  void UpdateChat(String msg)
  {
    for(int i = chat.Size() - 1; i > 0; i--)
	{
	  chat[i] = chat[i-1];
	  chatFade[i] = chatFade[i-1];
	}
	chat[0] = msg;
	chatFade[0] = 21;
  }
  
  static void ACS_UpdateChat(String msg)
  {
    let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(ev)
	{
      for(int i = ev.chat.Size() - 1; i > 0; i--)
	  {
	    ev.chat[i] = ev.chat[i-1];
	    ev.chatFade[i] = ev.chatFade[i-1];
	  }
	  ev.chat[0] = msg;
	  ev.chatFade[0] = 17;
	}
  }
  
  void UpdateItem(String msg)
  {
    for(int i = 0; i < item.Size(); i++)
	{
	  if(item[i] == "")
	  {
	    item[i] = msg;
		itemFade = 17.0;
		break;
	  }
	  else if(i == item.Size() - 1 && item[i] != "")
	  {
	    for(int j = 0; j < item.Size() - 1; j++)
		{
	      item[j] = item[j+1];
		}
		item[item.Size() - 1] = msg;
		itemFade = 12.0;
	  }
	}
  }

  override void WorldThingDamaged(WorldEvent e)
  {
    if(shield_disable)
	  return;
	  
    if(e.Thing && e.Thing is "DF_Monster" && DF_Monster(e.Thing).AP <= 0)
	{
	  let pmo = players[consoleplayer].mo;
	  if(pmo && e.DamageSource == pmo)
	  {
		int dmg = e.Thing.ApplyDamageFactor(e.DamageType, 100);
		if(dmg < 100)
		  SetShieldIcon();
		else if(dmg > 100)
		  SetHitMark("CritMark", 1.2);
	  }
	}
  }
  
  override void WorldThingDied(WorldEvent e)
  {
    if(players[consoleplayer].mo.health > 0 && e.Thing && e.Thing.bIsMonster)
	  SetHitMark("KillMark", 1.5);
	
	if(e.inflictor && (e.inflictor is "PistolPuff" || e.inflictor is "PiercePuff"))
	{
	  let pmo = DFPlayer(players[consoleplayer].mo);
	  if(pmo && DFPlayer(pmo).bBluePower)
	    pmo.CriticalSteal();
	}
	else if(e.inflictor && (e.inflictor is "ShotgunPuff" || e.inflictor is "ShockPuff"))
	{
	  let pmo = DFPlayer(players[consoleplayer].mo);
	  if(pmo && DFPlayer(pmo).bOrangePower)
	    pmo.AutoReload();
	}
  }
  
  override void PlayerDied(PlayerEvent e)
  {
    let pmo = players[consoleplayer].mo;
    if(!DFPlayer(pmo).cannotGameOver)
	{ 
      gameover = true;
	  gameoverBGAlpha = -1.0;
	  gameoverGFXAlpha = -2.0;
	  int cKey = Bindings.GetKeysForCommand("+Use");
	  gameoverKey = Bindings.NameKeys(cKey, 0);
	}
  }
  
  override void PlayerRespawned(PlayerEvent e)
  {
    gameover = false;
	gameoverBGAlpha = 0.0;
	gameoverGFXAlpha = 0.0;
  }
  
  override bool InputProcess(InputEvent e)
  {
    if(e.Type == InputEvent.Type_KeyDown)
	{
	  if(input_confirm)
	    SendNetworkEvent("DisableStartupPrompt");
	}
		
    return false;
  }
  
  override void WorldTick()
  {
    //TUTORIAL
	if(tip_time > 0)
	{
	  if(tip_alpha < 1.0)
	    tip_alpha += 0.1;
		
	  if(tip_alpha_flash > 0.0)
	    tip_alpha_flash -= 0.02;
		
	  tip_time--;
	}
	
    //HITMARKER
	if(hitAlpha > 0.0)
	{
	  if(hitAlpha <= 0.2)
	    hitAlpha -= 0.05;
	  else
	    hitAlpha -= 0.1;
	  
	  if(texName == "KillMark")
	    hitScale -= 0.025;
	}
	
	//MSG LOGS
	for(int i = 0; i < chat.Size(); i++)
	{
	  if(chatfade[i] > 0.1)
	    chatfade[i] -= 0.1;

	  if(chatfade[i] < 0.1)
	    chat[i] = "";
	}

	//ITEM LOGS
	if(itemFade ~== 0.1)
	{
	  itemFade -= 0.1;
	  for(int i = 0; i < item.Size(); i++)
	    item[i] = "";
	}
	else if(itemFade > 0.0)
	  itemFade -= 0.1;
	
	if(stimUpAlpha > 0.0)
	  stimUpAlpha -= 0.02;
	
	//Shield
	if(shieldAlpha > 0.0)
	  shieldAlpha -= 0.1;
	  
	//Armor integrity
	if(shieldBarAlpha > 0.0)
	  shieldBarAlpha -= 0.1;
	  
	//Blood Splatter
	if(spAlpha > 0.0)
	  spAlpha -= 0.01;
	  
	//PSI CAST
	if(PsiPunchTime > 0)
	{
	  PsiPunch = TexMan.CheckForTexture(String.Format("XLEFT%d", PsiPunchTimeMax - PsiPunchTime), TexMan.Type_Any);
	  PsiPunchTime--;
	}
	
	//ACTIVE SKILL
	if(skillTimeSec > 0)
	{
	  skillTimeSec--;
	  skillTimeMilliSec = skillTimeSec / 35.0;
	}
	
	//MUTATION PROGRESSION
	if(exp_alpha > 0.0)
	  exp_alpha -= 0.1;
	  
	if(exp_pad < 0.0)
	  exp_pad += 0.2;
	  
	//GAME OVER
	if(gameover)
	{
	  if(gameoverBGAlpha < 1.0)
	    gameoverBGAlpha += 0.05;
		
	  if(gameoverGFXAlpha < 1.0)
	    gameoverGFXAlpha += 0.05;
	}
	
	//PUNCH
	if(FSATime < FSAMax)
	{
	  FSATex = TexMan.CheckForTexture(String.Format("%s%d", FSATexName, FSATime), TexMan.Type_Any);
	  FSATime++;
	}
	
	if(mthClose)
	{
	  if(mthDist < 100.0)
	    mthDist += 20.0;
	  else
	  {
	    mthClose = false;
		mthBiteTime = 12;
	  }
	}
	
	if(mthBiteTime > 0)
	{
	  if(mthBiteTime == 1)
	  {
	    mthOpen = true;
		mthDist = 1.0;
	  }
	  
	  mthBiteTime--;
	}
	
	if(mthOpen)
	{
	  if(mthDist < 144)
	    mthDist *= 1.5;
	  else
	    mthOpen = false;
	}
	
	//INV GRID
	if(blockInvGridTic > 0)
	  blockInvGridTic--;
	  
	//KEY UPDATE
	if(updateKeyTic == 1)
	{
	  EventHandler.SendInterfaceEvent(0, "HUDUpdateKeys");
	  updateKeyTic--;
	}
	else if(updateKeyTic > 0)
	  updateKeyTic--;
	  
	//Stamina Warning
	if(warn_staminaF > 0.0)
	  warn_staminaF -= 0.1;
	  
	//Retaliation Warning
	if(warn_retalF > 0.0)
	  warn_retalF -= 0.1;
  }

  override void NetworkProcess (ConsoleEvent e)
  {
	Array<string> command;
	e.Name.Split (command, ":");
	string cmd = command[0];
	
	let pmo = players[consoleplayer].mo;
	if(cmd ~== "SwitchFireMode")
	{
	  let psp = players[consoleplayer].GetPSprite(PSP_WEAPON);
	  let wep = players[consoleplayer].ReadyWeapon;
	  if(wep && psp.CurState.InStateSequence(wep.ResolveState("Idle")))
		DF_Weapon(wep).TryChangeAmmo();
	}
	else if(cmd ~== "UseWithoutPickup")
	{
	  if(pmo && DFPlayer(pmo).item_focus)
	  {
	    let actr = DFPlayer(pmo).item_focus;
		blockInvGridTic = 2; //2 tic for safety
		if(actr is "CustomInventoryItem")
		{
		  if(actr is "Stim" || actr is "KeyItem")
		    return;
		
	      if(pmo.GiveInventory(actr.GetClassName(), 1)) //We don't have this item
		  {
		    let inv = pmo.FindInventory(actr.GetClassName());
		    if(inv)
		      pmo.UseInventory(inv);
				
		    actr.Destroy();
		  }
	      else if (!pmo.GiveInventory(actr.GetClassName(), 1) && pmo.CheckInventory(actr.GetClassName(), 1)) //We don't have room but we do have this item type
	      {
		    let inv = pmo.FindInventory(actr.GetClassName());
		    if(inv)
		      pmo.UseInventory(inv);
				
		    pmo.GiveInventory(actr.GetClassName(), 1);
			  
		    actr.Destroy();
	      }
	      else	//we don't have this item or have room for this item
	      {
		    CustomInventoryItem(actr).useFromFloor = true;
		    CustomInventoryItem(actr).CallTryPickup(pmo);
		  
		    let inv = pmo.FindInventory(actr.GetClassName());
		    if(inv)
		      pmo.UseInventory(inv);
	      }
		}
	  }
	  else if(pmo && DFPlayer(pmo).eatActr)
	  {
	    let wep = players[consoleplayer].ReadyWeapon;
		if(!level.isFrozen() && wep)
		  players[consoleplayer].SetPSprite(10, wep.ResolveState("MouthBite"));	
	  }
	}
	else if(cmd ~== "RepairItem") 
	{
	  RepairItem(command[1], command[2]);
	}
	else if(cmd ~== "QuickUse1")
	{
	  let itm = DFPlayer(pmo).quickUse1;
	  if(itm && itm.amount > 0)
	    pmo.UseInventory(itm);
	}
	else if(cmd ~== "QuickUse2")
	{
	  let itm = DFPlayer(pmo).quickUse2;
	  if(itm && itm.amount > 0)
	    pmo.UseInventory(itm);
	}
	else if(cmd ~== "QuickUse3")
	{
	  let itm = DFPlayer(pmo).quickUse3;
	  if(itm && itm.amount > 0)
	    pmo.UseInventory(itm);
	}
	else if(cmd ~== "QuickUse4")
	{
	  let itm = DFPlayer(pmo).quickUse4;
	  if(itm && itm.amount > 0)
	    pmo.UseInventory(itm);
	}
	else if(cmd ~== "FastHeal")
	{
	  if(pmo && DFPlayer(pmo).fHealCount > 0)
	  {
	    let psp = players[consoleplayer].FindPSprite(PSP_Weapon);
		let wep = players[consoleplayer].ReadyWeapon;
		if(psp && wep)
		{
		  if(DFPlayer(pmo).bBase_Wit && !psp.CurState.InStateSequence(wep.ResolveState("FastHeal")))
		    DF_Weapon(wep).TryFastHeal();	
		  else if(psp.CurState.InStateSequence(wep.ResolveState("Idle")))
		    DF_Weapon(wep).TryFastHeal();
		}
	  }
	}
	else if (cmd ~== "MapActivate")
	{
	  if(pmo && pmo.Health > 0)
		Menu.SetMenu("MapMenu");
	}
	else if (cmd ~== "LoadActivate")
	{
		Menu.SetMenu("DFLoadMenu");
	}
	else if(cmd ~== "CleanNote")
	{
	  if(pmo)
	  {
	    let itm = DFPlayer(pmo).note;
		pmo.A_GiveInventory(NoteItem(itm).noteInv, 1);
		DFPlayer(pmo).note = null;
		itm.Destroy();
	  }
	}
	else if(cmd ~== "ClearPicture")
	{
	  if(pmo)
	  {
        DFPlayer(pmo).pictureStr = "";
	    DFPlayer(pmo).pictureItemStr = "";
	  }
	}
	else if(cmd ~== "GiveItem")
	{
	  if(pmo)
	  {
        pmo.A_GiveInventory(command[1], 1);
	  }
	}
	else if(cmd ~== "SetAnimation")
	{
	  SetFSAnimation(command[1], command[2].ToInt(10), command[3].ToInt(10), command[4].ToInt(10), command[5].ToDouble());
	  if(pmo)
	    pmo.A_SetBlend("White", 0.3, 75);
	}
	else if(cmd ~== "UpdateKeys")
	  updateKeyTic = 2;
	else if(cmd ~== "DisableStartupPrompt")
	  input_confirm = false;
  }
  
  override void InterfaceProcess(ConsoleEvent e)
  {
    Array<string> command;
	e.Name.Split (command, ":"); 
	
	if(command[0] ~== "HUDUpdateFireMode")
	{
	  DFStatusBar(StatusBar).fireMode = "Fire Mode: " .. command[1];
	  DFStatusBar(StatusBar).fireModeAlpha = 5.0;
	}
    else if(e.name ~== "HUDUpdateKeys")
	{
	  DFStatusBar(StatusBar).UpdateHotkeys(); 
	}
	else if(command[0] ~== "HUDWeaponBar")
	{
	  int slot = command[1].ToInt(10);
	  int xofs = command[2].ToInt(10);
	  if(slot > 0 && slot < 5)
	  {
	    if(DFStatusBar(StatusBar).wpnBarAlpha <= 0.0)
		  DFStatusBar(StatusBar).wpnBarY = 12;
		
		DFStatusBar(StatusBar).wpnBarAlpha = 8.0;
	    DFStatusBar(StatusBar).wpnSelAlpha = 1.0;
	    DFStatusBar(StatusBar).wpnSelX = -480 + (192 * slot);
		
		if(xofs > 0)
		{
		  DFStatusBar(StatusBar).wpnPopUpX = xofs;
		  DFStatusBar(StatusBar).wpnPopUpY = -138;
		}
		else
		{
		  DFStatusBar(StatusBar).wpnPopUpX = xofs;
		  DFStatusBar(StatusBar).wpnPopUpY = -66;
		}
	  }
	}
	else if(e.name ~== "PortraitHealFlash")
	  DFStatusBar(StatusBar).face_heal_alpha = 1.0;
    else if(e.name ~== "ResetPhyExp")
	  DFStatusBar(StatusBar).expIntp.Reset(0);
	else if(e.name ~== "ItemSpaceUpdate" && blockInvGridTic <= 0)
	{
	  if(DFStatusBar(StatusBar).item_slot_alpha <= 0.0)
	  {
	    DFStatusBar(StatusBar).itemSlotScaleInc = 0.04;
		DFStatusBar(StatusBar).itemSlotScale = 1.5;
	  }
	  else
	  {
	    DFStatusBar(StatusBar).itemSlotScaleInc = 0.08;
		DFStatusBar(StatusBar).itemSlotScale = 1.4;
	  }
	  DFStatusBar(StatusBar).item_slot_alpha = 4.4;
	}
  }
