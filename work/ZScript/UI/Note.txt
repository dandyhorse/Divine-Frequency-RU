Class NoteItem : Actor
{
  bool pickup;
  string noteInv, noteSnd, content;
  
  property CanPickup: pickup;
  property NoteInv: noteInv;
  Property ReadSound: noteSnd;
  property Text: content;
  
  override bool Used(Actor user)
  {
    if(user is "DFPlayer")
	{
	  DFPlayer(user).note = self;
	  user.A_StartSound(self.noteSnd, CHAN_AUTO, CHANF_UI);
	  Menu.SetMenu("NoteReaderMenu");
	  return true;
	}
	
	return false;
  }
  
  Default
  {
    //$Category "Item/Notes"
	//$Color 5
    NoteItem.CanPickup False;
	NoteItem.ReadSound "Note";
  }
}

Class NoteReaderMenu : UIGenericMenu
{
  UIBoxTextures sbFG, sbBG, sbHov;
  UIScrollContainer nScroll;
  UILabel nContent, nHelp;
  bool canTake;
  String text;
  double tAlpha;
  int xofs;
  Font fnt;
  UIImage noteBG;
  
  string getKey(string keybind) 
  {
    Array<int> keyInts;
    Bindings.GetAllKeysForCommand(keyInts, keybind);
    return Bindings.NameAllKeys(keyInts);
  }
  
  override void Init(Menu parent)
  {
    Super.Init(parent);
	
	PlayerPawn pmo = players[consoleplayer].mo;
	let itm = DFPlayer(pmo).note;
	text = NoteItem(itm).content;
	canTake = NoteItem(itm).pickup;
	
	fnt = Font.GetFont("RU_DF_FONT");
	tAlpha = 0.0;
	xofs = 550;
	
	//MenuSound("Note");
	SetBaseResolution((720, 400));
	DrawImageBackground();
	
	sbFG = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/ScrollFG.png", false);
	sbBG = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/ScrollBG.png", false);
	sbHov = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/ScrollHover.png", false);
	
	nContent = new('UILabel').Create
	(
	  (0, 0),
	  (0, 0),
	  text: "",
	  fnt: fnt,
	  alignment: UILabel.AlignType_TopLeft,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.45
	);
	nContent.setAlpha(tAlpha);
	nContent.Pack(mainframe);
	
	String infoContent = text;   
	if(infoContent.Left(1) ~== "$")
	  infoContent = StringTable.Localize(text);
	  
	nContent.setText(infoContent);
		  
	BrokenLines Lines = fnt.BreakLines(infoContent, int(300 * 2.0));
	double sAreaHeight = Lines.Count() * (fnt.GetHeight() * 0.45);
	nScroll = new("UIScrollContainer").Create(
	(xofs, 64),
	(245, 300),
	4, sAreaHeight, 12.0,
	sbFG,
	scrollBarHover: sbHov,
	scrollBarClick: sbFG,
	scrollBg: sbBG,
	scrollArea: nContent
	);
	nScroll.Pack(mainframe);
	
	String nHelpText = String.Format("\cD[Esc] \cJLeave");
	
	if(canTake)
	{
	  Array<String> keyList;
	  getKey("+Use").Split(keyList, "\cM, ");
	  nHelpText = String.Format("\cD[%s] \cJPick up | \cD[Esc] \cJLeave", keyList[0]);
	}
	
	nHelp  = new('UILabel').Create
	(
	  (470, 364),
	  (245, 36),
	  text: nHelpText,
	  fnt: fnt,
	  alignment: UILabel.AlignType_TopLeft,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.45
	);
	nHelp.Pack(mainframe);
  }
  
  override bool OnUIEvent(UIEvent ev)
  {
	if(ev.Type == UIEvent.Type_KeyDown)
	{
	  if(!canTake)
	    return false;
		
	  Array<String> keyList;
	  getKey("+Use").Split(keyList, "\cM, "); // -________________________-
	  
	  for(int i = 0; i < keyList.Size(); i++)
	    if(ev.KeyString == keyList[i])
		{
		  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		  if(ev)
		    ev.SendNetworkEvent("CleanNote"); 
		  
		  MenuSound("Note/Up");
		  Close();
		}
	}
	
	if(ev.type == UIEvent.Type_RButtonDown)
	  Close();
	
	return Super.OnUIEvent(ev);
  }
  
  override void Ticker()
  {
	Super.Ticker();
	
	if(xofs > 470)
	  if(nContent && nScroll)
	  {
	    xofs -= 10;
	    nScroll.setPosX(xofs);
		
		if(tAlpha < 1.0)
		{
		  tAlpha += 0.1;
		  nContent.setAlpha(tAlpha);
		}
	  }
  }
  
  void DrawImageBackground()
  {
	UIImage bg = UIImage(new("UIImage")).Create((450, 0), (270, 400), image: "Graphics/NoteReadBG.png");
	bg.Pack(mainframe);
  }
}

Class PictureMenu : UIGenericMenu
{
  string pic, pic_item;
  double bg, rotation, xofs, yofs, factor, factor_extra;
  Font fnt;
  bool canTake;
  
  string getKey(string keybind) 
  {
    Array<int> keyInts;
    Bindings.GetAllKeysForCommand(keyInts, keybind);
    return Bindings.NameAllKeys(keyInts);
  }
  
  override void Init(Menu parent)
  {
    Super.Init(parent);
	
	PlayerPawn pmo = players[consoleplayer].mo;
	if(pmo)
	{
	  pic = DFPlayer(pmo).pictureStr;
	  if(DFPlayer(pmo).pictureItemStr != "")
	  {
	    pic_item = DFPlayer(pmo).pictureItemStr;
	    canTake = true;
	  }
	}
	
	rotation = 8.0;
	bg = 0.4;
	xofs = 152.0;
	yofs = 116.0;
	factor = 0.234;
	
	String nHelpText = String.Format("\cD[Esc] \cJLeave");
	if(pic_item != "")
	{
	  Array<String> keyList;
	  GetKey("+Use").Split(keyList, "\cM, ");
	  nHelpText = String.Format("\cD[%s] \cJPick up | \cD[Esc] \cJLeave", keyList[0]);
	}
	
	MenuSound("Note");
	SetBaseResolution((720, 400));
	fnt = Font.GetFont("RU_DF_FONT");
	UILabel nHelp = new('UILabel').Create
	(
	  (516, 385),
	  (200, 36),
	  text: nHelpText,
	  fnt: fnt,
	  alignment: UILabel.AlignType_CenterRight,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.45
	);
	nHelp.Pack(mainframe);
  }
  
  override void handleBack()
  {
    Eventhandler.SendNetworkEvent("ClearPicture");
    Super.handleBack();
  }
  
  override bool OnUIEvent(UIEvent ev)
  {
	if(ev.Type == UIEvent.Type_KeyDown)
	{
	  if(!canTake)
	    return false;
		
	  Array<String> keyList;
	  getKey("+Use").Split(keyList, "\cM, "); // -________________________-
	  
	  for(int i = 0; i < keyList.Size(); i++)
	    if(ev.KeyString == keyList[i])
		{
		  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		  if(ev)
		    ev.SendNetworkEvent(String.Format("GiveItem:%s", pic_item)); 
		  
		  MenuSound("Note/Up");
		  Close();
		}
	}
	
	if(ev.type == UIEvent.Type_RButtonDown)
	  Close();
	
	return Super.OnUIEvent(ev);
  }
  
  override void Drawer()
  {
    Screen.Dim(Color(0, 0, 0), bg, 0, 0, Screen.GetWidth(), Screen.GetHeight());
	
    Super.Drawer();
	
	Screen.DrawTexture(TexMan.CheckForTexture(pic), false, xofs, yofs,
	DTA_ScaleX, factor,
	DTA_ScaleY, factor,
	DTA_KeepRatio, true,
	DTA_320x200, true,
	DTA_CenterOffset, true,
	DTA_Alpha, 0.7,
	DTA_Rotate, rotation);
  }
  
  override void Ticker()
  {
	Super.Ticker();
	
	if(bg < 0.9)
	  bg += 0.05;
	  
	if(rotation > 0)
	  rotation -= 1.0;
	  
	if(rotation <= 2 && factor < 0.24)
	  factor += 0.0015;

	if(yofs > 100)
	  yofs -= 2.0;
	  
	if(xofs < 160)
	  xofs += 1.0;
  }
}
