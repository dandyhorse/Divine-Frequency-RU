class DamageNumHandler : StaticEventHandler 
{
  Vector3 SpawnPos, oldPos, newPos;
	
  override void WorldThingDamaged(WorldEvent e)
  {
	if(Cvar.GetCVar("damageNum_enable", players[consoleplayer]).GetInt() > 0 && e.Thing != players[consoleplayer].mo && e.Thing.bIsMonster)
	{
	  if(e.DamageType == "ShotGib" || e.DamageType == "Void" || e.DamageType == "SGShock")
	    return;
		
	  spawnPos = e.Thing.pos + (0, 0, e.Thing.height-16);
	  Actor dam = e.Thing.Spawn("DamNum", spawnPos);
			
	  if(dam)
	  {
		DamNum(dam).damage = e.Damage;
				
		if(e.damagetype != "None")
		  DamNum(dam).color = Stringtable.Localize("$" .. e.damagetype);
		else
		  DamNum(dam).color = "White";
	  }
    }
  }
	
  override void renderOverlay(RenderEvent e) 
  {
    if(automapactive)
	  return;
        
    GMMatrix worldToClip = GMMatrix.worldToClip(e.viewPos, e.viewAngle, e.viewPitch, e.viewRoll, players[consoleplayer].FOV, Screen.GetAspectRatio());
    Actor DmgActor;
    ThinkerIterator iter = ThinkerIterator.create("Actor", Thinker.STAT_DEFAULT);
    while((DmgActor = Actor(iter.next())) != Null) 
	{
	  if(DmgActor is "DamNum")
	  {
		Vector3 adjustedWorldPos = e.viewPos + LevelLocals.vec3Diff(e.viewPos, GMVectorUtil.lerpVec3(DmgActor.Prev, DmgActor.Pos, e.Fractic));
		Vector3 ndcPos = worldToClip.multiplyVector3(adjustedWorldPos + (0, 0, DmgActor.height / 2.0)).asVector3();
		if(ndcPos.z <= 1 && ndcPos.z >= -1)
		{
		  Font fnt = "RU_DF_FONT";
		  String damage = String.Format("%d", DamNum(DmgActor).damage);
		  String color = DamNum(DmgActor).color;
		  double alpha = DmgActor.Alpha;
		  double fs = clamp(1.8 - (players[consoleplayer].mo.Distance3D(DmgActor) * 0.001), 0.8, 1.8);

		  Vector2 drawPos = GMGlobalMaths.ndcToViewPort(ndcPos);
		  Screen.DrawText(fnt, Font.FindFontColor(color), drawPos.x, drawPos.y, damage,
		  DTA_KeepRatio, true,
		  DTA_LegacyRenderStyle, STYLE_ADD, 
		  DTA_Alpha, alpha,
		  DTA_ScaleX, fs,
		  DTA_ScaleY, fs * 1.1);
		}
	  }
    }
  }
}

// Based on DamNum by Xaser Acheron

Class DamNum : Actor
{
  String color;
  int damage;
  bool isCrit;
  
  override void BeginPlay()
  {
    Super.BeginPlay();
	
	isCrit = false;
    alpha = frandom(0.8, 1.0);
	double velocity = frandom(1.0, 2.0);
	int ang = random(0,359);
    vel = (velocity * cos(ang), velocity * sin(ang), frandom(3.0, 4.0));
  }
  
  Default
  {
    Gravity 0.4;
	+NOINTERACTION
  }
  States
  {
    Spawn:
      TNT1 A 1
	  {
	    A_FadeOut(0.035);
	    vel.z -= Gravity;
	  }
	  Wait;
  }
}
