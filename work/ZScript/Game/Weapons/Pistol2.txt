  override void AttachToOwner(Actor other)
  {
    if(other)

	  DFPlayer(other).wep2 = true;

	    hasAlt = true;
	}
	  
	Super.AttachToOwner(other);
  }

  action void A_PSIInsect()
  {
    FTranslatedLineTarget t;
    LineAttack(angle, 2048, pitch, 10, "Normal", "PistolAltShot", LAF_ISMELEEATTACK, t);
	if(t.linetarget && t.linetarget.bIsMonster && !DF_Monster(t.linetarget).bPossessImmune && !DF_Monster(t.linetarget).bPossessed)
	{
	  t.linetarget.A_GiveInventory("SwarmToken", 1);
	  let actr = Spawn("InsectSwarm", t.linetarget.pos + (0,0,t.linetarget.height * 0.5));
	  if(actr)
	  {
	    InsectSwarm(actr).zAnchor = t.linetarget.height * 0.5;
		actr.master = t.linetarget;
	  }
	}
  }
  
  action void A_FireHandgun()
  {
	if(!invoker.upAcc && invoker.isSpraying)
	{
	  invoker.sXY = 7;
	  invoker.sZ = 7;
	}
	else
	{
	  invoker.sXY = 1;
	  invoker.sZ = 1;
	}
	
	if(invoker.wep_fireMode == invoker.wep_primary)
	{
	  int damg = invoker.dmg;
	  if(invoker.wep_cond > 33 && invoker.wep_cond < 82)
	  {
        double damgModf = (82 - invoker.wep_cond) * 0.31;
        damg = invoker.dmg - floor(damgModf);
	  }
	  else if(invoker.wep_cond <= 33)
	    damg = 35;
	  
	  A_RailAttack(damg, 0, 0, "", "", RGF_SILENT, 0, "PistolPuff", invoker.sXY, invoker.sZ);
	  if(DFPlayer(self).bPhy_Gland)
	    A_RailAttack(0, 0, 0, "", "", RGF_SILENT, 0, "PoisonPuff", invoker.sXY, invoker.sZ);
	}
	else
	{
	  A_FireBullets(invoker.sXY, invoker.sZ, -1, 10, "PiercePuff", FBF_NORANDOM);
	}
	
	invoker.Degrade(33);
	A_StartSound(invoker.snd, CHAN_WEAPON);

	if(invoker.Ammo1.Amount == 1)
	  A_StartSound("NA_Pist", CHAN_AUTO, volume: 0.6);
	
	A_AlertMonsters();
	A_TakeInventory("BulletLoaded",1);
	
	invoker.recoilControl = DFPlayer(invoker.owner).Str;
	if(invoker.upAcc)
	{
	  invoker.recoilControl = clamp(invoker.recoilControl + 3, 3, 4);
	}
	if(invoker.wep_AmmoType == 0)
	  A_SpawnCasing("PistolCasing", (51, 28, -7), (random(-1,0),random(5,6),random(6,7)));
	  
	if(!DFPlayer(self).bRecoilPower && invoker.recoilControl < 4)
	{
	  double pitch_recoil = 20;
	  double push_recoil = 8;
	  if(invoker.wep_AmmoType == 1)
	  {
	    pitch_recoil += 20 - (invoker.recoilControl * 5);
	    push_recoil += 32 - (invoker.recoilControl * 5);
	  }
	  
	  A_QuakeEx(2, 2, 3, 5, 0, 256, "DSEMPTY");
	  A_SetPitch(pitch - (pitch_recoil - (invoker.recoilControl * 5)), SPF_INTERPOLATE);
	  A_ZoomFactor(1.03, ZOOM_INSTANT);
	  A_Recoil(push_recoil - (invoker.recoilControl * 2));
	  
	  if(invoker.recoilControl < 2)
	  {
	    double angAdd = 12 - (invoker.recoilControl * 2);
	    A_SetAngle(angle + random(-angAdd, angAdd), SPF_INTERPOLATE);
	  }
	}
	else
	{
	  A_QuakeEx(1, 1, 2, 5, 0, 256, "DSEMPTY");
	}
  }

Class HandgunReloader : Inventory
{
  Weapon wep;
  int loaded, magsize;
  
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	if(owner)
	{
	  wep = Handgun(owner.FindInventory("Handgun"));
	  if(wep)
	  {
	     loaded = wep.Ammo1.Amount;
		 if(loaded <= 0)
		   magsize = wep.Ammo1.MaxAmount - 1;
		 else
		   magsize = wep.Ammo1.MaxAmount;
	  }
	}
  }

  override void DoEffect()
  {
    Super.DoEffect();
	
	if(owner && wep)
	{
	  if(wep.Ammo2.Amount <= 0) //Out of bullets, stop
	    Destroy();
	
	  if(loaded >= magsize) //Mag full, stop
	    Destroy();
	  
	  owner.A_GiveInventory("BulletLoaded", 1);
	  if(owner && !DFPlayer(owner).bDuel_Buff)
	    owner.A_TakeInventory(wep.Ammo2.GetClassName(), 1);
		
	  loaded++;
	}
  }
  
  Default
  {
    Inventory.Amount 1;
	Inventory.MaxAmount 1;
  }
}

Class PierceAmmo : Ammo
{
  bool enable_swap;
  override bool HandlePickup(Inventory item)
  {
    bool result = Super.HandlePickup(item);
    if(result && !enable_swap)
    {
	  if(owner)
	  {
	    let inv = Handgun(owner.FindInventory("Handgun"));
		if(inv)
		{
	      inv.wep_have_swap = true;
		  enable_swap = true;
		}
	  }
    }
    return result;
  }
  Default
  {
	Inventory.Amount 1;
	Inventory.MaxAmount 30;
	Inventory.Icon "WEP2B";
  }
}

Class SwarmToken : Inventory
{
  override void PostBeginPlay()
  {
    if(owner && owner is "DF_Boss")
	  Destroy();
	  
	if(owner && owner is "DF_Monster" && !DF_Monster(owner).bPossessImmune)
	{
	  if(!owner.bFriendly && !DF_Monster(owner).bPossessed)
	  {
	    owner.bFriendly = true;
		DF_Monster(owner).bPossessed = true;
		BlockThingsIterator it = BlockThingsIterator.Create(owner, 1024);
		while(it.Next())
		{
		  Actor mo = it.thing;
	  
		  if(mo == owner)
			continue;
      
		  if(!mo.bShootable || mo.bFriendly)
			continue;
		
		  if(mo.Health <= 0)
			continue;
			
		  if(mo is "DF_Boss")
		    continue;
			
		  if(mo is "DF_Monster" && mo.CheckSight(owner))
		  {
		    if(mo.target == null || (mo.target && mo.target == players[consoleplayer].mo))
			  mo.target = owner;
		  }
	    }
	  }
	  else
	    Destroy();
	}
	else
	  Destroy();
  }
  
  override void DoEffect()
  {
    if(owner && GetAge() % 20 == 0)
	  owner.A_DamageSelf(1, "Sting");
	
	if(GetAge() >= 2096)
	{
	  if(owner)
	  {
	    owner.bFriendly = false;
		DF_Monster(owner).bPossessed = false;
	  }
		
	  Destroy();
	}
	  
    Super.DoEffect();
  }
  
  Default
  {
    Inventory.MaxAmount 1;
  }
}

Class SwarmUnit : Actor
{
  bool Left, Up;
  int xAnchor, zAnchor, angleRate, pitchOsc;
  
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	A_SetScale(frandom(1.0,1.5));
	
	if(master && zAnchor > master.height * 0.4)
	  Up = random(0,1);
	  
	if(Up)
	  pitchOsc = random(1,4);
  }
  
  override void Tick()
  {
    Super.Tick();
	
	//A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("INSECT"), STYLE_NONE, 0, 10, 6);
	if(GetAge() >= 2096 || !master || (master && master.Health <= 0))
	  Destroy();
	  
	A_Warp(AAPTR_MASTER, xAnchor, 0, zAnchor, angle, WARPF_ABSOLUTEANGLE|WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE);
	angle += Left ? angleRate : -angleRate;
	if(abs(angle) > 360)
	  angle -= Left ? 360.0 : -360.0;
	  
	if(Up)
	  zAnchor += (sin(angle)) * pitchOsc;
  }
  
  Default
  {
    //RenderStyle "AddStencil";
	//StencilColor "846e00";
	+BRIGHT
    +NOINTERACTION
  }
  States
  {
    Spawn:
	  BUGF AB 1;
	  Loop;
  }
}
