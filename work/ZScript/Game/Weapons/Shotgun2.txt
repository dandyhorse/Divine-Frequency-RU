  override void AttachToOwner(Actor other)
  {
    if(other)
	{
	  DFPlayer(other).wep3a = true;
	  
	  if(DFPlayer(other).bPsi_SG)
	    hasAlt = true;
	}
	  
	Super.AttachToOwner(other);
  }
  
  override void DetachFromOwner()
  {
    if(owner)
	  DFPlayer(owner).wep3a = false;
	
    Super.DetachFromOwner();
  }
  
  override void BeginPlay()

  override void DoEffect()
  {
    Super.DoEffect();
	
	if(burst_timer > 0)
	{
	  burst_timer--;
	}
	else if(burst_timer == 1)
	{
	  burst_mult = 1.0;
	  burst_timer--;
	}
  }
  
  void AddCharge()
  {
    if(EnhancedShot < 3)
	  EnhancedShot++;
  }
  
  action void A_PSIShock(Actor targ)
  {
    if(targ)
	{
	  invoker.AddCharge();
	  //GiveInventory("BaseArmor", 15);
	  A_SetBlend("Cyan", 0.1, 15);
	  if(DFPlayer(self).armorStack < DFPlayer(self).armorMax)
	  {
		DFPlayer(self).armorStack++;
	    GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		if(ev)
		  ev.UpdateChat("\cD+1 Armor Stack.");
	  }	
      targ.A_GiveInventory("DamageTrackerSGSP", damage);
	  targ.A_StartSound("ElecHit1", CHAN_AUTO, volume: 1.0, attenuation: 0.4);
	  targ.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("EXPFC0"), STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE|SPF_LOCAL_ANIM, 5, 300, zoff: targ.height * 0.5, startalphaf: 0.05, startroll: random(0,359));
	  for(int i = 2; i < 5; i++)
	  {
	    bool style = random(0,1);
	    targ.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture(String.Format("SPARK%d", i)), style ? STYLE_Translucent : STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE|SPF_ROLL|SPF_LOCAL_ANIM, 23, random(366,439), 0,
		zoff: targ.height * 0.5, startalphaf: 0.8, fadestepf: 0.0, startroll: random(0,359), rollvel: frandom(-0.5,0.5));
	  }
	  DF_Monster(targ).bStunned = true;
	  DF_Monster(targ).bStunSoft = true;
	  DF_Monster(targ).StunTime = 35*2;
	  targ.A_DamageSelf(random(1, 10), "Stun");
	  BlockThingsIterator it = BlockThingsIterator.Create(self, 1280); //Credit: Boondorl, for the whole thing
	  
	  while(it.Next())
	  {
		if(it.thing && it.thing is "DF_Monster" && it.thing.bIsMonster && it.thing.health > 0 && Distance3DSquared(it.thing) <= 1638400) //Is this monster alive and in range?
		  if(!it.thing.bDormant && it.thing.bShootable && !it.thing.bFriendly) //Can we damage this entity? should we?
		    if(CheckSight(it.thing, SF_IGNOREVISIBILITY|SF_SEEPASTSHOOTABLELINES|SF_IGNOREWATERBOUNDARY)) //Can we even see it?
			  if(!DF_Monster(it.thing).bStunned) //is it already stunned?
		      {
			    Actor msl = tracer.Spawn("StunBall", targ.pos + (0, 0, DF_Monster(targ).sHeight * 0.6));
			    if(msl)
				{
				  StunBall(msl).charge = 1;
				  msl.tracer = it.thing;
				}	
				break;
			  }
	  }
	}
  }

  action void A_FireShotgun2()
  {  
    invoker.wVol = 1.0;
	A_AlertMonsters();
	A_Overlay(PSP_FLASH, "Flash");
	A_TakeInventory("ShellLoaded",1);
	
	if(invoker.AltFireMode)
	{
	  if(DFPlayer(self).bOrangePower)
	  {
	    if(invoker.burst_timer <= 0)
		  invoker.burst_mult = 1.0;
		else
		  invoker.burst_mult += 0.25;
		  
		invoker.burst_timer = 70;
	  }
	  
	  if(DFPlayer(self).Str < 2)
	  {
	    int hitChance = random(0,100);
	    if(hitChance < 15 - (DFPlayer(self).Str * 5))
	    {
	      A_DamageSelf(1, "Drowning");
	      GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	      if(ev)
	        ev.UpdateChat(String.Format("· %s \cJsmacked themselves in the face for 1 damage.", self.GetTag()));
	    }
	  }
	}
	
	double spXY = 32 - (0.22 * invoker.wep_cond); //10 min
	double spZ = 10 - (0.05 * invoker.wep_cond); //5 min
	
	if(invoker.firstPickup || invoker.EnhancedShot > 0)
	{
	  int damg = 9 * random(7,8);
	  if(invoker.hasAtom)
	    damg = 72;
		
	  invoker.EnhancedShot--;
	  A_FireBullets(spXY, spZ, 8, damg * invoker.burst_mult, "ShockPuff", FBF_NORANDOM);
	  A_FireBullets(0, 0, 1, 0, "ShockPuffLayer");
	  A_StartSound("SHT_FIR2", CHAN_AUTO);
	}
	else
	{
	  for(int i = 0; i < 8; i++)
	  {
	    FTranslatedLineTarget t;
	    double pangle = angle;
	    double slope = BulletSlope();
		int damg = 8 * random(3,4);
		if(invoker.hasAtom)
		  damg = 32;
		  
	    pangle += spXY * Random2[cabullet]() / 255.;
	    slope += spZ * Random2[cabullet]() / 255.;
		if(DFPlayer(self).bPhy_Gland)
		  LineAttack(pangle, PLAYERMISSILERANGE, slope, random(6,8), "Acid", "ShotgunPuff", 0, t);
		  
	    let puff = LineAttack(pangle, PLAYERMISSILERANGE, slope, damg * invoker.burst_mult, "Hitscan", "ShotgunPuff", 0, t);
		if(t.linetarget && t.linetarget is "DF_Monster" && t.linetarget.health <= 0)
		{
		  invoker.wVol = clamp(0.6 + (Distance2D(t.linetarget) * 0.001), 0.6, 1.0);
		}
		
		if(!DFPlayer(self).bS_Up1)
		{
	      if(t.linetarget && t.linetarget is "DF_Monster" && !(t.linetarget is "DF_Boss"))
	        if(DF_Monster(t.linetarget).Armored && !DFPlayer(self).bOrangePower)
              invoker.wVol = 0.65;
		}
	  }
	  if(!invoker.AltFireMode)
	    A_StartSound(invoker.def_fire_snd, CHAN_AUTO, volume: invoker.wVol);
	  else
	    A_StartSound("SHT_FIR4", CHAN_AUTO, volume: 0.5);
	}
	invoker.Degrade(40, false);
	
	int str = DFPlayer(self).Str;
	if(DFPlayer(self).bS_Up2)
	  str += 2;
	  
	if(!DFPlayer(self).bRecoilPower && str < 4)
	{
	  if(str < 2)
	    A_QuakeEx(3, 3, 4, 10, 0, 256, "DSEMPTY");
	  else
	    A_QuakeEx(2, 2, 3, 10, 0, 256, "DSEMPTY");
		
	  if(invoker.SFMult > 1.0)
	  {
	    double angAdd = (18 - (str * 4)) * invoker.SFMult;
		double pit = (28 - (str * 6)) * invoker.SFMult;
	    A_SetAngle(angle + random(-angAdd, angAdd), SPF_INTERPOLATE);
		A_SetPitch(pitch - pit, SPF_INTERPOLATE);
	  }
	  else
	  {
	    A_SetAngle(angle - random(0, 16 - (str * 4)), SPF_INTERPOLATE);
		A_SetPitch(pitch - (32 - (str * 8)), SPF_INTERPOLATE);
	  }
	  A_ZoomFactor(1.2, ZOOM_INSTANT);
	  A_Recoil(12 - (str * 3));
	}
	else
	  A_QuakeEx(1, 1, 2, 8, 0, 256, "DSEMPTY");
  }
  
  action void A_TryReload()
  {
    let ammo = FindInventory("ShotgunAmmo");
	let loaded = FindInventory("ShellLoaded");
	if((ammo && ammo.amount > 0) && loaded)
	{
      int load = 1;
	
      if(DFPlayer(self).bS_Up2 && ammo.amount > 1 && loaded.amount <= loaded.MaxAmount - 2)
	    load = 2;
	  
	  A_TakeInventory("ShotgunAmmo", load);
	  A_GiveInventory("ShellLoaded", load);
	}
  }
  
  void PreFire()
  {
    if(owner)
	{
      for(int i = 0; i < 8; i++)
	  {
	    FLineTraceData t;
	    double pangle = owner.angle;
	    double slope = owner.BulletSlope();
	    double spXY = 32 - (0.22 * wep_cond); //10 min
	    double spZ = 10 - (0.05 * wep_cond); //5 min
	    pangle += spXY * Random2[cabullet]() / 255.;
	    slope += spZ * Random2[cabullet]() / 255.;
	    bool hit = owner.LineTrace(pangle, PLAYERMISSILERANGE, slope, 0, 33, data: t);
		if(hit && t.HitActor && t.HitActor is "DF_Boss")
		{
		  if(DF_Monster(t.HitActor).parry_time > 0)
		    owner.player.mo.FireWeapon(FindState("OldFire"));
			break;
		}
	    else if(hit && t.HitActor && t.HitActor is "Stars" && (Stars(t.HitActor).phase1 || Stars(t.HitActor).phase3))
	    {
		  if(t.HitActor.InStateSequence(t.HitActor.CurState, t.HitActor.ResolveState("See")))
		  {
		    t.HitActor.bShootable = false;
		    t.HitActor.SetStateLabel("FastEvade");
			break;
		  }
		}
	  }
	}
  }

Class StunBall : Actor
{
  bool empower;
  int rollAmt, charge;
  
  override void BeginPlay()
  {
    Super.BeginPlay();
	
	bSPRITEFLIP = random(0,1);
	rollAmt = randompick(random(-10,-6), random(6,10));
  }

  override void Tick()
  {
    Super.Tick();
	
    if(!tracer)
	  A_Remove(AAPTR_DEFAULT, RMVF_MISSILES);
		  
	 A_SeekerMissile(0, 90, SMF_PRECISE);
	 A_SetRoll(roll + rollAmt, SPF_INTERPOLATE);
  }
  
  override bool CanCollideWith(Actor other, bool passive)
  {
    Super.CanCollideWith(other, passive);
	
	if(other is "DF_Monster" && other == tracer)
	  return true;
	  
	return false;
  }
  
  Default
  {
    Radius 4;
	Height 8;
    Projectile;
    Speed 35;
	XScale 0.1;
	YScale 0.09;
	DamageType "Stun";
    +SEEKERMISSILE
	+HITTRACER
	+FORCEXYBILLBOARD
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  ELB1 ABCDEFGHIJKLMNOPQRSTUVWXYZ 2 Bright;
	  ELB2 ABCDEFGHIJ 2 Bright;
	  Loop;
	Death:
	  TNT1 A 1
	  {
	    if(tracer)
		{
		  let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		  if(empower)
		  {
		    tracer.A_SpawnItemEx("ElectroBolt", flags: SXF_SETMASTER);
		  }
		  else
		  {
		    if(DFPlayer(players[consoleplayer].mo).armorStack < DFPlayer(players[consoleplayer].mo).armorMax)
			{
		      DFPlayer(players[consoleplayer].mo).armorStack++;
		      ev.UpdateChat("\cD+1 Armor Stack.");
			}
		  }
		
		  DF_Monster(tracer).bStunned = true;
	      DF_Monster(tracer).bStunSoft = true;
	      DF_Monster(tracer).StunTime = 35*2;
	      tracer.A_DamageSelf(random(1, 10), "Stun");
		  for(int i = 2; i < 5; i++)
		  {
		    bool style = random(0,1);
			tracer.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture(String.Format("SPARK%d", i)), style ? STYLE_Translucent : STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE|SPF_ROLL|SPF_LOCAL_ANIM, 23, random(366,439), 0,
			zoff: tracer.height * 0.5, startalphaf: 0.8, fadestepf: 0.0, startroll: random(0,359), rollvel: frandom(-0.5,0.5));
		  }
		  tracer.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("EXPFC0"), STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE|SPF_LOCAL_ANIM, 5, 300, zoff: tracer.height * 0.5, startalphaf: 0.05, startroll: random(0,359));
		  let wep = players[consoleplayer].ReadyWeapon;
		  if(wep is "TrenchGun")
		    TrenchGun(wep).AddCharge();

		  if(charge <= 0)
		    return;
		  
	      BlockThingsIterator it = BlockThingsIterator.Create(self, 1280); //Credit: Boondorl, for the whole thing
		  while(it.Next())
		  {
		    if(it.thing && it.thing is "DF_Monster" && it.thing.bIsMonster && it.thing.health > 0 && Distance3DSquared(it.thing) <= 1638400) //Is this monster alive and in range?
		      if(!it.thing.bDormant && it.thing.bShootable && !it.thing.bFriendly) //Can we damage this entity? should we?
		        if(CheckSight(it.thing, SF_IGNOREVISIBILITY|SF_SEEPASTSHOOTABLELINES|SF_IGNOREWATERBOUNDARY)) //Can we even see it?
			      if(!DF_Monster(it.thing).bStunned) //is it already stunned?
		          {
			        Actor msl = tracer.Spawn("StunBall", tracer.pos + (0, 0, DF_Monster(tracer).sHeight * 0.6));
			        if(msl)
				    {
				      StunBall(msl).empower = empower;
				      StunBall(msl).charge = self.charge -1;
				      msl.tracer = it.thing;
				    }	
				    break;
			}
		  }
		}
	  }
	  Stop;
  }
}
