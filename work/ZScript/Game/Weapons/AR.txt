Class AR : DF_Weapon
{
  bool isSpraying, isEmpty, inf_ammo;
  int spread, recoilControl, muzLayer, burstCount;
  int inf_ammo_tic;
  
  override void AttachToOwner(Actor other)
  {
    if(other)
	{
	  DFPlayer(other).wep4b = true;
	  if(DFPlayer(other).bHigh_Wit)
	    hasAlt = true;
	}
	
	Super.AttachToOwner(other);
  }
  
  action void A_FireAR()
  {
    A_StartSound("AR/Fire", CHAN_WEAPON, volume: 1.0);
	A_AlertMonsters();
	
	if(!DFPlayer(self).bRecoilPower && invoker.isSpraying)
	{
	  invoker.spread = 10.0 - (DFPlayer(self).Str * 2);
	  A_QuakeEx(1, 1, 2, 5, 0, 256, "DSEMPTY");
	}
	else
	  invoker.spread = 1;
	
	if(DFPlayer(self).bPhy_Gland)
	  A_FireBullets(invoker.spread, invoker.spread, 1, 0, "ARPoisonPuff", FBF_NORANDOM);
	  
	A_FireBullets(invoker.spread, invoker.spread, 1, 100, "ARPuff", FBF_NORANDOM);
	A_SpawnCasing("PistolCasing", (51, 28, -7), (random(-1,0),random(5,6),random(6,7)));
	invoker.recoilControl = DFPlayer(self).Str;
	double recoilModf = invoker.recoilControl + (0.05 * invoker.wep_cond);
	if(!DFPlayer(self).bRecoilPower && invoker.recoilControl < 4)
	{
	  A_QuakeEx(1, 1, 2, 5, 0, 256, "DSEMPTY");
	  A_SetPitch(pitch - (8 - recoilModf), SPF_INTERPOLATE);
	  if(!DFPlayer(self).bRecoilPower)
	    A_ZoomFactor(0.94, ZOOM_INSTANT);
	  
	  if(invoker.recoilControl < 2)
	    A_Recoil(1);
	  
	  if(invoker.recoilControl < 3)
	  {
	    double angAdd = 8 - recoilModf;
	    A_SetAngle(angle + random(-angAdd, angAdd), SPF_INTERPOLATE);
	  }
	}
	if(!invoker.inf_ammo)
	{
	  A_TakeInventory("HeavyLoaded",1);
	  invoker.Degrade(40);
	}
  }
  
  override void BeginPlay()
  {
    Super.BeginPlay();
	
	isSpraying = false;
	hasAlt = false;

	wep_order = 5;
	wep_name = "Автомат";
	wep_info = "$AR_Info";
	wep_ability = "$AR_Ability";
	wep_quirk = "$AR_Quirk";
	wep_dmgType = "???";
	wep_primary = "Burst-Fire";
	wep_secondary = "Full Auto";
	wep_AmmoType = 0;
	wep_cond = 100;
	wep_condMax = 100;
	wep_protectMax = 10;
	wep_deg = 1;
	wep_alticon = "W_Alt5";
	wep_ammoicon1 = "WEP5B"; 
	wep_psiCost = 7;
	wep_have_swap = true;
  }
  
  override void DoEffect()
  {
    if(owner && inf_ammo)
	{
	  if(inf_ammo_tic < 35)
	  {
	    inf_ammo_tic++;
	  }
	  else
	  {
	    inf_ammo_tic = 0;
		DFPlayer(owner).inf_ammo_time--;
		if(DFPlayer(owner).inf_ammo_time <= 0)
		  inf_ammo = false;
	  }
	}
    Super.DoEffect();
  }

Class ARPuff : BulletzPuff 
{
  Default
  {
    +ALWAYSPUFF
    +PUFFONACTORS
    DamageType "Pen";
  }
  States
  {
    Spawn:
	  TNT1 A 0 NoDelay
	  {
	    A_DoSpark("PuffSpark");
	    if(tracer && tracer.bIsMonster)
		  return ResolveState("XDeath");
		  
		return ResolveState(null);
	  }
	  Goto Super::Spawn;
    XDeath:
	  TNT1 A 0
	  {
	    if(tracer && tracer.bIsMonster && !DF_Monster(tracer).Armored)
		{
		  A_StartSound("Armored/Pierce");
		  A_SpawnItemEx("BloodHaze"); //Haze to enhance point of impact
		  A_SpawnItemEx("DFBlood"); //Point of impact
		  A_SpawnItemEx("DFBloodSmall3", yvel: frandom(-2.0,2.0), zvel: frandom(-2.0,2.0)); //Point of impact
	      A_SpawnItemEx("DFBloodSpray", xvel: frandom(-3.0, -2.0), zvel: frandom(2.0, 3.5), angle: random(-32,32)); //Spray blood
		
		  for(int i = 0; i < random(8,12); i++)
		    A_SpawnItemEx("BloodParticle", 12, frandom(-16,16), 0, frandom(3,5), frandom(-2,2), frandom(1,3));
	    }
		else
		{
		  A_DoSpark("PuffSpark");
		  A_StartSound("Bullet/Ricochet", CHAN_AUTO, 0, 1.0, 0.6);
		  return ResolveState("Sparks");
		}
		   
	    return ResolveState(null);
	  }
	  Stop;
  }
}
