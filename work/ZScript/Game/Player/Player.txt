Class InfectionToken2 : Inventory
{
  override void DoEffect()
  {
    Super.DoEffect();
	
    if(GetAge() % 420 == 0)
	{
	  DFPlayer(owner).SetToxin(1);
	  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
		ev.UpdateChat(String.Format("%s's \cJinfection spreads...", owner.GetTag()));
	}

	Inventory stToken = FindInventory("Stamina");
	if(stToken && stToken.Amount == stToken.MaxAmount)
	  return;
	
	if(GetAge() % 2 == 0)
	  A_TakeInventory("Stamina", 1);
  }
  
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
	
	if(other is "DFPlayer")
	{
	  //DFPlayer(other).bInfectionPower = true;
	  DFPlayer(other).ForwardMove1 += 1.0;
	  DFPlayer(other).ForwardMove2 += 1.0;
	  DFPlayer(other).SideMove1 += 1.0;
	  DFPlayer(other).SideMove2 += 1.0;
	}
	
	GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(ev)
	  ev.UpdateChat(String.Format("%s \cJcontracted a serious illness...", other.GetTag()));
  }
  
  override void DetachFromOwner()
  {
    Super.DetachFromOwner();
	
	GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(owner && ev)
	{
	  ev.UpdateChat(String.Format("%s \cDCures \cJtheir infection.", owner.GetTag()));
	  //DFPlayer(owner).bInfectionPower = false;
	  DFPlayer(owner).ForwardMove1 -= 1.0;
	  DFPlayer(owner).ForwardMove2 -= 1.0;
	  DFPlayer(owner).SideMove1 -= 1.0;
	  DFPlayer(owner).SideMove2 -= 1.0;
	}
  }
  
  override void ModifyDamage (int damage, Name damageType, out int newdamage, bool passive, Actor inflictor, Actor source, int flags)
  {
    if(!passive && damage > 0)
	{
	  if(damageType == "Melee")
	  {
	    newdamage = damage + 10;
	  }
      else if(damageType == "Daze")
	  {
	    newdamage = damage + 20;
	  }
      else if(damageType == "LeftHook")
	  {
	    newdamage = damage + 50;
	  }
	  else if(damageType == "Gauntlet")
	  {
	    newdamage = damage + 40;
	  }
    }
  }
  
  Default
  {
    Inventory.Amount 1;
	Inventory.MaxAmount 1;
	+Inventory.UNDROPPABLE;
	+Inventory.UNTOSSABLE;
  }
}

Class RandomMover : Inventory
{
  bool isMoving, onBreak;
  int tDuration, tAngle, iLifetime;
  
  override void BeginPlay()
  {
    Super.BeginPlay();
	
	iLifetime = random(2,4) * 35;
	tDuration = random(4,45);
	tAngle = randompick(random(70,120), random(250,290));
  }
  
  override void DoEffect()
  {
    Super.DoEffect();
	
	if(!owner)
	  return;
	  
	let player = owner.player;
    if(!player)
      return;
	  
	if(owner.health <= 0)
	  Destroy();
	
	if(owner)
	{
	  if(DFPlayer(owner).tox < (DFPlayer(owner).toxMax * 0.9))
	    Destroy();
	}
		
	if(owner && player.cmd.buttons & (BT_FORWARD|BT_BACK|BT_MOVELEFT|BT_MOVERIGHT))
	{
	  if(tDuration > 0)
	  {
	    owner.Thrust(0.5, owner.angle + tAngle);
	    tDuration--;
	  }
	  else
	  {
	    tDuration = random(10,45);
		tAngle = randompick(random(90,120), random(250,290));
	  }
	}
  }
  
  Default
  {
    Inventory.Amount 1;
	Inventory.MaxAmount 1;
	+Inventory.UNDROPPABLE;
	+Inventory.UNTOSSABLE;
  }
}

  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod)
  {
    //Before we do anything - we should be evil first
	if(bPhy_Reflect && source && source is "DF_Monster")
	{
	  double dam_refl = 0.2;
	  dam_refl += Phy * 0.1;
	  source.A_DamageSelf(dam_refl * GetMaxHealth(true), "Reflect");
	}
	
    //Immunity
	if(buff_dmg_immune > 0 && mod != "InstantDeath")
	  return 0;
	
    //Difficulty Modifier
    damage *= CVar.FindCVar("modf_damage").GetFloat();
	
	GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(iFrames > 0 && mod != "Extreme" || iFrames > 0 && mod != "Drowning")
	{
	  if(ev)
		ev.UpdateChat(String.Format("· \cFDash (Well-Timed): \cRDamage (%d = \cI0\cR).", damage));
		
	  MovementHandler mv = MovementHandler(EventHandler.Find("MovementHandler"));
	  if(mv)
	    mv.dodgeCD = 0;
	  
	  A_SetBlend("White", 0.1, 24);
	  damage = 0;
	}
	else if(mod == "Physical")
	{
	  ////PSI SHIELD
	  if(psi_shield_health > 0)
	  {
	    double dam = damage;
		damage *= 0.5;
		psi_shield_health -= damage;
		if(ev)
		  ev.UpdateChat(String.Format("· \cFPsi-Shield (%d): \cRDamage (%d\cI-%d \cR= %d)", dam - damage, dam, dam - damage, damage));
	  }
	  ////BLOCKING
	  let wep = players[consoleplayer].ReadyWeapon;
	  if(wep is "Hands")
	  {
	    if(Hands(wep).block_hold && source)
		{
		  if(source && source is "DF_Monster")
		  {
		    int stam_drain = damage * 4.0;
			if(damage <= 0)
			  stam_drain = 0;  
			  
			Hands(wep).HitBlocked(stam_drain);
			
			//if(owner && DFPlayer(owner).bAdvanced_Str && perfect_time > 0)
			if(bAdvanced_Str && Hands(wep).perfect_time < 8 && !Hands(wep).riposte_land)
			{
			  Hands(wep).focusActr = source;
			  FireWeapon(wep.FindState("Riposte"));
			  if(ev)
				ev.UpdateChat(String.Format("· \cFPerfect-Counter (%d): \cRDamage (%d\cI-%d \cR= %d)", damage, damage, damage, 0));
				
			  return Super.DamageMobj(inflictor, source, 0, mod);
			}
		  }
		  double dam = damage;
		  damage *= 0.5;
		  //console.printf("DMG:%d BLOCK:%d", damage, damage * 4.0);
		  if(ev)
		    ev.UpdateChat(String.Format("· \cFBlocking (%d): \cRDamage (%d\cI-%d \cR= %d)", dam - damage, dam, dam - damage, damage));
		}
	  }
	  if(armorStack <= 0 && source)
	    Bleed(AngleTo(source));
	}
	else if(mod == "Energy")
	{
	  if(psi_shield_health > 0)
	  {
	    double dam = damage;
		damage *= 0.5;
		psi_shield_health -= damage;
		if(ev)
		  ev.UpdateChat(String.Format("· \cFPsi-Shield (%d): \cRDamage (%d\cI-%d \cR= %d)", dam - damage, dam, dam - damage, damage));
	  }
	  
	  let wep = players[consoleplayer].ReadyWeapon;
	  if(wep is "Hands")
	  {
	    if(Hands(wep).block_hold)
		{
		  double original_dam = damage;
		  if(source && source is "DF_Monster")
			Hands(wep).HitBlocked(damage * 2.5);
		  //Mitigate damage
		  damage *= 0.5;
		  
		  if(ev)
		    ev.UpdateChat(String.Format("· \cFBlocking (%d): \cRDamage (%d\cI-%d \cR= %d)", original_dam - damage, original_dam, original_dam - damage, damage));
		}
	  }
	  double dam = damage;
	  double mitg = dam * armor_entropy;
	  damage = damage - mitg;
	}
	else if(mod == "Explode" && bResistExplode)
	  damage = 8;
	
	//Finished standard stuff - proceed with armor
	if(armorStack > 0 && damage > 0 && mod == "Physical")
	{
	  double dam = damage;
	  double absorb = dam * armor_mult;
	  dam = dam - absorb;
	  dam -= armor_flat;
	  if(dam < 0)
		dam = 0;
		
	  if(ev)
		ev.UpdateChat(String.Format("· \cFArmor (%d): \cRDamage (%d\cI-%d \cR= %d)", damage - dam, damage, damage - dam, dam));
		
	  armorStack--;
      damage = Max(0, dam);
	}
	if(armorStack <= 0)
	  armorStack = 0;
	  
	//BUT MUH WILL IS STRONGER THAN FLESH AND STEEL TYPE SHIT
	if(buff_withstand && fist_fighting)
	{
      withstand_mult -= 0.15;
	  withstand_stack++;
	  withstand_time = 20;
	  
	  if(withstand_mult <= 0.4)
	  {
	    withstand_mult = 0.4;
		withstand_stack = 4;
	  }
	  
      if(damageType != "Drowning")
      {
	    double dam = damage;
	    damage *= withstand_mult;
		if(ev)
		  ev.UpdateChat(String.Format("· \cFWithstand (%d): \cRDamage (%d\cI-%d \cR= %d)", dam - damage, dam, dam - damage, damage));
	  }
	}
	  
	//Use stim if gonna die regardless
	if(damage >= health && fHealCount > 0)
	{
	  damage = health - 1;
	  buff_dmg_immune = 70;
	  let psp = player.GetPSprite(PSP_WEAPON);
	  if(psp)
	    psp.SetState(player.ReadyWeapon.FindState("DeathHeal"));
	}
	
	return Super.DamageMobj(inflictor, source, damage, mod);
  }

  override void Tick()
  {
    Super.Tick();
	
	if(time_sec >= 35)
	{
	  //Do the things
	  time_sec = 0;
	  
	  //Water stuff
	  if(!silentFall && waterlevel > 0)
	    A_StartSound("player/swim", CHAN_AUTO, volume: 0.75);
	}
	else
	  time_sec++;
	
	if(pos.z == floorz || bOnMObj) //thank you Nash for giving me the wisdom and insight
	{
	  vel.x *= 0.9;
	  vel.y *= 0.9;
	}
	
	//MUSIC VOLUME
	if(TimeVolume > 0)
	{
	  CurVolume += fStepVolume;
	  SetMusicVolume(CurVolume);
	  TimeVolume--;
	}
	
    if(healAmount > 0)
	{
	  HealThing(1);
	  healAmount--;
	}
	
	if(buff_dmg_immune > 0)
	  buff_dmg_immune--;
	  
	if(withstand_time > 0)
	{
	  if(withstand_tic == 35)
	  {
	    withstand_time--;
	    withstand_tic = 0;
	  }
	  else
	  {
	    withstand_tic++;
	  } 
	}
	else
	{
	  withstand_mult = 1.0;
	  withstand_stack = 0;
	}
	
	if(radAmt > 0)
	{
	  if(rad > 0)
	    rad--;
		
	  radAmt--;
	}
	
	if(radBuild)
	{
	  if(radTime > 0)
	    radTime--;
	  else
	  {
	    radTime = radInterval;
		SetRadiation(1);
	  }
	}
	
	if(toxAmt > 0)
	{
	  if(tox > 0)
	    tox--;
		
	  toxAmt--;
	}
	
	if(hActr)
	{
	  if(scn_TimeCheck == 35)
	  {
	    scn_TimeCheck = 0;
	    if(hActr.CheckIfSeen())
	      hActr = null;
	  }
	  scn_TimeCheck++;
	}
	
	if(iFrames > 0)
	{
	  iFrames--;
	  bNOBLOOD = true;
	}
	else
	  bNOBLOOD = false;
	  
	if(tox >= toxMax * 0.5)

	}
	
	if(toxinDmgCD > 0)

	TryLineTrace();
	
	if((Player.cmd.buttons & BT_USE) && !(Player.oldbuttons & BT_USE))

  void TryLineTrace()
  {
	Vector3 start = pos + (0,0,player.viewheight);
	Vector3 dir = (AngleToVector(angle, cos(pitch)), -sin(pitch));
	if(lTr)
    {
		lTr.Trace(start, CurSector, dir, 160, 0);
        if(lTr.results.hitActor)
		{
		  if(lTr.results.hitActor is "Inventory" && Distance3D(lTr.results.hitActor) <= 160)
		  {
		    item_focus = lTr.results.hitActor; //Focus the item
			
		    useContextObjTag = lTr.results.hitActor.GetTag();
			useContextPrompt = "\cUPick up";
			if(lTr.results.hitActor is "Consumable")
			  altUseContextPrompt = "\cUUse";
			else
			  altUseContextPrompt = "";

			return;
		  }
		  //I CONJURE THEE!! SPAGHETTI GODs OF THE FAR REALM!!!
		  item_focus = null; //Did not hit any inventory item. Remove the focus immediately
		  if((lTr.results.hitActor is "NoteItem") && Distance3D(lTr.results.hitActor) <= 96 + lTr.results.hitActor.radius)
		  {
		    useContextObjTag = lTr.results.hitActor.GetTag();
			
			if(lTr.results.hitActor is "Recorder")
			  useContextPrompt = "Listen";
			else
			  useContextPrompt = "Read";
			  
			altUseContextPrompt = "";
			return;
		  }
		  else if((lTr.results.hitActor is "PictureItem") && Distance3D(lTr.results.hitActor) <= 96 + lTr.results.hitActor.radius)
		  {
		    useContextObjTag = lTr.results.hitActor.GetTag();
			useContextPrompt = "Examine";
			altUseContextPrompt = "";
			return;
		  }
		  else if(lTr.results.hitActor is "Breakable" && Distance3D(lTr.results.hitActor) <= 96 + lTr.results.hitActor.radius)
		  {
		    useContextObjTag = lTr.results.hitActor.GetTag();
			useContextPrompt = "Destroy";
			altUseContextPrompt = "";
			return;
		  }
		  else //Object hit isn't anything interesting
	      {
			eatActr = null;
		    useContextObjTag = "";
			UseContextPrompt = "";
			altUseContextPrompt = "";
	      }
		}
	    else //Hit nothing
	    {
		  item_focus = null;
		  eatActr = null;
		  useContextObjTag = "";
		  UseContextPrompt = "";
		  altUseContextPrompt = "";
	    }
	}
	
	if(scTr)
	{
	  scTr.master = self;
	  scTr.Trace(start, CurSector, dir, 2000, 0);  
	  if(scTr.results.hitActor is "DF_Monster" && scTr.results.HitActor.bIsMonster && scTr.results.hitActor.Health > 0)
	    if(!DF_Monster(scTr.results.hitActor).unScannable && !scTr.results.hitActor.bDormant && !scTr.results.hitActor.bBoss)
		  hActr = scTr.results.HitActor; //Make this a valid highlight pointer
	}
	
	if(cTr)
	{
	  cTr.Trace(start, CurSector, dir, 160, 0);
      if(bPhyOrigin && cTr.results.hitActor is "Corpse" && Distance3D(cTr.results.hitActor) <= 96 + cTr.results.hitActor.radius)
	  {
		eatActr = cTr.results.hitActor;
		String tag = cTr.results.hitActor.GetTag() .. " \cD(Fresh)";
		if(Corpse(cTr.results.hitActor).user_stale)
		  tag = cTr.results.hitActor.GetTag() .. " \cJ(Stale)";
			
		useContextObjTag = tag;
		UseContextPrompt = "";
		altUseContextPrompt = "Consume";
	  }
	}
  }

class CorpseTracer : LineTracer
{
  override ETraceStatus TraceCallback()
  {
	switch (results.HitType)
    {
      case TRACE_HitWall:
        if(!(results.hitLine.flags & Line.ML_BLOCKEVERYTHING) && !(results.hitLine.flags & Line.ML_BLOCKSIGHT))
		  if((results.hitLine.flags & Line.ML_TWOSIDED) && results.tier == TIER_Middle)
		    break;
		
      case TRACE_HasHitSky:
      case TRACE_HitFloor:
      case TRACE_HitCeiling:
        return TRACE_Stop;
        break;
            
      case TRACE_HitActor:
        if(results.HitType == TRACE_HitActor)
		{
		  if(results.HitActor is "Corpse")
		    return TRACE_Stop;
		}
    }  
    return TRACE_Skip;
  }
}

class LootTracer : LineTracer
{
  override ETraceStatus TraceCallback()
  {
	switch (results.HitType)
    {
      case TRACE_HitWall:
        if(!(results.hitLine.flags & Line.ML_BLOCKEVERYTHING) && !(results.hitLine.flags & Line.ML_BLOCKSIGHT))
		  if((results.hitLine.flags & Line.ML_TWOSIDED) && results.tier == TIER_Middle)
		    break;
		
      case TRACE_HasHitSky:
      case TRACE_HitFloor:
      case TRACE_HitCeiling:
        return TRACE_Stop;
        break;
            
      case TRACE_HitActor:
        if(results.HitType == TRACE_HitActor)
		{
		  if(results.HitActor is "Inventory" || results.HitActor is "NoteItem" || results.HitActor is "Breakable")
		    return TRACE_Stop;
		}
    }  
    return TRACE_Skip;
  }
}

class ScanTracer : LineTracer
{
    Actor master;
    
    override ETraceStatus TraceCallback()
    {
        switch (results.HitType)
        {
            case TRACE_HitWall:
                if (!(results.hitLine.flags & Line.ML_BLOCKEVERYTHING) &&
                    (results.hitLine.flags & Line.ML_TWOSIDED) &&
                    results.tier == TIER_Middle)
                {
                    break;
                }
            case TRACE_HasHitSky:
            case TRACE_HitFloor:
            case TRACE_HitCeiling:
                return TRACE_Stop;
                break;
            
            case TRACE_HitActor:
                if (results.hitActor != master &&
                    (results.hitActor.bIsMonster || results.hitActor.Health > 0))
                {
                    return TRACE_Stop;
                }
                results.hitActor = null;
                break;
        }
        
        return TRACE_Skip;
    }
}
