Class DF_StaticObj : Actor
{
  String SpriteName;
  property SpriteName : SpriteName;
  
  void CreateVT()
  {
    TextureID tex = TexMan.CheckForTexture(SpriteName);
    let p = Level.SpawnVisualThinker("VThinker");
	if(p)
	{
	  p.Texture = tex;
	  p.pos = pos;
	  p.vel = vel;
      p.SetRenderStyle(GetRenderStyle());
	  if(!bFORCEXYBILLBOARD)
	    p.flags = SPF_NO_XY_BILLBOARD;
	  if(bSPRITEFLIP)
	    p.bXFlip = true;
	  p.alpha = alpha;
	  p.Scale = scale;
	}
  }
  
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	CreateVT();
  }
}

  override void PostBeginPlay()
  {
    double scale = frandom(0.8, 1.0);
	self.scale.x = scale;
	self.scale.y = abs(scale);
	
    tic = random(10,12);
	Super.PostBeginPlay();
  }
  
  override void Tick()
  {
    if(tic >= 0)
	  tic--;
