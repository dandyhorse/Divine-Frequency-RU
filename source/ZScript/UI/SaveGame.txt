//Made by a cat
class Saver : StaticEventHandler
{
	const MAX_SAVES = 3;
	Transient bool active;
	Transient double iconAlpha;
	Transient int saveTic;

	static void Autosave(bool hide = false)
	{
		EventHandler.SendNetworkEvent(hide ? "AutosaveHidden" : "Autosave");
	}

	static void Quicksave()
	{
		EventHandler.SendNetworkEvent("MQuicksave");
	}

	override void NetworkProcess(ConsoleEvent e)
	{
		if(e.Name ~== "Autosave" || e.Name ~== "AutosaveHidden") 
		  EventHandler.SendInterfaceEvent(e.Player, "DoAutosave");
		else if(e.Name ~== "MQuicksave")
		  EventHandler.SendInterfaceEvent(e.Player, "DoQuicksave");
		else if(e.Name ~== "ManualSave")
	      EventHandler.SendInterfaceEvent(e.Player, "DoManualsave");
		  
		//Do fancy shit if it's not a quiet/hidden save
		if(e.Name ~== "Autosave" || e.Name ~== "MQuicksave" || e.Name ~== "Manualsave")
		{
			players[e.Player].mo.A_StartSound("Save", CHAN_AUTO, CHANF_UI, volume: 0.7);
			active = true;
			saveTic = 70;
			iconAlpha = 1.0;
		}
	}

	override void InterfaceProcess(ConsoleEvent e)
	{
		if(e.Name ~== "DoAutosave")
			SaveGame("(Autosave)");
		else if(e.Name ~== "DoQuicksave")
			SaveGame("(Quicksave)");
		else if(e.Name ~== "DoManualsave")
			SaveGame("");
	}

	static ui void SaveGame(string tag)
	{
		let pmo = DFPlayer(players[consoleplayer].mo);
		if(!pmo)
			return;

		let sgm = SaveGameManager.GetManager();
		
		if(tag != "") //Preserve manual saves
		{
			int count = 0;
			for(int i = 0; i < sgm.SavegameCount(); i++)
			{
				if(sgm.GetSavegame(i).SaveTitle.Left(tag.Length()) == tag)
					count++;
			}

			for(int i = sgm.SavegameCount() - 1; i >= 0 && count >= MAX_SAVES; i--)
			{
				if(sgm.GetSavegame(i).SaveTitle.Left(tag.Length()) == tag)
				{
					sgm.RemoveSaveSlot(i);
					count--;
				}
			}
		}
		sgm.DoSave(0, String.Format("%s %s - %s", tag, pmo.CharName, pmo.Location));
	}

	override void WorldTick()
	{
		if(!active)
			return;

		saveTic--;

		if(saveTic <= 40)
			iconAlpha -= 0.025;

		if(saveTic <= 0)
		{
			active = false;
			iconAlpha = 0.0;
			saveTic = 0;
		}
	}

	override void RenderOverlay(RenderEvent e)
	{
		if(!active)
			return;

		Screen.DrawTexture(TexMan.CheckForTexture("SaveIcon"), false, 305, 20,
		DTA_ScaleX, 0.15,
		DTA_ScaleY, 0.17,
		DTA_KeepRatio, true,
		DTA_320x200, true,
		DTA_CenterOffset, true,
		DTA_Alpha, iconAlpha);
	}
}
