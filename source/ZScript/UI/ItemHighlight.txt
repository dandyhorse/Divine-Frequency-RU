Class ItemHighlightHandler : EventHandler
{
  Actor targetHighlight;
  bool tex_setup, drawable;
  TextureID bTex;
  ui Vector2 screenCoord[2];
  int hRadius, hHeight;
  state state_prev, state_cur;
  
  override void WorldTick()
  {
    Super.WorldTick();
	
	let pmo = players[consoleplayer].mo;
	if(pmo && DFPlayer(pmo).item_focus)
	{
	  targetHighlight = DFPlayer(pmo).item_focus;
	  state_cur = targetHighlight.CurState;
      if(state_cur != state_prev)
      {
        state_prev = state_cur;
        TextureID tex = targetHighlight.CurState.GetSpriteTexture(0, scale: targetHighlight.Scale);
        if(tex.IsValid())
        {
		  if(TexMan.GetName(tex) ~== "0000A0")
		  {
		    targetHighlight = null;
			drawable = false;
		  }
		  else
		  {
            Vector2 texSize = TexMan.GetScaledSize(tex);
            hRadius = texSize.x * targetHighlight.scale.x;
            hHeight = texSize.y * targetHighlight.scale.y;
			drawable = true;
		  }
        }
      }
	}
	else
	  targetHighlight = null;
	  
	if(!tex_setup)
	{
	  bTex = TexMan.CheckForTexture("PRJBOXTL");
	  tex_setup = true;
	}
  }
	
  override void RenderUnderlay(RenderEvent e)
  {
	if(automapactive != 0)
	  return;
	  
	let pmo = players[consoleplayer].mo;
	if(!pmo)
	  return;
	  
	if(!targetHighlight)
	  return;
	  
	if(!drawable)
	  return;
	  
	double fovH = (e.camera.player) ? e.camera.player.FOV : e.camera.cameraFOV;
	GMMatrix worldToClip = GMMatrix.worldToClip(e.viewPos, e.viewAngle, e.viewPitch, e.viewRoll, fovH, Screen.GetAspectRatio());
	
	double halfRad = hRadius* .5;
	for (let corner = 0; corner < 4; corner++)
	{
	  vector3 objPos = targetHighlight.pos;
	  if(targetHighlight.bFLOATBOB)
	    objPos.z += targetHighlight.GetBobOffset();
	  
	  let vert = e.ViewPos + LevelLocals.Vec3Diff(e.ViewPos, objPos + (sin(e.viewAngle) * halfRad * (corner & 1? 1. : -1.), cos(e.viewAngle) * halfRad * (corner & 1? -1. : 1.), corner & 2? 0 : hHeight));
	  let ndc = worldToClip.multiplyVector3(vert).asVector3();
			
	  if(ndc.Z > 1. || ndc.Z < -1.)
		return;
			
	  let cornerCoord = GMGlobalMaths.ndcToViewPort(ndc);
	  if(!corner)
		screenCoord[0] = screenCoord[1] = cornerCoord;
	  else
	  {
		screenCoord[0].X = min(screenCoord[0].X, cornerCoord.X);
		screenCoord[0].Y = min(screenCoord[0].Y, cornerCoord.Y);
		screenCoord[1].X = max(screenCoord[1].X, cornerCoord.X);
		screenCoord[1].Y = max(screenCoord[1].Y, cornerCoord.Y);
	  }
	}
		
	let texSize = TexMan.GetScaledSize(bTex) * (min(Screen.GetWidth(), Screen.GetHeight()) / 1000.0);	
	{
	  let centerCoord = screenCoord[0] + ((screenCoord[1] - screenCoord[0]) * 0.5);
	  screenCoord[0].X = min(screenCoord[0].X, centerCoord.X - texSize.X);
	  screenCoord[0].Y = min(screenCoord[0].Y, centerCoord.Y - texSize.Y);
	  screenCoord[1].X = max(screenCoord[1].X, centerCoord.X + texSize.X);
	  screenCoord[1].Y = max(screenCoord[1].Y, centerCoord.Y + texSize.Y);
	}
		
	let cornerAdds = (screenCoord[1] - screenCoord[0]) * 0.5 * 0.4;
	let cornerAdd = min(cornerAdds.X, cornerAdds.Y);
	screenCoord[0].X -= cornerAdd;
	screenCoord[1].X += cornerAdd;
	screenCoord[0].Y -= cornerAdd;
	screenCoord[1].Y += cornerAdd;
		
	for(let corner = 0; corner < 4; corner++)
	  Screen.DrawTexture(
	  bTex, false,
	  screenCoord[corner & 1].X - (corner & 1? texSize.X : 0.0),
	  screenCoord[!!(corner & 2)].Y - (corner & 2? texSize.Y : 0),
	  DTA_FlipX, corner & 1,
	  DTA_FlipY, !!(corner & 2),
	  DTA_DestWidthF, texSize.X,
	  DTA_DestHeightF, texSize.Y
    );
  }
}
