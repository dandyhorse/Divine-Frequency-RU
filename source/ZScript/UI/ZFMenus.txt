class ZFMenuEHandler : EventHandler
{
	static const string StrTitle[] = { "Savage", "Tempered Soul", "Indomitable Soul", "Deathless Incarnation" }; //Forward Orientation
	static const string WitTitle[] = { "Acolyte", "Occultist", "Mastermind", "The Eye of Illumination" }; //Upoward Orientation
	static const string PhyTitle[] = { "Failure", "Predator", "Eldritch Horror", "Collective Darkness" }; //Downward Orientation
	
	override void WorldLoaded(WorldEvent e)
	{
	  for(int i = 0; i < players.Size(); i++)
      {
        let pmo = players[i].mo;
        if(pmo && pmo is "DFPlayer" && DFPlayer(pmo).bBase_Wit)
	    {
	      ThinkerIterator ConFinder = ThinkerIterator.Create("Consumable");
	      Actor mo;
	      while(mo = Consumable(ConFinder.Next()))
		    Consumable(mo).flash = true;
		}
		
		if(pmo && pmo.FindInventory("LanternToken"))
		{
		  DFPlayer(pmo).lite = pmo.Spawn("LanternLight", pmo.pos);
		  DFPlayer(pmo).lite.master = pmo;
		}
	  }
	}
	
	override void InterfaceProcess(ConsoleEvent e)
    {
		PlayerPawn pmo = players[consoleplayer].mo;
		if(e.Name ~== "HUDUpdateQBTex")
		{
			if(pmo)
			{
				//Check quickbind 1
				if(DFPlayer(pmo).quickUse1)
				{
					DFStatusBar(StatusBar).qb1 = true;
					DFStatusBar(StatusBar).qb1_img = TexMan.CheckForTexture(DFPlayer(pmo).quickUse1.invIcon);
					
				    if(!(DFPlayer(pmo).quickUse1 is "Skill"))
						DFStatusBar(StatusBar).qb1_amount = String.Format("%d", DFPlayer(pmo).quickUse1.amount);
					else
						DFStatusBar(StatusBar).qb1_amount = "";
						
					if(DFPlayer(pmo).quickUse1.amount > 0)
						DFStatusBar(StatusBar).qb1_alpha = 1.0;
					else
						DFStatusBar(StatusBar).qb1_alpha = 0.5;
				}
				else
				{
					DFStatusBar(StatusBar).qb1 = false;
				}
				//Check quickbind 2
				if(DFPlayer(pmo).quickUse2)
				{
					DFStatusBar(StatusBar).qb2 = true;
					DFStatusBar(StatusBar).qb2_img = TexMan.CheckForTexture(DFPlayer(pmo).quickUse2.invIcon);
					
				    if(!(DFPlayer(pmo).quickUse2 is "Skill"))
						DFStatusBar(StatusBar).qb2_amount = String.Format("%d", DFPlayer(pmo).quickUse2.amount);
					else
						DFStatusBar(StatusBar).qb2_amount = "";
						
					if(DFPlayer(pmo).quickUse2.amount > 0)
						DFStatusBar(StatusBar).qb2_alpha = 1.0;
					else
						DFStatusBar(StatusBar).qb2_alpha = 0.5;
						
				}
				else
				{
					DFStatusBar(StatusBar).qb2 = false;
				}
				//Check quickbind 3
				if(DFPlayer(pmo).quickUse3)
				{
					DFStatusBar(StatusBar).qb3 = true;
					DFStatusBar(StatusBar).qb3_img = TexMan.CheckForTexture(DFPlayer(pmo).quickUse3.invIcon);
						
				    if(!(DFPlayer(pmo).quickUse3 is "Skill"))
						DFStatusBar(StatusBar).qb3_amount = String.Format("%d", DFPlayer(pmo).quickUse3.amount);
					else
						DFStatusBar(StatusBar).qb3_amount = "";
						
					if(DFPlayer(pmo).quickUse3.amount > 0)
						DFStatusBar(StatusBar).qb3_alpha = 1.0;
					else
						DFStatusBar(StatusBar).qb3_alpha = 0.5;
				}
				else
				{
					DFStatusBar(StatusBar).qb3 = false;
				}
				//Check quickbind 4
				if(DFPlayer(pmo).quickUse4)
				{
					DFStatusBar(StatusBar).qb4 = true;
					DFStatusBar(StatusBar).qb4_img = TexMan.CheckForTexture(DFPlayer(pmo).quickUse4.invIcon);
					
				    if(!(DFPlayer(pmo).quickUse4 is "Skill"))
						DFStatusBar(StatusBar).qb4_amount = String.Format("%d", DFPlayer(pmo).quickUse4.amount);
					else
						DFStatusBar(StatusBar).qb4_amount = "";
						
					if(DFPlayer(pmo).quickUse4.amount > 0)
						DFStatusBar(StatusBar).qb4_alpha = 1.0;
					else
						DFStatusBar(StatusBar).qb4_alpha = 0.5;
				}
				else
				{
					DFStatusBar(StatusBar).qb4 = false;
				}
			}
		}
		else if (e.Name ~== "HUDUpdateHealth")
		{
			if(pmo)
			{
				DFStatusBar(StatusBar).plr_hp = pmo.Health;
				DFStatusBar(StatusBar).plr_max_hp = pmo.GetMaxHealth(true);
				DFStatusBar(StatusBar).plr_health = String.Format("%d/%d", pmo.Health, pmo.GetMaxHealth(true));
			}
		}
		else if (e.Name ~== "HUDUpdatePortrait")
		{
			DFStatusBar(StatusBar).Update_New_Portait();
			DFStatusBar(StatusBar).tox_nfo = String.Format("%d/%d", DFPlayer(pmo).tox, DFPlayer(pmo).toxMax);
			DFStatusBar(StatusBar).rad_nfo = String.Format("%d/%d", DFPlayer(pmo).rad, DFPlayer(pmo).radMax);
		}
		else if (e.Name ~== "HUDUpdateStim")
		{
			if(pmo)
			{
				let inv = pmo.FindInventory("Stim");
				if(inv)
				{
					DFStatusBar(StatusBar).stim_nfo = String.Format("%d/%d", Stim(inv).amt, Stim(inv).maxAmt);
					if(Stim(inv).amt <= 0)
					{
						DFStatusBar(StatusBar).stim_nfo = String.Format("\cU0/%d", Stim(inv).maxAmt);
					}
				}
			}
		}
	}
	
	override void NetworkProcess(ConsoleEvent e)
	{
		PlayerPawn pmo = players[e.Player].mo;

		if (pmo)
		{
			Array<string> command;
			e.Name.Split (command, ":");
			
			/*
			/////////////////////////////////////////
			////////////////////////////////////////
			///////////     SET NAME      ////////
			/////////////////////////////////////
			*/
			if((command [0] ~== "SetName"))
			{
				pmo.SetTag("\cF" .. command[1]);
				DFPlayer(pmo).CharName = command[1];
			}
			
			/*
			/////////////////////////////////////////
			////////////////////////////////////////
			///////////    GUN STORE      ////////
			/////////////////////////////////////
			*/
			if((command [0] ~== "UpgradeGun"))
			{
				pmo.GiveInventory(command[1], 1);
				if(command.Size() > 2)
					pmo.GiveInventory(command[2], 1);
			}
			
			/////////////////////////////////////////
			////////////////////////////////////////
			///////////   LEVEL UP     ///////////
			/////////////////////////////////////
			if(command [0] ~== "LevelUpBoi")
			{
				DFPlayer(pmo).level++;
				DFPlayer(pmo).tune_max++;
				
				let inv = pmo.FindInventory("Attunement_Check");
				if(inv)
				  inv.Amount = 1;
				  
				if(command [1] ~== "StrToken")
				{
					DFPlayer(pmo).Str += 1;
				    if(DFPlayer(pmo).statBase == "Str")
					{
					  DFPlayer(pmo).porName = String.Format("P_STR%d", DFPlayer(pmo).Str);
					  DFPlayer(pmo).title = StrTitle[DFPlayer(pmo).Str - 1];
					}
					  
					pmo.GiveInventory("StrToken", 1);
					
					if(DFPlayer(pmo).Str == 1)
					{
					  DFPlayer(pmo).bBase_Str = true;
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Str1");
					  
					  if(DFPlayer(pmo).bStrOrigin)
					    DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Str1_og");
					}
					
					if(DFPlayer(pmo).Str == 2)
					{
					  DFPlayer(pmo).bAdvanced_Str = true;
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Str2");
					  
					  if(DFPlayer(pmo).bStrOrigin)
					  {
					    DFPlayer(pmo).bStrOrigin_Buff = true;
						DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Str2_og");
					  }
					}
					
					if(DFPlayer(pmo).Str == 3)
					{
					  DFPlayer(pmo).bHigh_Str = true;
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Str3");
					  pmo.GiveInventory("Skill_IronWill", 1);
					}
					
					if(DFPlayer(pmo).Str == 4)
					{
					  DFPlayer(pmo).bMax_Str = true;
					  DFPlayer(pmo).bResistToxin = true;
					  pmo.GiveInventory("Skill_Undying", 1);
					}
				}
				
				if(command [1] ~== "WitToken")
				{
					DFPlayer(pmo).Wit += 1;
					if(DFPlayer(pmo).statBase == "Wit")
					{
					  DFPlayer(pmo).porName = String.Format("P_WIT%d", DFPlayer(pmo).Wit);
					  DFPlayer(pmo).title = WitTitle[DFPlayer(pmo).Wit -1];
					}
					pmo.GiveInventory("WitToken", 1);
					pmo.ACS_NamedExecute("Wits_Update");
					if(DFPlayer(pmo).Wit == 1)
					{
					  DFPlayer(pmo).bBase_Wit = true;
					  DFPlayer(pmo).bAutoReload = true;
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Wit1");
					  ThinkerIterator KeyFinder = ThinkerIterator.Create("Consumable");
					  Actor mo;
					  while(mo = Consumable(KeyFinder.Next()))
					    Consumable(mo).flash = true;
						
					  if(DFPlayer(pmo).bWitOrigin)
					  {
					    DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Wit1_og");
					    DFPlayer(pmo).tune_max++;
						pmo.GiveInventory("Skill_Radar", 1);
					  }
					}
					  
					if(DFPlayer(pmo).Wit == 2)
					{
					  DFPlayer(pmo).bAdvanced_Wit = true;
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Wit2");
					  
					  if(DFPlayer(pmo).bWitOrigin)
					  {
					    DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Wit2_og");
					    DFPlayer(pmo).hasScanner = true;
						DFPlayer(pmo).tune_max++;
						pmo.GiveInventory("Skill_Shield", 1);
					  }
					  
					  if(DFPlayer(pmo).statBase == "Wit")
					    DFPlayer(pmo).porScan = "P_WEYE1";
					  
					  let ps = pmo.FindInventory("Psi");
					  if(ps)
					    ps.MaxAmount += 10;
					}
					
					if(DFPlayer(pmo).Wit == 3)
					{
					  DFPlayer(pmo).bHigh_Wit = true;
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Wit3");
					  
					  if(DFPlayer(pmo).bWitOrigin)
					  {
						DFPlayer(pmo).tune_max++;
						DFPlayer(pmo).bWitOrigin_Buff = true;
						DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Wit3_og");
					  }
					  let w_ssg = pmo.FindInventory("SSG");
					  if(w_ssg)
						SSG(w_ssg).hasAlt = true;
						
					  let w_ar = pmo.FindInventory("AR");
					  if(w_ar)
						AR(w_ar).hasAlt = true;
						
					  if(DFPlayer(pmo).statBase == "Wit")
					    DFPlayer(pmo).porScan = "P_WEYE2";
					}
					
					if(DFPlayer(pmo).Wit == 4)
					{
					  DFPlayer(pmo).bMax_Wit = true;
					}

					if(DFPlayer(pmo).Wit >= 2)
					{
					  pmo.ForwardMove1 += 0.07;
					  pmo.ForwardMove2 += 0.07;
					  pmo.SideMove1 += 0.07;
					  pmo.SideMove2 += 0.07;
					}
				}
				
				if(command [1] ~== "PhyToken")
				{
					DFPlayer(pmo).Phy += 1;
					if(DFPlayer(pmo).statBase == "Phy")
					{
				      DFPlayer(pmo).porName = String.Format("P_PHY%d", DFPlayer(pmo).Phy);
					  DFPlayer(pmo).title = PhyTitle[DFPlayer(pmo).Phy -1];
					}
					
					pmo.GiveInventory("PhyToken", 1);
					//pmo.GiveInventory("HealthUpgrade", 10);
					DFPlayer(pmo).tox = 0;
					DFPlayer(pmo).rad = 0;
					DFPlayer(pmo).toxMax += 10;
					DFPlayer(pmo).radMax += 10;

					if(DFPlayer(pmo).Phy == 1)
					{
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Phy1");
					  DFPlayer(pmo).bBase_Phy = true;
					  pmo.GiveInventory("Mutation", 3);
					  if(DFPlayer(pmo).bPhyOrigin)
					  {
					    DFPlayer(pmo).invMax += 3;
					    DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Phy1_og");
					  }
					}
					
					if(DFPlayer(pmo).Phy == 2)
					{
					  DFPlayer(pmo).bAdvanced_Phy = true;
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Phy2");
					  pmo.GiveInventory("Mutation", 5);
					  if(DFPlayer(pmo).statBase == "Phy")
						DFPlayer(pmo).porHPa = "P_PHYHP2";
					}
					  
					if(DFPlayer(pmo).Phy == 3)
					{
					  DFPlayer(pmo).bHigh_Phy = true;
					  DFPlayer(pmo).char_upgrade.AppendFormat(":$NF_Phy3");
					  pmo.GiveInventory("Mutation", 10);
					}
					
					if(DFPlayer(pmo).Phy == 4)
					{
					  DFPlayer(pmo).bMax_Phy = true;
					  if(DFPlayer(pmo).statBase == "Phy")
						DFPlayer(pmo).porHPa = "P_PHYHP3";
					}
				}
				
				pmo.FindInventory("ExaltXp").Amount -= command[2].toInt(10);
				SendInterfaceEvent(0, "HUDUpdatePortrait");
			}
			
			/*
			/////////////////////////////////////////
			////////////////////////////////////////
			///////////   INVENTORY     ///////////
			/////////////////////////////////////
			*/
			if(command [0] ~== "UseItem")
			{
				let itm = pmo.FindInventory(command[1]);
				if(itm)
				{
				  pmo.UseInventory(itm);
				  pmo.A_StartSound(command[2], CHAN_AUTO, CHANF_UI);
				  EventHandler.SendInterfaceEvent(0, "HUDUpdateQBTex");
				}
			}

			if(command [0] ~== "UseTonic")
			{
				let itm = pmo.FindInventory("Stim");
				if(itm && Stim(itm).amt > 0)
				{
				  pmo.UseInventory(itm);
				  pmo.A_StartSound("StimHeal", CHAN_AUTO, CHANF_UI);
				}
			}
			
			if(command [0] ~== "UseFlare")
			{
				let itm2 = pmo.FindInventory("Lantern");
				if(itm2 && Lantern(itm2).on)
				{
				  Lantern(itm2).invIcon = "Graphics/Inventory Icon/Lantern.png";
				}
				
				let itm = pmo.FindInventory("EmergencyLight");
				if(itm)
				{
				  pmo.UseInventory(itm);
				  if(EmergencyLight(itm).on)
				    CustomInventoryItem(itm).invIcon = "Graphics/Inventory Icon/Light_On.png";
				  else
				    CustomInventoryItem(itm).invIcon = "Graphics/Inventory Icon/Light.png";
				}
			}
			
			if(command [0] ~== "UseLantern")
			{
				let itm2 = pmo.FindInventory("EmergencyLight");
				if(itm2 && EmergencyLight(itm2).on)
				{
				  EmergencyLight(itm2).invIcon = "Graphics/Inventory Icon/Light.png";
				}
				
				let itm = pmo.FindInventory("Lantern");
				if(itm)
				{
				  pmo.UseInventory(itm);
				  if(Lantern(itm).on)
				    CustomInventoryItem(itm).invIcon = "Lant_On";
				  else
				    CustomInventoryItem(itm).invIcon = "Lant";
				}
			}
			
			if(command [0] ~== "DropItem")
			{
				let itm = pmo.FindInventory(command[1]);
				if(itm)
				{
				  pmo.DropInventory(itm, 1);
				  pmo.A_StartSound("SmallItem/Up", CHAN_AUTO, CHANF_UI);
				  EventHandler.SendInterfaceEvent(0, "HUDUpdateQBTex");
				}
			}
			
			if(command [0] ~== "BindQuickUse")
			{
				let itm = CustomInventoryItem(pmo.FindInventory(command[2]));
				if(itm)
				{
				  if(command[1] ~== "1")
				    DFPlayer(pmo).quickUse1 = itm;
					
				  if(command[1] ~== "2")
				    DFPlayer(pmo).quickUse2 = itm;
					
				  if(command[1] ~== "3")
				    DFPlayer(pmo).quickUse3 = itm;
					
				  if(command[1] ~== "4")
				    DFPlayer(pmo).quickUse4 = itm;
					
				  EventHandler.SendInterfaceEvent(0, "HUDUpdateQBTex");
				}
			}
			if(command [0] ~== "DuelExit")
			{
				pmo.ACS_NamedExecute("Duel_Exit");
			}
		}
	}
}
