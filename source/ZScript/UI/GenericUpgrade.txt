Class BagUpgrade : InventoryItem
{
  bool user_no_flash;
  Default
  {
    //$Category "Item/Ammo"
	//$Color 6
	Inventory.PickupMessage "";
	Inventory.PickupSound "";
	+INVENTORY.NOSCREENFLASH
	+FORCEXYBILLBOARD
	Tag "Item Pouch";
	Radius 9;
	Height 19;
	XScale 0.36;
	YScale 0.3;
  }
  States
  {
    Spawn:
	  PUCH A 55;
	  PUCH A 0
	  {
	    if(!invoker.user_no_flash)
		{
	      let actr = Spawn("LootFlash", pos);
	      if(actr)
	      {
		    actr.scale = scale;
		    actr.frame = frame;
		    actr.sprite = sprite;
		    if(bFORCEXYBILLBOARD)
		      actr.bFORCEXYBILLBOARD = true;
		  }
		}
	  }
	  Loop;
    Pickup:
	  TNT1 A 0
	  {
	    DFPlayer(self).invMax++;
		A_GiveInventory("UIChatDipper", 1);
		A_StartSound("Bag_Up", CHAN_AUTO, volume: 0.7);
		ACS_NamedExecute("Bag_Upgrade1", 0, 1);
		
		if(itemFlash_amount > 0.0)
		  A_SetBlend("#ffffff", itemFlash_amount, 5);
	  }
	  Stop;
  }
}

Class UIChatDipper : Inventory
{
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
	
	let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(ev)
	  ev.itemChatDip = true;
  }
  
  override void DoEffect()
  {
    Super.DoEffect();
	if(GetAge() > 175)
	{
	  let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
	    ev.itemChatDip = false;
		
	  Destroy();
	}
  }
  
  Default
  {
    Inventory.Amount 1;
    +Inventory.UNDROPPABLE
  }
}

Class PhyPurger : Inventory
{
  int tic;
  
  override void DoEffect()
  {
    Super.DoEffect();
	if(owner && tic >= 175)
	{
	  DFPlayer(owner).SetToxin(-1);
	  DFPlayer(owner).SetRadiation(-1);
	  tic = 0;
	}
	
	tic++;
  }
  
  Default
  {
    Inventory.Amount 1;
    +Inventory.UNDROPPABLE
  }
}

Class ChickenUpgrade : Inventory //Level 5 ability. Stack with chicken buff
{
  int Tic;
  
  override void PostBeginPlay()
  {
	Tic = 0;
  }
  
  override void DoEffect()
  {
    if(Tic >= 140)
	{
	  owner.A_GiveInventory("Health", 1);
	  Tic = 0;
	}
	Tic++;
	Super.DoEffect();
  }
  
  override void OwnerDied()
  {
    Tic = 0;
    Destroy();
	Super.OwnerDied();
  }
  
  Default
  {
    Inventory.MaxAmount 1;
    +Inventory.UNDROPPABLE
    -Inventory.INVBAR
  }
}

Class ArmorRegen : Inventory
{
  int Tic;
  
  override void PostBeginPlay()
  {
	Tic = 0;
  }
  
  override void DoEffect()
  {
    if(Tic >= 1575)
	{
	  if(owner && DFPlayer(owner).armorStack < DFPlayer(owner).armorMax)
	  {
	    DFPlayer(owner).armorStack++;
		GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		if(ev)
		  ev.UpdateChat("\cD+1 Armor Stack.");
	  }
	  Tic = 0;
	}
	Tic++;
	Super.DoEffect();
  }
  
  override void OwnerDied()
  {
    Tic = 0;
    Destroy();
	Super.OwnerDied();
  }
  
  Default
  {
    Inventory.MaxAmount 1;
    +Inventory.UNDROPPABLE
    -Inventory.INVBAR
  }
}

Class DamageResistUpgrade : Inventory
{
  override void ModifyDamage (int damage, Name damageType, out int newdamage, bool passive, Actor inflictor, Actor source, int flags)
  {
    if(passive && damage > 0)
      newdamage = Max(0, damage * 0.85);
  }
  
  Default
  {
    Inventory.MaxAmount 1;
	+INVENTORY.UNDROPPABLE
	+INVENTORY.QUIET
	-INVENTORY.INVBAR
  }
}

Class Astro_Gift1 : Inventory
{
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
	
	if(owner)
	{
	  DFPlayer(owner).bUQ_Mag = true;
	  let pAmmo = other.FindInventory("BulletLoaded");
	  let wpn = Handgun(other.FindInventory("Handgun"));

	  {
		wpn.cSize += 4;
		pAmmo.MaxAmount += 4;
		other.A_GiveInventory("BulletLoaded", pAmmo.MaxAmount);
		other.A_GiveInventory("M2_AstroGift", 1); //marker
	  }
	}
	
	Destroy();
  }
  
  Default
  {
    Inventory.Amount 1;
  }
}

Class UpgradeHandler : EventHandler
{
  static void UpgradeShotgun()
  {
    let pmo = players[consoleplayer].mo;
	if(pmo)
	{
      let wpn = Trenchgun(pmo.FindInventory("Trenchgun"));
	  if(wpn)
	  {
	    wpn.hasAlt = true;
	    wpn.firstPickup = false;
	  }
	}
  }
  override void NetworkProcess (ConsoleEvent e)
  {	
	let pmo = players[consoleplayer].mo;
	if(pmo)
	{
	  Array<String> cmd;
	  String commandList = e.Name;
	  commandList.Split(cmd, ":");

	  if(cmd[0] ~== "MutateCost")
	  {
	    pmo.A_TakeInventory("Mutation", cmd[1].ToInt(10));
	  }
	  else if(e.Name ~== "phy_levelup")
	  {
	    int lvl = DFPlayer(pmo).phyLevel;
	    if(lvl == 2 || lvl == 7 || lvl == 10 || lvl == 12 || lvl == 13)
		{
	      pmo.ACS_NamedExecute("Phy_LevelUp_HPM");
	      pmo.GiveInventory("UIChatDipper2", 1);
		  pmo.GiveInventory("HealthUpgrade", 10);
	      pmo.GiveInventory("Mutation", 1);
		}
		else if(lvl == 4 || lvl == 5 || lvl == 8 || lvl == 11)
		{
		  pmo.ACS_NamedExecute("Phy_LevelUp_M");
	      pmo.GiveInventory("UIChatDipper2", 1);
		  pmo.GiveInventory("Mutation", 1);
		}
		else if(lvl == 3)
		{
		  pmo.ACS_NamedExecute("Phy_LevelUp_A1");
	      pmo.GiveInventory("UIChatDipper2", 1);
		  pmo.GiveInventory("Skill_Blob", 1);
		}
		else if(lvl == 6)
		{
		  pmo.ACS_NamedExecute("Phy_LevelUp_A2");
	      pmo.GiveInventory("UIChatDipper2", 1);
		  pmo.GiveInventory("Skill_Gland", 1);
		}
		else if(lvl == 9)
		{
		  pmo.ACS_NamedExecute("Phy_LevelUp_A3");
	      pmo.GiveInventory("UIChatDipper2", 1);
		  pmo.GiveInventory("Skill_Vomit", 1);
		}
		else
		{
		  pmo.ACS_NamedExecute("Phy_LevelUp");
	      pmo.GiveInventory("UIChatDipper2", 1);
		  pmo.GiveInventory("Mutation", 1);
		}
	  }
	  else if(e.Name ~== "b_Parasite")
	  {
	    DFPlayer(pmo).bBrainPower = true;
		pmo.ACS_NamedExecute("NewEntry");
	  }
	  else if(e.Name ~== "m_Health")
	  {
	    pmo.GiveInventory("HealthUpgrade", 20);
		pmo.GiveInventory("ArmorRegen", 1);
		pmo.GiveInventory("Skill_HP1", 1);
		let amount = pmo.FindInventory("Stamina");
		if(amount)
		  amount.MaxAmount += 300;
		  
		if(DFPlayer(pmo).bPhy_Mind)
		  pmo.GiveInventory("HealthUpgrade", 30);
		  
		DFPlayer(pmo).bPhy_HP = true;
	  }
	  else if(e.Name ~== "m_BigBrain")
	  {
	    pmo.GiveInventory("Skill_HP2", 1);
		let smart = pmo.FindInventory("PSI"); //double the amount of psi
		if(smart)
		  smart.MaxAmount += 15;
		
		if(DFPlayer(pmo).bPhy_HP)
		  pmo.GiveInventory("HealthUpgrade", 30);
		  
		DFPlayer(pmo).bPhy_Mind = true;
	  }
	  else if(e.Name ~== "m_Caltrop")
	  {
		pmo.A_GiveInventory("Skill_Caltrop", 1);
		DFPlayer(pmo).bPhy_Caltrop = true;
	  }
	  else if(e.Name ~== "m_Flux")
	  {
		pmo.A_GiveInventory("Skill_Blast", 1);
		DFPlayer(pmo).bPhy_Flux = true;
	  }
	  else if(e.Name ~== "m_Hunger")
	  {
	    pmo.GiveInventory("Skill_Hunger", 1);
		DFPlayer(pmo).bPhy_Hunger = true;
		DFPlayer(pmo).PhyHeal = 40;
	  }
	  else if(e.Name ~== "m_MorePocket")
	  {
	    pmo.GiveInventory("Skill_Pocket", 1);
		DFPlayer(pmo).invMax += 8;
		DFPlayer(pmo).bPhy_MorePocket = true;
	  }
	  else if(e.Name ~== "m_Recoil")
	  {
	    pmo.GiveInventory("Skill_Focus", 1);
		DFPlayer(pmo).bRecoilPower = true;
		DFPlayer(pmo).bPhy_Tendon = true;
	  }
	  else if(e.Name ~== "m_Reflect")
	  {
		pmo.GiveInventory("Skill_Reflect", 1);
		DFPlayer(pmo).bPhy_Reflect  = true;
	  }
	  else if(e.Name ~== "w_HandgunAuto")
	  {
	    let pAmmo = pmo.FindInventory("BulletLoaded");
		let wpn = Handgun(pmo.FindInventory("Handgun"));
		if(pAmmo && wpn)
		{
		  /////////// Full-auto
		  wpn.FullAuto = true;
		  if(wpn.wep_fireMode == wpn.wep_primary)
		    wpn.wep_ammoicon1 = "WEP5A";
	      else
	        wpn.wep_ammoicon1 = "WEP2E";
		  //////////// Mag size
		  wpn.cSize += 10;
		  pAmmo.MaxAmount += 10;
		  pmo.A_GiveInventory("BulletLoaded", pAmmo.MaxAmount);
		  //////////// Recoil
		  wpn.sXY = 1;
		  wpn.sZ = 1;
		  wpn.upAcc = true;
		  DFPlayer(pmo).bP_Up1 = true;
		}
	  }
	  else if(e.Name == "w_HandgunProtect")
	  {
	    let wpn = Handgun(pmo.FindInventory("Handgun"));
		if(wpn)
		{
		  ///////////// Durability upgrade
		  wpn.heatProtection = true;
		  if(wpn.wep_deg == 5)
			wpn.wep_deg = 3;
		  else
			wpn.wep_deg = 2;
		  ////////////// Ammo capacity
		  let pAmmo = pmo.FindInventory("PistolAmmo");
		  let ppAmmo = pmo.FindInventory("PierceAmmo");
		  if(pAmmo && ppAmmo)
		  {
		    pAmmo.MaxAmount += 15;
			ppAmmo.MaxAmount += 15;
		  }
		  ////////////// Explosion Resistance
		  DFPlayer(pmo).bResistExplode = true;
		  DFPlayer(pmo).bP_Up2 = true;
		}
	  }
	  else if(e.Name ~== "Psi_Handgun")
	  {
		let wpn = Handgun(pmo.FindInventory("Handgun"));
		if(wpn)
		{
		  ////////////// Fast switching
		  wpn.fast_switch = true;
		  ////////////// Absorb life
		  DFPlayer(pmo).bBluePower = true;
		  ////////////// Change icon
		  wpn.Icon = TexMan.CheckForTexture("WI_2P");
		}
		DFPlayer(pmo).bP_Up3 = true;
		
		let inv = pmo.FindInventory("Psi");
		if(inv)
		{
		  inv.MaxAmount -= 3;
		  if(inv.Amount > inv.MaxAmount)
		    inv.Amount = inv.MaxAmount;
		}
	  }
	  else if(e.Name ~== "w_ShotgunVel")
	  {
	    let wpn = Trenchgun(pmo.FindInventory("Trenchgun"));
		if(wpn)
		{
		  wpn.def_fire_snd = "SHT_FIR5";
		  wpn.hasAtom = true;
		  wpn.wep_deg = 0;
		  wpn.wep_cond = 100;
		  wpn.isInDanger = false;
		  wpn.snd = "SHT_FIR3";
		  DFPlayer(pmo).bS_Up1 = true;
		}
	  }
	  else if(e.Name ~== "w_ShotgunReload")
	  {
	    ////////////////// Extra ammo
		let sAmmo = pmo.FindInventory("ShotgunAmmo");
		if(sAmmo)
		  sAmmo.MaxAmount += 9;
		/////////////////// Full Tube 
		let sLoaded = pmo.FindInventory("ShellLoaded");
		if(sLoaded)
		{
		  sLoaded.MaxAmount += 3;
		}
		let wpn = Trenchgun(pmo.FindInventory("Trenchgun"));
		if(wpn)
		{
		  wpn.wep_psiCost = 2;
		}
		//////////////////// Dual Shell Reload & Psi Cost Reduction
		DFPlayer(pmo).bS_Up2 = true;
	  }
	  else if(e.Name ~== "Psi_Shotgun")
	  {
	    DFPlayer(pmo).bOrangePower = true;
		DFPlayer(pmo).bS_Up3 = true;
		////////////// Change icon
		let wpn = Trenchgun(pmo.FindInventory("Trenchgun"));
		if(wpn)
		  wpn.Icon = TexMan.CheckForTexture("WI_3P");
	  }
	  else if(e.Name ~== "w_SprayLethal")
	  {
	    let wpn = SprayCan(pmo.FindInventory("SprayCan"));
		if(wpn)
		{
		  wpn.hasLethal = true;
		  DFPlayer(pmo).bSp_Up1 = true;
		}
	  }
	  else if(e.Name ~== "w_SpraySave")
	  {
	    let wpn = SprayCan(pmo.FindInventory("SprayCan"));
		if(wpn)
		{
		  wpn.hasReduce = true;
		  wpn.gasType = "GasMissileBig";
		  DFPlayer(pmo).bSp_Up2 = true;
		}
	  }
	  else if(e.Name ~== "Psi_Spray")
	  {
	    DFPlayer(pmo).bRedPower = true;
		DFPlayer(pmo).bSp_Up3 = true;
		let inv = pmo.FindInventory("Psi");
		if(inv)
		{
		  inv.MaxAmount -= 6;
		  if(inv.Amount > inv.MaxAmount)
		    inv.Amount = inv.MaxAmount;
		}
		  
		////////////// Change icon
		let wpn = SprayCan(pmo.FindInventory("SprayCan"));
		if(wpn)
		  wpn.Icon = TexMan.CheckForTexture("WI_5P");
	  }
	  else if(e.Name ~== "Adjust_Attunement")
	  {
	    DFPlayer(pmo).tune++;
	    let inv = pmo.FindInventory("Attunement_Check");
		if(inv)
		{
		  if(DFPlayer(pmo).tune < DFPlayer(pmo).tune_max)
		    inv.Amount = 1;
		  else
		    inv.Amount = 0;
		}
	  }
	  else if(e.Name ~== "CloseShop")
	  {
	    pmo.A_StopSound();
	    if(DFPlayer(pmo).upgActr)
	    {
	      DFPlayer(pmo).upgActr.SetStateLabel("Death");
	      DFPlayer(pmo).upgActr = null;
	    }
	  }
	}
  }
}
