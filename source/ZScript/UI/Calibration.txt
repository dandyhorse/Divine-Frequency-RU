class CalibrationHandler : EventHandler
{
  int tic;
  bool active;

  static void Run()
  {
    let ev = CalibrationHandler(EventHandler.Find("CalibrationHandler"));
	if(ev)
	{
	  ev.tic = 0;
	  ev.active = true;
	  Menu.SetMenu("CalibrationMenu");
	}
  }
  
  static void RunVHSOption()
  {
    let ev = CalibrationHandler(EventHandler.Find("CalibrationHandler"));
	if(ev)
	  Menu.SetMenu("VHSOptionMenu");
  }
  
  static void RunScaleOption()
  {
    let ev = CalibrationHandler(EventHandler.Find("CalibrationHandler"));
	if(ev)
	  Menu.SetMenu("UIScaleOptionMenu");
  }
  
  override void WorldTick()
  {
    if(!active)
	  return;
	  
	Super.WorldTick();
	
	if(tic == 1)
	  EventHandler.SendNetworkEvent("CalibrationExit");
	else if(tic == 2)
	{
	  tic = 0;
	  active = false;
	}
	
	tic++;
  }
  
  override void NetworkProcess(ConsoleEvent e)
  {
    if(e.Name == "CalibrationExit")
    {
	  active = false;
	  
      let saveConfig = CVar.FindCvar("calibrate");
	  if(saveConfig)
	    saveConfig.SetBool(true);
		
	  let pmo = players[consoleplayer].mo;
	  if(pmo)
	    pmo.ACS_NamedExecute("Calibration_Done", 0);
	}
	else if(e.Name == "VHSOptionExit")
    {
	  let pmo = players[consoleplayer].mo;
	  if(pmo)
	    pmo.ACS_NamedExecute("VHSOption_Done", 0);
	}
	else if(e.Name == "ScaleOptionExit")
    {
	  let pmo = players[consoleplayer].mo;
	  if(pmo)
	    pmo.ACS_NamedExecute("ScaleOption_Done", 0);
	}
  }
  
  override void RenderOverlay(RenderEvent e)
  {
    if(active)
	{
	  if(tic >= 1)
	    Screen.Dim(Color(0,0,0), 1.0, 0, 0, Screen.GetWidth(), Screen.GetHeight());
	}
  }
}

Class VHSOptionMenuHandler : UIHandler
{
  String commandList;
  VHSOptionMenu menu;

  override void ButtonClickCommand(UIButton caller, Name command)
  {
	Array<String> cmd;
	cmd.Clear();
	commandList = command;
	commandList.Split(cmd, ":");

	if(cmd[0] ~== "PrevVHSOption")
	{
	  menu.vhs_option_index--;
	  if(menu.vhs_option_index < 0)
	    menu.vhs_option_index = 3;
		
	  menu.vhs_option_text.setText(menu.vhs_options[menu.vhs_option_index]);
	  menu.update_option = true;
	}
	else if(cmd[0] ~== "NextVHSOption")
	{
	  menu.vhs_option_index++;
	  if(menu.vhs_option_index > 3)
	    menu.vhs_option_index = 0;
		
	  menu.vhs_option_text.setText(menu.vhs_options[menu.vhs_option_index]);
	  menu.update_option = true;
	}
	else if(cmd[0] ~== "NextTearOption")
	{
	  if(menu.vhs_tear_value)
	    menu.vhs_tear_value = false;
	  else
	    menu.vhs_tear_value = true;
	  
	  menu.vhs_tear_text.setText(menu.vhs_tearing[menu.vhs_tear_value]);
	  menu.update_option = true;
	}
	else if(cmd[0] ~== "Close")
	{
	  EventHandler.SendNetworkEvent("VHSOptionExit");
	  Menu.Close();
	}
  }
}

Class UIScaleOptionMenuHandler : UIHandler
{
  String commandList;
  UIScaleOptionMenu menu;
  
  void UpdateMenu()
  {
    let cv = CVar.FindCVar("vid_scalefactor");
	if(cv)
	  cv.SetFloat(menu.scale_value);
		
    Menu.MenuSound("M_Click2");
	Menu.scale_text.setText(String.Format("%.1f", menu.scale_value));
  }

  override void ButtonClickCommand(UIButton caller, Name command)
  {
	Array<String> cmd;
	cmd.Clear();
	commandList = command;
	commandList.Split(cmd, ":");
	if(cmd[0] ~== "ScaleUp")
	{
	  menu.scale_value -= 0.1;
	  if(menu.scale_value < 0.3)
	    menu.scale_value = 1.4;
		
	  UpdateMenu();
	}
	else if(cmd[0] ~== "ScaleDown")
	{
	  menu.scale_value += 0.1;
	  if(menu.scale_value > 1.4)
	    menu.scale_value = 0.3;
		
	  UpdateMenu();
	}
	else if(cmd[0] ~== "Close")
	{
	  EventHandler.SendNetworkEvent("ScaleOptionExit");
	  Menu.Close();
	}
  }
}
