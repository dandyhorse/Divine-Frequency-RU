Class CodexMenu : UIGenericMenu
{
  CodexMenuHandler handler;
  Array<DF_Weapon> Weapons;
  UILabel wInfo, wStat, wAbility, wQuirk;
  UILabel wInfoHeader, wAbilityHeader ,wQuirkHeader;
  UIScrollContainer wModScroll, wCharScroll, wQuirkScroll;
  UIBoxTextures sbFG, sbBG, sbHov;
  Font fnt, efnt;
  
  override void Init(Menu parent)
  {
    Super.Init(parent);
		
	PlayerPawn pmo = players[consoleplayer].mo;
	fnt = Font.GetFont("DF_FONT");
	efnt = Font.GetFont("dialoguefont");
	handler = new("CodexMenuHandler");
    handler.menu = self;
		
	SetBaseResolution((720, 400));
	CreateMenuTabs();
	
	Weapons.Clear();

	for(Inventory item = DFPlayer(pmo).Inv; item; item = item.Inv)
	{
	  if(item is "DF_Weapon")
	    Weapons.Push(DF_Weapon(item));
	}
	
	for(int i = 0; i < Weapons.Size() -1; i++)
	{
	  if(DF_Weapon(Weapons[i]).wep_order > DF_Weapon(Weapons[i + 1]).wep_order)
	  {
	    DF_Weapon swap = DF_Weapon(Weapons[i]);
		Weapons[i] = Weapons[i + 1];
		Weapons[i + 1] = swap;
		i = -1;
	  }
	}
	
	DrawImageBackground();
	
	let list = UIListFrame(new("UIListFrame")).Create((2,66), (100,324), 0.0);
	list.pack(mainframe);
	
	let inactiveTexture = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/Bestiary/Button.png", false);
	let hoverTexture = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/Bestiary/ButtonHover.png", false);
	sbFG = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/ScrollFG.png", false);
	sbBG = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/ScrollBG.png", false);
	sbHov = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/ScrollHover.png", false);
	  
	for(int i = 0; i < Weapons.Size(); i++)
	{
	  String command = String.Format("PasteEntry:%s:%s:%s", 
	  DF_Weapon(Weapons[i]).wep_info, 
	  DF_Weapon(Weapons[i]).wep_ability, 
	  DF_Weapon(Weapons[i]).wep_quirk
	  );
	  
	  let btn = UIButton(new("UIButton")).Create(
	  (0, 0),
	  (100, 12),
	  DF_Weapon(Weapons[i]).wep_name,
	  handler,
	  command,
	  inactive : inactiveTexture,
	  hover: hoverTexture,
	  click: inactiveTexture,
	  fnt: fnt,
	  textscale: 0.45,
	  alignment: UIButton.AlignType_CenterLeft
	  );
	  btn.pack(list);
	}
	
	//Draw the Information header
	wInfoHeader = new('UILabel').Create
	(
	  (116, 86),
	  (197, 16),
	  text: "Information:",
	  fnt: fnt,
	  alignment: UILabel.AlignType_TopLeft,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.6
	);
	wInfoHeader.Pack(mainframe);
		
	//Draw the Information
	wInfo = new('UILabel').Create
	(
	  (116, 106),
	  (187, 252),
	  text: "",
	  fnt: fnt,
	  alignment: UILabel.AlignType_TopLeft,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.4
	);
	wInfo.Pack(mainframe);
	
	//Draw the Secondary Fire Header
	wAbilityHeader = new('UILabel').Create
	(
	  (321, 86),
	  (197, 16),
	  text: "Secondary Fire:",
	  fnt: fnt,
	  alignment: UILabel.AlignType_TopLeft,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.6
	);
	wAbilityHeader.Pack(mainframe);
	
	//Draw the Secondary Fire Info
	wAbility = new('UILabel').Create
	(
	  (321, 106),
	  (184, 252),
	  text: "",
	  fnt: fnt,
	  alignment: UILabel.AlignType_TopLeft,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.4
	);
	wAbility.Pack(mainframe);
	
	//Draw the Extra Intel
	wQuirkHeader = new('UILabel').Create
	(
	  (526, 86),
	  (197, 16),
	  text: "Additional Mechanics:",
	  fnt: fnt,
	  alignment: UILabel.AlignType_TopLeft,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.6
	);
	wQuirkHeader.Pack(mainframe);
	
	//Draw the Extra Intel
	wQuirk = New("UILabel").Create(
	  (526, 106),
	  (187, 252),
	  text: "",
	  fnt: fnt,
	  alignment: UILabel.AlignType_TopLeft,
	  wrap: true,
	  autoSize: true,
	  textScale: 0.4
	);
	wQuirk.Pack(mainframe);
	
	//Create the button to check upgrades
	let info_upgrade = UIButton(new("UIButton")).Create(
	(0, 0),
	(100, 12),
	"\cFEnhancements",
	handler, "Info_Upgrade",
	inactive : inactiveTexture,
	hover: hoverTexture,
	click: inactiveTexture,
	fnt: fnt,
	textscale: 0.45,
	alignment: UIButton.AlignType_CenterLeft);
	info_upgrade.pack(list);
  }
  
  override bool OnUIEvent(UIEvent ev)
  {
    if(ev.Type == UIEvent.Type_KeyDown)
	  if(ev.KeyString ~== "	")
		Close();
		
	if(ev.type == UIEvent.Type_RButtonDown)
	  Close();
		  
	return Super.OnUIEvent(ev);
  }
  
  override void Drawer()
  {
    Screen.Dim(Color(8, 8, 8), 0.6, 0, 0, Screen.GetWidth(), Screen.GetHeight());
    Super.Drawer();
  }
  
  void DrawImageBackground()
  {
    UIImage sb = UIImage(new("UIImage")).Create((0, 64), (104, 328), image: "Graphics/Codex/SelectionBar.png");
	sb.Pack(mainframe);
	
	UIImage bi1H = UIImage(new("UIImage")).Create((112, 86), (197, 16), image: "Graphics/Codex/Bestiary/bInfoHeader.png");
	bi1H.Pack(mainframe);
	UIImage bi1 = New("UIIMage").Create((112, 104), (197, 256), "Graphics/Codex/Bestiary/bInfo.png");
	bi1.Pack(mainframe);

	UIImage bi2H = UIImage(new("UIImage")).Create((317, 86), (197, 16), image: "Graphics/Codex/Bestiary/bInfoHeader.png");
	bi2H.Pack(mainframe);
	UIImage bi2 = New("UIIMage").Create((317, 104), (197, 256), "Graphics/Codex/AbilityBG.png");
	bi2.Pack(mainframe);
	
	UIImage bi3H = UIImage(new("UIImage")).Create((522, 86), (197, 16), image: "Graphics/Codex/Bestiary/bInfoHeader.png");
	bi3H.Pack(mainframe);
	UIImage bi3 = New("UIIMage").Create((522, 104), (197, 256), "Graphics/Codex/Bestiary/bInfo.png");
	bi3.Pack(mainframe);
  }
  
  void CreateMenuTabs()
  {
    UIBoxTextures inactiveTexture = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/TabInactive.png", false);
	UIBoxTextures hoverTexture = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/TabHover.png", false);
	
	UIFrame tabFrame = UIFrame(new("UIFrame")).Create((199, 0), (322, 24));
	tabFrame.Pack(mainframe);
	
	UIImage tbg = UIImage(new("UIImage")).Create((0, 0), (322, 24), image: "Graphics/Codex/TabBG.png");
	tbg.Pack(tabFrame);
	
	UIButton inv = UIButton(new("UIButton")).Create((0, 0), (64, 24), "Inventory", handler, "ChangeMenu:InventoryMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	inv.pack(tabFrame);
	
	UIButton wep = UIButton(new("UIButton")).Create((64, 0), (64, 24), "Weapons", handler, "ChangeMenu:CodexMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	wep.pack(tabFrame);
	
	UIButton bes = UIButton(new("UIButton")).Create((128, 0), (64, 24), "Bestiary", handler, "ChangeMenu:BestiaryMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	bes.pack(tabFrame);
	
	UIButton note = UIButton(new("UIButton")).Create((192, 0), (64, 24), "Map", handler, "ChangeMenu:MapMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	note.pack(tabFrame);
	
	UIButton help = UIButton(new("UIButton")).Create((256, 0), (64, 24), "Help", handler, "ChangeMenu:HelpMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	help.pack(tabFrame);
  }
}

Class CodexMenuHandler : UIHandler
{
  String commandList;
  CodexMenu menu;
  
  void Print_Mod()
  {
    if(menu.wModScroll)
	  menu.wModScroll.Unpack();
	  
    string w_mod;
	let pmo = DFPlayer(players[consoleplayer].mo);
	if(!pmo)
	  return;
	
	if(pmo.bP_Up1)
	  w_mod.AppendFormat(StringTable.Localize("$NF_FullAuto"));
	  
	if(pmo.bP_Up2)
	  w_mod.AppendFormat(StringTable.Localize("$NF_Dur"));
	
	if(pmo.bP_Up3)
	  w_mod.AppendFormat(StringTable.Localize("$NF_PsiBlue"));
	
	if(pmo.bS_Up1)
	  w_mod.AppendFormat(StringTable.Localize("$NF_TV"));
	
	if(pmo.bS_Up2)
	  w_mod.AppendFormat(StringTable.Localize("$NF_Tube"));
	
	if(pmo.bS_Up3)
	  w_mod.AppendFormat(StringTable.Localize("$NF_PsiOrange"));
	
	if(pmo.bSp_Up1)
	  w_mod.AppendFormat(StringTable.Localize("$NF_Lethal"));
	
	if(pmo.bSp_Up2)
	  w_mod.AppendFormat(StringTable.Localize("$NF_Chem"));
	
	if(pmo.bSp_Up3)
	  w_mod.AppendFormat(StringTable.Localize("$NF_PsiRed"));
	
	BrokenLines Lines = menu.fnt.BreakLines(w_mod, int(187 * 2.5));
	double sAreaHeight = Lines.Count() * (menu.fnt.GetHeight() * 0.4);
	menu.wModScroll = new("UIScrollContainer").Create(
	(116, 106), (191, 252),
	4, sAreaHeight, 12.0,
	menu.sbFG,
	scrollBarHover: menu.sbHov,
	scrollBarClick: menu.sbFG,
	scrollBg: menu.sbBG,
	scrollArea: menu.wInfo);
	menu.wModScroll.Pack(menu.mainframe);
	menu.wInfo.setText(w_mod);
  }
  
  void Print_Char()
  {
    if(menu.wCharScroll)
	  menu.wCharScroll.Unpack();
	
	string w_char;
	Array<String> upg_list;
	let pmo = DFPlayer(players[consoleplayer].mo);
	if(pmo)
	{
	  string upg_string = pmo.char_upgrade;
	  upg_string.Split(upg_list, ":");
	}
	
	for(int i = 0; i < upg_list.Size(); i++)
	{
	  string localized_str = StringTable.Localize(upg_list[i]);
	  w_char.AppendFormat(localized_str);
	}
	
	BrokenLines Lines = menu.fnt.BreakLines(w_char, int(187 * 2.5));
	double sAreaHeight = Lines.Count() * (menu.fnt.GetHeight() * 0.4);
	menu.wCharScroll = new("UIScrollContainer").Create(
	(321, 106), (191, 252),
	4, sAreaHeight, 12.0,
	menu.sbFG,
	scrollBarHover: menu.sbHov,
	scrollBarClick: menu.sbFG,
	scrollBg: menu.sbBG,
	scrollArea: menu.wAbility);
	menu.wCharScroll.Pack(menu.mainframe);
	menu.wAbility.setText(w_char);
  }
  
  void Print_Unique()
  {
	let pmo = DFPlayer(players[consoleplayer].mo);
	if(!pmo)
	  return;
	  
    string w_unique;
	
	if(pmo.bUQ_Mag)
	  w_unique.AppendFormat(StringTable.Localize("$NF_ADB"));
	
	if(pmo.bUQ_DualFire)
	  w_unique.AppendFormat(StringTable.Localize("$NF_Prism"));
	  
	menu.wQuirk.setText(w_unique);
  }
  
  void Display_Upgrade()
  {
    Menu.MenuSound("M_Click1");
    menu.wInfoHeader.setText("\cFWeapon Modifications:");
	menu.wAbilityHeader.setText("\cFCharacter Ascension:");
	menu.wQuirkHeader.setText("\cFUnique:");
	menu.wInfo.setText("");
	menu.wAbility.setText("");
	menu.wQuirk.setText("");
	//Print weapon mods
	Print_Mod();
	Print_Char();
	Print_Unique();
  }

  override void ButtonClickCommand(UIButton caller, Name command)
  {
	Array<String> cmd;
	cmd.Clear();
	commandList = command;
	commandList.Split(cmd, ":");

	if(cmd[0] ~== "ChangeMenu")
	{
	  if(cmd[1] != "CodexMenu")
	  {
	    Menu.Close();
	    Menu.SetMenu(cmd[1]);
		Menu.MenuSound("M_Click1");
	  }
	}
	  
	if(cmd[0] == "PasteEntry")
	{
	  menu.wInfoHeader.setText("Information:");
	  menu.wAbilityHeader.setText("Secondary Fire:");
	  menu.wQuirkHeader.setText("Additional Mechanics:");
		
	  menu.MenuSound("M_Click2");
	  if(cmd[1] ~== "$DB_Info")
	  {
	    menu.wInfo.setFont(menu.efnt);
		menu.wInfo.setTextScale(0.343);
		menu.wInfo.setLineSpacing(2.0);
		menu.wQuirk.setFont(menu.efnt);
		menu.wQuirk.setTextScale(0.343);
	  }
	  else
	  {
	    menu.wInfo.setFont(menu.fnt);
		menu.wInfo.setTextScale(0.4);
		menu.wInfo.setLineSpacing(0);
		menu.wQuirk.setFont(menu.fnt);
		menu.wQuirk.setTextScale(0.4);
	  }
	  menu.wInfo.setText(StringTable.Localize(cmd[1]));
	  menu.wAbility.setText(StringTable.Localize(cmd[2]));
	  menu.wQuirk.setText(StringTable.Localize(cmd[3]));
	  
	  if(menu.wModScroll)
	  {
	    menu.wModScroll.Unpack();
	    String wepinfo_string = StringTable.Localize(cmd[1]);
	    BrokenLines Lines = menu.fnt.BreakLines(wepinfo_string, int(187 * 2.5));
	    double sAreaHeight = Lines.Count() * (menu.fnt.GetHeight() * 0.4);
	    menu.wModScroll = new("UIScrollContainer").Create(
	    (116, 106),
	    (191, 252),
	    4, sAreaHeight, 12.0,
	    menu.sbFG,
	    scrollBarHover: menu.sbHov,
	    scrollBarClick: menu.sbFG,
	    scrollBg: menu.sbBG,
	    scrollArea: menu.wInfo);
	    menu.wModScroll.Pack(menu.mainframe);
	  }
	  
	  if(menu.wCharScroll)
	  {
	    menu.wCharScroll.Unpack();
	    String altfire_string = StringTable.Localize(cmd[2]);
	    BrokenLines Lines = menu.fnt.BreakLines(altfire_string, int(187 * 2.5));
	    double sAreaHeight = Lines.Count() * (menu.fnt.GetHeight() * 0.4);
	    menu.wCharScroll = new("UIScrollContainer").Create(
	    (321, 106),
	    (191, 252),
	    4, sAreaHeight, 12.0,
	    menu.sbFG,
	    scrollBarHover: menu.sbHov,
	    scrollBarClick: menu.sbFG,
	    scrollBg: menu.sbBG,
	    scrollArea: menu.wAbility);
	    menu.wCharScroll.Pack(menu.mainframe);
	  }
	  
	  if(menu.wQuirkScroll)
	  {
	    menu.wQuirkScroll.Unpack();
	    String quirk_string = StringTable.Localize(cmd[3]);
	    BrokenLines Lines = menu.fnt.BreakLines(quirk_string, int(187 * 2.5));
	    double sAreaHeight = Lines.Count() * (menu.fnt.GetHeight() * 0.4);
	    menu.wQuirkScroll = new("UIScrollContainer").Create(
	    (526, 106),
	    (191, 252),
	    4, sAreaHeight, 12.0,
	    menu.sbFG,
	    scrollBarHover: menu.sbHov,
	    scrollBarClick: menu.sbFG,
	    scrollBg: menu.sbBG,
	    scrollArea: menu.wQuirk);
	    menu.wQuirkScroll.Pack(menu.mainframe);
	  }
	}
	else if(cmd[0] ~== "Info_Upgrade")
	  Display_Upgrade();
  }
}
