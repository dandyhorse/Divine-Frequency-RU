 Class MapMenu : UIGenericMenu
{
  UIImage map_img, map_pos;
  MapMenuHandler handler;
  Font fnt;
  String img_path, img_hl;
  double hl_alpha, hl_alpha_threshold, hl_incr;

  override void Init(Menu parent)
  {
    Super.Init(parent);
	
	PlayerPawn pmo = players[consoleplayer].mo;
	if(pmo)
	{
	  img_path = DFPlayer(pmo).map_item;
	  img_hl = DFPlayer(pmo).map_highlight;
	  hl_alpha_threshold = DFPlayer(pmo).map_threshold;
	  hl_incr = (1.0 - hl_alpha_threshold) / 5.0;
	  if(img_path == "Graphics/Maps/Map_00.png")
	  {
	    img_hl = "";
	  }
	}
	
	hl_alpha = 1.0;
	fnt = Font.GetFont("DF_FONT");
	handler = new("MapMenuHandler");
    handler.menu = self;
	SetBaseResolution((720, 400));
	map_img = UIImage(new("UIImage")).Create((0,8), (720, 400), img_path, UIImage.AlignType_Center, (0.38,0.38));
	map_img.Pack(mainframe);
	if(img_hl != "")
	{
	  map_pos = UIImage(new("UIImage")).Create((0,8), (720, 400), img_hl, UIImage.AlignType_Center, (0.38,0.38));
	  map_pos.Pack(mainframe);
	}
	CreateMenuTabs();
  }
  
  override void Ticker()
  {
    if(hl_alpha > hl_alpha_threshold)
	{
	  hl_alpha -= hl_incr;
	  if(map_pos)
	    map_pos.setAlpha(hl_alpha);
	}
	
    Super.Ticker();
  }
  
  override bool OnUIEvent(UIEvent ev)
  {
    if(ev.Type == UIEvent.Type_KeyDown)
	  if(ev.KeyString ~== "	")
		Close();
	
	if(ev.type == UIEvent.Type_RButtonDown)
	  Close();
		  
	return Super.OnUIEvent(ev);
  }
  
  void CreateMenuTabs()
  {
    UIBoxTextures inactiveTexture = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/TabInactive.png", false);
	UIBoxTextures hoverTexture = UIBoxTextures(new('UIBoxTextures')).createSingleTexture("Graphics/Codex/TabHover.png", false);
	
	UIFrame tabFrame = UIFrame(new("UIFrame")).Create((199, 0), (322, 24));
	tabFrame.Pack(mainframe);
	
	UIImage tbg = UIImage(new("UIImage")).Create((0, 0), (322, 24), image: "Graphics/Codex/TabBG.png");
	tbg.Pack(tabFrame);
	
	UIButton inv = UIButton(new("UIButton")).Create((0, 0), (64, 24), "Inventory", handler, "ChangeMenu:InventoryMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	inv.pack(tabFrame);
	
	UIButton wep = UIButton(new("UIButton")).Create((64, 0), (64, 24), "Weapons", handler, "ChangeMenu:CodexMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	wep.pack(tabFrame);
	
	UIButton bes = UIButton(new("UIButton")).Create((128, 0), (64, 24), "Bestiary", handler, "ChangeMenu:BestiaryMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	bes.pack(tabFrame);
	
	UIButton note = UIButton(new("UIButton")).Create((192, 0), (64, 24), "Map", handler, "ChangeMenu:MapMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	note.pack(tabFrame);
	
	UIButton help = UIButton(new("UIButton")).Create((256, 0), (64, 24), "Help", handler, "ChangeMenu:HelpMenu",
	inactive : null,
	hover: hoverTexture,
	click: null,
	fnt: fnt,
	textscale: 0.5,
	alignment: UIButton.AlignType_Center
	);
	help.pack(tabFrame);
  }
}

Class MapMenuHandler : UIHandler
{
  String commandList;
  MapMenu menu;
  
  override void ButtonClickCommand(UIButton caller, Name command)
  {
	Array<String> cmd;
	cmd.Clear();
	commandList = command;
	commandList.Split(cmd, ":");

	if(cmd.size() < 2)
	  return;
	  
	if(cmd[0] ~== "ChangeMenu")
	{
	  if(cmd[1] != "MapMenu")
	  {
	    Menu.Close();
	    Menu.SetMenu(cmd[1]);
		Menu.MenuSound("M_Click1");
	  }
	}
  }
}
