Class bInfo : Inventory
{
  String entryImage, entryLore, entryEcho, entryText, entryRes;
  property Image: entryImage;
  property Lore: entryLore;
  property Text: entryText;
  property Res: entryRes;
  property Echo: entryEcho;
  
  Default
  {
    Inventory.Amount 1;
	Inventory.MaxAmount 1;
	+INVENTORY.UNDROPPABLE
	+INVENTORY.UNTOSSABLE
  }
}

Class BestiaryPack : Inventory
{
  override void AttachToOwner(Actor other)
  {
    other.A_GiveInventory("b_Enlighten1", 1);
	other.A_GiveInventory("b_Enlighten2", 1);
	other.A_GiveInventory("b_Enlighten3", 1);
	other.A_GiveInventory("b_AnThit", 1);
	other.A_GiveInventory("b_Colony", 1);
	other.A_GiveInventory("b_Enlighten4", 1);
	other.A_GiveInventory("b_Moulter", 1);
	other.A_GiveInventory("b_Corvid", 1);
	other.A_GiveInventory("b_Ghost", 1);
	other.A_GiveInventory("b_Proxy", 1);
	other.A_GiveInventory("b_Enlighten5", 1);
	other.A_GiveInventory("b_GKC", 1);
	other.A_GiveInventory("b_Kin", 1);
	Destroy();
  }
  
  Default
  {
    Inventory.Amount 1;
  }
}
