  override bool OnUIEvent(UIEvent ev)
  {
    if(ev.Type == UIEvent.Type_KeyDown)
	  if(ev.KeyString ~== "	")
		Close();
		
	if(ev.type == UIEvent.Type_RButtonDown)
	  Close();
		  
    for(int i = 0; i < bootuns.Size(); i++)
	{
	  if(bootuns[i].isHovered())
	  {
		vector2 pos = mainframe.screenToRel((ev.MouseX, ev.MouseY));
		GPopUp.setPos(pos + (8, 8));
		GPopUp.show();
		break;
	  }
		
	  if(i == bootuns.Size() -1 && !bootuns[i].isHovered())
	  {
	    GInfo.setText("");
		GPopUp.hide();
	  }
	}
	  
	return Super.OnUIEvent(ev);
  }

 Class BestiaryMenuHandler : UIHandler
{
  String commandList;
  BestiaryMenu menu;
  
  override void elementHoverChanged(UIElement caller, Name command, bool unhovered)
  {
	if(!unhovered)
	{
	  Array<String> cmd;
	  cmd.Clear();
	  String commandList = command;
	  commandList.Split(cmd, ":");
	  
	  if(cmd.size() < 2)
	    return;
	  
	  if(cmd[0] == "Wiki")
	    menu.GInfo.setText(StringTable.Localize(cmd[1]));
	}
  }

  override void ButtonClickCommand(UIButton caller, Name command)
  {
	Array<String> cmd;
	cmd.Clear();
	commandList = command;
	commandList.Split(cmd, ":");

	if(cmd.size() < 2)
	  return;
	  
	if(cmd[0] ~== "ChangeMenu")
	{
	  if(cmd[1] != "BestiaryMenu")
	  {
	    Menu.Close();
	    Menu.SetMenu(cmd[1]);
		Menu.MenuSound("M_Click1");
	  }
	}
	  
	if(cmd[0] == "PasteEntry")
	{
	  menu.MenuSound("M_Click2");
	  menu.bImage.setImage(cmd[1]);
	  menu.bLore.setText(StringTable.Localize(cmd[2]));
	  menu.bPhy.setText(StringTable.Localize(cmd[3]));
	  String infoContent = StringTable.Localize(cmd[2]);

	  if(menu.nScroll)
	    menu.nScroll.Unpack();
	  
	  BrokenLines Lines = menu.fnt.BreakLines(infoContent, int(189 * 2.45));
	  double sAreaHeight = Lines.Count() * (menu.fnt.GetHeight() * 0.4);
	  
	  menu.nScroll = new("UIScrollContainer").Create(
	    (526, 106),
	    (189, 248),
	    4, sAreaHeight, 12.0,
	    menu.sbFG,
	    scrollBarHover: menu.sbHov,
	    scrollBarClick: menu.sbFG,
	    scrollBg: menu.sbBG,
	    scrollArea: menu.bLore
	  );
	  menu.nScroll.Pack(menu.mainframe);
	}
  }
}
