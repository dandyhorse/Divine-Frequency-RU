Class ElectroArc1 : Actor
{
  Actor ptr;
  
  override void PostBeginPlay()
  {
    if(ptr)
	{
	  double length = Distance2D(ptr) + (0.5 * ptr.radius);
	  double height = pos.z - (ptr.pos.z + (ptr.height / 2.0));
	  double scale = Distance3D(ptr) / 512.0;
	  
	  angle = AngleTo(ptr);
	  pitch = (height / length) * 100.0;
	  A_SetScale(scale);

Class StunToken : Inventory
{
  int lifeTime;
  
  override void BeginPlay()
  {
    lifeTime = 210;
  }
  
   override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
    other.A_StartSound("Stun/Active", 24, CHANF_LOOP , 1.0);
  }
  
  override void DetachFromOwner()
  {
    if(owner)
	  owner.A_StopSound(24);
  }
  
  override void DoEffect()
  {
    Super.DoEffect();
	
	if(owner && owner.health > 0)
	  SetOrigin(owner.pos, false);
	else
	  Destroy();
	
	if(lifeTime <= 0)
	  Destroy();
	  
	if(owner && lifeTime % 2 == 0)
	{
	  for(int i = 0; i < 3; i++)
	    owner.A_SpawnParticleEx(
			"FFFFFF",
			TexMan.CheckForTexture("SPARKZ"),
			style: STYLE_Add,
			flags: SPF_ROLL|SPF_FULLBRIGHT|SPF_RELATIVE|SPF_LOCAL_ANIM,
			lifetime: 10,
			size: random(24,32),
			angle: 0,
			xoff: random(0-owner.radius, owner.radius),
			yoff: random(0-owner.radius, owner.radius),
			zoff:random(8, (owner.height-4)),
			velx: FRandom (-0.5,0.5),
			vely: FRandom (-0.5,0.5),
			velz: FRandom (-0.5,0.5),
			accelx: 0,
			accely: 0,
			accelz: 0,
			startalphaf: 1.0,
			fadestepf: 0.0,
			sizestep: 0.1,
			startroll: random(0,359),
			rollvel: 0,
			rollacc: 0
		  );
		
	  int chance = random(0,256);
	  if(chance <= 24)
	  {
	    owner.A_StartSound("ElecPop1", CHAN_AUTO, attenuation: 0.5);
		owner.A_SpawnItemEx("StunSparks", flags: SXF_NOCHECKPOSITION|SXF_SETMASTER);
	    owner.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("STFLA0"), STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE, 19, 160, 0, zoff: 0.5 * owner.height, startalphaf: 0.45, fadestepf: 0.03, sizestep: 8, startroll: random(0,359));
	    owner.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("SPARK2"), STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE|SPF_LOCAL_ANIM, 19, 512, 0, random(-0.6 * owner.radius, 0.6 * owner.radius), random(-0.6 * owner.radius, 0.6 * owner.radius), random(24, 0.8 * owner.height), startroll: random(0,359));
	    owner.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("SPARK3"), STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE|SPF_LOCAL_ANIM, 19, 512, 0, random(-0.6 * owner.radius, 0.6 * owner.radius), random(-0.6 * owner.radius, 0.6 * owner.radius), random(24, 0.8 * owner.height), startroll: random(0,359));
	    owner.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("SPARK4"), STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE|SPF_LOCAL_ANIM, 19, 512, 0, random(-0.6 * owner.radius, 0.6 * owner.radius), random(-0.6 * owner.radius, 0.6 * owner.radius), random(24, 0.8 * owner.height), startroll: random(0,359));
	  }
	}
	
	lifeTime--;
  }
  
  Default

  }

Class StunTokenSoft : Inventory
{
  int lifeTime;
  
  void LayerFlash(Actor actr, int style, double alpha, string col)
  {
    if(actr)
	{
      double ang = actr.AngleTo(players[consoleplayer].mo);
      Vector3 SpawnPos = Vec3Offset(1 * cos(ang), 1 * sin(ang), 0);
      let layer = Spawn("StunVictimLayer", SpawnPos);
	  if(layer)
	  {
	    layer.A_SetRenderStyle(alpha, style);
	    layer.SetShade(col);
	    layer.angle = actr.angle;
	    layer.frame = actr.frame;
	    layer.scale = actr.scale;
	    layer.sprite = actr.sprite;
	    layer.master = actr;
	  }
	}
  }
  
  override void BeginPlay()
  {
    lifeTime = 210;
  }
  
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
    other.A_StartSound("Stun/Active", 24, CHANF_LOOP , 1.0);
  }
  
  override void DetachFromOwner()
  {
    if(owner)
	  owner.A_StopSound(24);
  }
  
  override void DoEffect()
  {
    Super.DoEffect();
	
	if(owner)
	{
	  if(owner.health <= 0)
	    return;
	
	  if(GetAge() == 1)
	    LayerFlash(owner, STYLE_AddStencil, 0.7, "0066CC");
	  else if(GetAge() == 5)
	    LayerFlash(owner, STYLE_AddStencil, 0.7, "CCE5FF");
	  else if(GetAge() == 8)
	    LayerFlash(owner, STYLE_AddStencil, 0.7, "0066CC");
	  else if(GetAge() == 12)
	    LayerFlash(owner, STYLE_AddStencil, 0.7, "0066CC");
	
	  if(GetAge() < 4)
	  {
	    for(int i = 0; i < 8; i++)
	    {
	      double ang = random(-20,20);
		  if(players[consoleplayer].mo)
		    ang += AngleTo(players[consoleplayer].mo);
		  
	      owner.A_SpawnItemEx("ElectricSpark",
		  random(-owner.radius, owner.radius), random(-owner.radius, owner.radius), random(32, owner.height),
		  frandom(11.0,15.6), frandom(-5.0,5.0), frandom(4.0,10.0), ang);
	    }
	  }
	  
	  if(lifeTime % 2 == 0)
	  {
	    bool add = random(0,1);
	    int amount = 3;
	    int siz = random(24,32);
	    if(owner.height > 150)
	    {
	      amount = 6;
		  siz = random(32,48);
	    }
	    for(int i = 0; i < amount; i++)
	    {
	      owner.A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("SPARKZ"),
		  add ? STYLE_Add : STYLE_Normal, SPF_ROLL|SPF_FULLBRIGHT|SPF_RELATIVE|SPF_LOCAL_ANIM, 10, siz, 0,
		  random(0-owner.radius, owner.radius), random(0-owner.radius, owner.radius), random(8, (owner.height-4)),
		  FRandom (-0.5,0.5), FRandom (-0.5,0.5), FRandom (-0.5,0.5),
		  startalphaf: 1.0, fadestepf: 0.0, sizestep: 0.1, startroll: random(0,359));
	    }
	  }
	}
	
	lifeTime--;
	
	if(lifeTime <= 0)
	  Destroy();
  }
  
  Default
  {
    Inventory.Amount 1;
  }
}
