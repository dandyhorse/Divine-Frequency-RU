Class Tool : CustomInventoryItem
{
  override void BeginPlay()
  {
    isTool = true;
	canUse = true;
  }
  
  Default
  {
    //$Category "Item/Tools"
	//$Color 1
  }
  
  States
  {
    Pickup:
	  TNT1 A 0 A_StartSound(invoker.PickupSnd, CHAN_AUTO);
	  Stop;
  }
}

Class ArmorKit : Consumable
{
  Actor layer;
  
  override void BeginPlay()
  {
    invName = "Blastema Cells | \cFEsoteric";
	invIcon = "Graphics/Inventory Icon/Cell.png";
	invText = "$ArmorRepair";
	invLore = "$ArmorRepairLore";
	invSnd = "GAUZEUP";
	invLog = "\cQBlastema Cells Case";
	PickupSnd = "ATD_Up";
	healAmount = 0;
  }
  
  Default
  {
	Tag "\cFBlastema Cells";
	Radius 15;
	Height 35;
	Scale 0.2;
	+FORCEXYBILLBOARD
  }

  States
  {
	Spawn:
      ABLT A -1;
      Stop;
    Use:
      TNT1 A 1
	  {
		A_UseConsumable("GAUZEUP");
		GiveInventory("BaseArmor", 50);
		
		GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		if(ev)
		{
		  ev.UpdateChat(String.Format("%s \cJuses Blastema Cells...", self.GetTag()));
		  ev.UpdateChat(String.Format("%s \cDgained \cJ50 Armor Points", self.GetTag()));
		}
	  }
	  Stop;
  }
}
