Class KeyBase : InventoryItem
{
  override void AttachToOwner(Actor other)
  {
    GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(other && ev)
	  ev.UpdateChat(String.Format("%s \cJpicked up a key.", other.GetTag()));
		  
    Super.AttachToOwner(other);
  }
  
  Default
  {
    Radius 15;
    Height 15;
	Inventory.Amount 1;
    Inventory.PickupMessage "";
    Inventory.PickupSound "Key_Up";
    +FLATSPRITE 
    +FLOORCLIP
    +SOLID
    +NOGRAVITY
    -INVENTORY.INVBAR
  }
  States
  {
    Use:
	  TNT1 A 0;
	  Fail;
  }
}
