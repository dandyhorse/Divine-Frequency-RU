Class Stim : Tool
{
  int amt, maxAmt;
  
  override bool TryPickup(in out Actor toucher)
  {
	if(toucher && toucher is "DFPlayer")
	  DFPlayer(toucher).fHealCount = amt;

	return Super.TryPickup(toucher);
  }
  
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
	
	let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(ev)
	  ev.PromptTutorial("Tip_Injector");
  }
  
  override void BeginPlay()
  {
    Super.BeginPlay();
    amt = 1;
	maxAmt = 1;
    invName = "\cGPsykhus Injector\cJ";
	invIcon = "Graphics/Inventory Icon/Stim.png";
	invText = "$Stim";
	invLore = "$StimLore";
	invSnd = "";
	PickupSnd = "Stim_Up";
  }
  
  Default
  {
    //$Category "Item/Portable"
	//$Color 1
    Tag "\cGPsykhus Injector";
	Radius 18;
	Height 27;
	Scale 0.2;
	Inventory.Amount 1;
    Inventory.MaxAmount 1;
	Inventory.PickupSound "";
	Inventory.PickupMessage "";
	+INVENTORY.KEEPDEPLETED
	+INVENTORY.UNDROPPABLE
	+COUNTITEM
	+FORCEXYBILLBOARD
  }
  States
  {
    Spawn:
	  TONC B -1 Light("AntidoteLight");
	  Stop;
    Use:
	  TNT1 A 0
	  {
		if(invoker.amt > 0)
		{
		  if(players[consoleplayer].cheats & (CF_FROZEN|CF_TOTALLYFROZEN))
	        return;
			
		  A_GiveInventory("Stamina", 999);
		  DFPlayer(self).armorStack = DFPlayer(self).armorMax;
		  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		  if(ev)
		  {
		    ev.UpdateChat(String.Format("· %s \cJuses \cGPsykhus Injector.", self.GetTag()));
			ev.UpdateChat(String.Format("· %s \cJis \cDEuphoric \cF(Massive Health Regeneration).", self.GetTag()));
			ev.UpdateChat("\cD+900 Stamina.");
			ev.UpdateChat(String.Format("\cD+%d Armor Stack.", DFPlayer(self).armorMax));
			ev.SendInterfaceEvent(0, "PortraitHealFlash");
		  }
		  
		  invoker.amt--;
		  if(self && self is "DFPlayer")
		    DFPlayer(self).fHealCount = invoker.amt;
			
		  //////////////////////////Do that good healing
		  let item = FindInventory("StimHeal");
	      if(item)
	        StimHeal(item).regenTimer = 0;
	      else
	        A_GiveInventory("StimHeal", 1);
		  
		  //////////////////////////Do some ocky ass shit
		  if(DFPlayer(self).bPhy_Mind)
		    GiveInventory("HealthUpgrade", 5);
			
		  /////////////////////////Pretty Effects
		  if(!players[consoleplayer].mo.bINVULNERABLE)
		    A_SetBlend("LightBlue", 0.3, 15);
			
		  let fx = FindInventory("TonicHealFX");
		  if(fx)
		    TonicHealFX(fx).Tic = 0;
		  else
		    A_GiveInventory("TonicHealFX", 1);
		}
	  }
	  Fail;
  }
}

Class StimHeal : Inventory
{
  int Tic;
  int regenTimer;
  
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
	
	if(other)
	  DFPlayer(other).buff_stim = true;
  }
  
  override void PostBeginPlay()
  {
    Tic = 0;
    regenTimer = 0;
    Super.PostBeginPlay();
  }
  
  override void DoEffect()
  {
	if(regenTimer == 100)
	{
	  if(owner)
	    DFPlayer(owner).buff_stim = false;
	  
	  Destroy();
	}
	
	if(Tic == 7)
	{
	  owner.GiveBody(50);
	  regenTimer++;
	  Tic = 0;
	}
	
    Tic++;
  }
  
  Default
  {
  Inventory.Amount 1;
  -INVENTORY.INVBAR
  }
}

Class Elixir : AmmoItem
{
  Default
  {
    Radius 16;
    Height 28;
	XScale 0.17;
	YScale 0.2;
    Tag "\cFLifeblood Elixir";
    Inventory.Amount 1;
    Inventory.PickupMessage "";
    Inventory.PickupSound "";;
	-FORCEXYBILLBOARD
	//+BRIGHT
  }
  States
  {
    Spawn:
	  ELIX AB 4 A_SpawnParticleEx("FF0000", TexMan.CheckForTexture("SPFLD0"), STYLE_AddShaded, SPF_FULLBRIGHT|SPF_RELATIVE, 35, 6, 0, random(-7,7), random(-7,7), random(0,18), velz: .2);
	  Loop;
	Pickup:
	  TNT1 A 0
	  {
	    let item = FindInventory("Stim");
		if(item)
		{
		  if(Stim(item).amt == Stim(item).maxAmt)
		  {
		    if(!CheckInventory("StimFullMsg", 1))
		      A_GiveInventory("StimFullMsg", 1);

			return invoker.ResolveState("Full");
		  }
		  else
		  {
		    Stim(item).amt += 1;
			DFPlayer(self).fHealCount = Stim(item).amt;
			A_SetBlend("Red", 0.2, 20);
			A_StartSound("Stim_Up2", CHAN_AUTO, volume: 1.0);
			GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
			if(ev)
			{
			  ev.UpdateItem(String.Format("+1 \cGPsykhus Injector Charge"));
			  ev.stimUpAlpha = 1.0;
			}
		  }
		}
		else
		{
		  if(!CheckInventory("NoStimMsg", 1))
		    A_GiveInventory("NoStimMsg", 1);

		  return invoker.ResolveState("Full");
		}

		return invoker.ResolveState(null);
	  }
	  TNT1 A 0 A_RailWait;
	  Stop;
  }
}

Class StimRefill : Inventory
{
  override void AttachToOwner(Actor other)
  {

  }
  
  Default

	Inventory.Amount 1;
  }
}

Class Psykhus : Actor
{
  Default
  {
    //$Category "Item/Consumables"
	//$Color 1
    Radius 4;
    Height 4;
    +NOINTERACTION
  }
  States
  {
    Spawn:
      TNT1 A 4 Light("CPillPlusLight")
	  {
	    let pmo = players[consoleplayer].mo;
		if(pmo)
		{
	      if(Distance3D(pmo) < 1500 && !CheckIfSeen())
	      {
	        A_SpawnItemEx("StimFX4", random(-24, 24), random(-24, 24), random(2, 40), 0, 0, 1);
	        A_SpawnItemEx("StimFX4", random(-24, 24), random(-24, 24), random(2, 40), 0, 0, 1);
	      }
		  if(Distance3D(pmo) <= 144)
		  {
		    let item = pmo.FindInventory("Stim");
			if(item && Stim(item).amt < Stim(item).maxAmt)
		      return ResolveState("Death");
			else if(item && Stim(item).amt == Stim(item).maxAmt)
			  if(!pmo.CheckInventory("StimFullMsg", 1))
			    pmo.A_GiveInventory("StimFullMsg", 1);
		  }
		}
		return ResolveState(null);
	  }
	  Wait;
	Death:
	  TNT1 A 1
	  {
	    let pmo = players[consoleplayer].mo;
		if(pmo)
		{
	      let item = pmo.FindInventory("Stim");
		  if(item)
		  {
			Stim(item).amt += 1;
			DFPlayer(pmo).fHealCount = Stim(item).amt;
			pmo.A_SetBlend("Light Blue", 0.3, 20);
			
			GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
			if(ev)
			{
			  ev.UpdateItem(String.Format("+1 \cGHealth Stim Charge"));
			  ev.stimUpAlpha = 1.0;
			}
			  
			A_StartSound("Stim_Up", CHAN_AUTO);

			for(int i = 0; i < 22; i++)
			  A_SpawnItemEx("StimFX4", random(-24, 24), random(-24, 24), random(2, 40), random(-4, 4), random(-4, 4), random(2, 3));
		  }
		}
	  }
	  Stop;
  }
}
