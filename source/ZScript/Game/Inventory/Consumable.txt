Class Consumable : CustomInventoryItem
{
  bool flash;
  int healAmount, flashTic;
  
  override void Tick()
  {
    Super.Tick();
	
	if(flash)
	{
	  if(flashTic == 55)
	  {
	    flashTic = 0;
	    let actr = Spawn("LootFlash", pos);
	    if(actr)
	    {
		  actr.scale = scale;
		  actr.frame = frame;
		  actr.sprite = sprite;
		  if(bFORCEXYBILLBOARD)
		    actr.bFORCEXYBILLBOARD = true;
		}
	  }
	  flashTic++;
	}
  }
  
  override bool Use(bool pickup)
  {
	bool cnd = Super.Use(pickup);
	if(cnd)
	  DFPlayer(owner).invCur--;
	  
	return cnd;
  }
  
  action void A_UseConsumable(string snd = "", string msg = "", string status = "")
  {
    A_StartSound(snd, CHAN_AUTO);
	A_SetBlend("White", 0.1, 12);
	
	double heal = invoker.healAmount * CVar.FindCVar("modf_heal").GetFloat();
	
	if(menuactive)
	{
	  HealThing(heal);
	  EventHandler.SendInterfaceEvent(0, "HUDUpdateHealth");
	}
	else
	  DFPlayer(self).DoHeal(heal);
	  
	GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(self && ev)
	{
	  if(msg != "")
	    ev.UpdateChat(String.Format("· %s \cJ%s \cF%s\cJ...", self.GetTag(), msg, invoker.GetTag()));
	  
	  if(status != "")
	    ev.UpdateChat(String.Format("· %s \cJ%s", self.GetTag(), status));
		
	  ev.UpdateChat(String.Format("\cD+%d Health.", heal));
	}
  }
  
  action void AddRad(int amount)
  {
    if(self && !DFPlayer(self).bStrOrigin_Buff)
	{
	  DFPlayer(self).SetRadiation(amount, true);
	  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
	    ev.UpdateChat(String.Format("\cR+%d Radiation.", amount));
	}
  }
  
  action void AddToxin(int amount)
  {
    if(self && !DFPlayer(self).bStrOrigin_Buff)
	{
	  DFPlayer(self).SetToxin(amount, true);
	  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
	    ev.UpdateChat(String.Format("\cR+%d Toxin.", amount));
	}
  }
  
  action void RemoveRad(int amount)
  {
    if(self)
	  DFPlayer(self).RadHeal(amount);
	  
	GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(ev)
	  ev.UpdateChat(String.Format("\cD¯%d Radiation.", amount));
  }
  
  action void RemoveToxin(int amount)
  {
    if(self)
	  DFPlayer(self).ToxHeal(amount);
	  
	GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(ev)
	  ev.UpdateChat(String.Format("\cD¯%d Toxin.", amount));
  }
  
  void HealthTax(int dmg)
  {
    if(owner && !DFPlayer(owner).bStrOrigin_Buff)
	{
      owner.A_DamageSelf(dmg, "Drowning", DMSS_NOPROTECT|DMSS_NOFACTOR);
	  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(owner && ev)
	    ev.UpdateChat(String.Format("\cR¯%d Health.", dmg));
	}
	else
	{
	  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
		ev.UpdateChat(String.Format("· %s \cJresists all negative effect.", owner.GetTag()));
	}
  }

  Default
  {
    //$Category "Item/Consumables"
	//$Color 1
    Inventory.MaxAmount 60;
	Inventory.InterHubAmount 60;
    +COUNTITEM
  }
}

Class DirtyWaterFX : PuzzleItem
{
  int Tic, amt; //I'm intentionally leaving out Damage stacking because i am merifcul lel
  
  override void PostBeginPlay()
  {
    Tic = 0;
    Super.PostBeginPlay();
  }
  
  override void DoEffect()
  {
	if(Tic == 88)
	{
	  owner.A_StartSound("Poison1", CHAN_AUTO, 0, 0.4);
	  owner.A_SetBlend("#490013", 0.5, 175);
	  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(owner && !DFPlayer(owner).bStrOrigin_Buff)
	  {
        owner.A_DamageSelf(5, "Drowning", DMSS_NOPROTECT|DMSS_NOFACTOR);
	    if(ev)
	    {
		  ev.UpdateChat("\cR¯5 Health.");
		  ev.UpdateChat(String.Format("\cR+%d Toxin.", amt));
		  DFPlayer(owner).SetToxin(amt, true);
		}
	  }
	  else
	  {
		GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		if(ev)
		  ev.UpdateChat(String.Format("· %s \cJresists all negative effect.", owner.GetTag()));
	  }
	  
	  if(CountInv("DirtyWaterBuff") == 3)
	  {
	    DFPlayer(owner).toxMax += 50;
		if(ev)
		{
		  ev.UpdateChat(String.Format("%s \cJdeveloped an unsual tolerance to Toxin!!", owner.GetTag()));
		  ev.UpdateChat("\cFToxin's max. threshold increased by 50.");
		}
	  }
		
	  Destroy();
	}
	
	Tic++;
	
	Super.DoEffect();
  }
  
  Default
  {
  Inventory.Amount 1;
  Inventory.MaxAmount 1;
  -INVENTORY.INVBAR
  }
}

	invIcon = "REPAIR03";
	invText = "$GunRepairHelp";
	invLore = "$GunRepairHelpLore";
	invSnd = "Repair2";
	PickupSnd = "Rep_Up";
	canUse = false;
  }
  
  override bool TryPickup(in out Actor toucher)
  {
	if(CanPickup(toucher))
	{
	  if(toucher && toucher is "DFPlayer")
	    DFPlayer(toucher).bRepairBoost = true;
			
	  Super.TryPickup(toucher);
	  return true;
	}
	
	return false;
  }
  
  override void DetachFromOwner()
  {
    if(owner && owner is "DFPlayer")
	  DFPlayer(owner).bRepairBoost = false;
	  
	Super.DetachFromOwner();
  }
  
  override void OnDrop(Actor dropper)
  {
    if(dropper && dropper is "DFPlayer")
	  DFPlayer(dropper).bRepairBoost = false;
	  
	Super.OnDrop(dropper);
  }

Class Lube : Consumable
{
  override void BeginPlay()
  {
    Super.BeginPlay();
	
    invName = "Gun Oil";
	invIcon = "Graphics/Inventory Icon/GunOil.png";
	invText = "$GunOil";
	invLore = "$GunOilLore";
	invSnd = "GOil_Use";
	PickupSnd = "GOil_Up";
  }
  
  action void ApplyOil(DF_Weapon wpn)
  {
	if(wpn.wep_protectShot == wpn.wep_protectMax)
	{
	  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
	    ev.UpdateChat(String.Format("· %s \cJwasted some perfectly good \cFGun Oil\cJ...", self.GetTag()));
	}
	else
	{
	  wpn.wep_protectShot = wpn.wep_protectMax;
	  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
		ev.UpdateChat(String.Format("· %s \cJapplies \cFGun Oil\cJ to \cF%s.", self.GetTag(), wpn.GetTag()));
	}
  }
  
  Default
  {
	Inventory.MaxAmount 60;
	Tag "Gun Oil";
	Radius 13;
	Height 9;
	XScale 0.36;
	YScale 0.3;
	+FORCEXYBILLBOARD
  }
  States
  {
    Spawn:
	  GOIL C 0
	  {
	    let pmo = DFPlayer(players[consoleplayer].mo);
		if(pmo && pmo.bBase_Wit)
		{
	      let actr = Spawn("LootFlash", invoker.pos);
		  if(actr)
		  {
		    actr.scale = invoker.scale;
		    actr.frame = invoker.frame;
		    actr.sprite = invoker.sprite;
		    if(bFORCEXYBILLBOARD)
		      actr.bFORCEXYBILLBOARD = true;
		  }
		}
	  }
	  GOIL C 55;
	  Loop;
	Use:
	  TNT1 A 0
	  {
	    A_StartSound("GOil_Use", CHAN_AUTO);
		
		GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		if(ev)
		{
		  DF_Weapon wpn = DF_Weapon(player.ReadyWeapon);
		  if(wpn)
			ApplyOil(wpn);
		}
	  }
	  Stop;
  }
}
