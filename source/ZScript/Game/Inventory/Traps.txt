Class ElectroMine : Actor
{
  Actor fakeTrap;
  int power;
  
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	fakeTrap = Spawn("VoltPickup", pos);
	if(fakeTrap)
	{
	  VoltPickup(fakeTrap).charge = power;
	  VoltPickup(fakeTrap).link = self;
	}
  }
  
  override void Tick()
  {
    if(pos.z == floorz && power > 0)
	  if(GetAge() % 4 == 0)
	    A_SpawnItemEx("ElectroShockFX", random(-160, 160), random(-160, 160), random(0,8), zvel: frandom(0.1, 0.5));
	
	Super.Tick();
  }
  
  void DeliverShock(Actor mo)
  {
	A_StartSound("EMP_03", CHAN_AUTO, attenuation: 0.5);
					
	Actor arc1 = Spawn("ElectroArc1", pos + (0,0,16));
	if(arc1)
	  ElectroArc1(arc1).ptr = mo;
					  
	Actor arc2 = Spawn("ElectroArc2", pos + (0,0,16));
	if(arc2)
	  ElectroArc2(arc2).ptr = mo;
	
	if(!DF_Monster(mo).bStunResist)
	{
	  mo.A_StartSound("ElectroMine/Explode", CHAN_AUTO, volume: 0.5, attenuation: 0.4);
	  //mo.SetStateLabel("Stunned");
	  mo.DamageMobj(self, self, 10, "Stun", DMG_FORCED);
	  DF_Monster(mo).bStunned = true;
	  DF_Monster(mo).StunTime = 35*3;
	}
	else
	{
	  mo.A_SpawnItemEx("StunPrimary", zofs: mo.height / 2);
	}
  }
  
  bool CheckForTarget()
  {
    BlockThingsIterator it = BlockThingsIterator.Create(self, 192); //Credit: Boondorl, for the whole thing
	bool hit = false;
	while(!hit && it.Next())
	  if(it.thing && it.thing is "DF_Monster" && it.thing.bIsMonster && it.thing.health > 0 && Distance3DSquared(it.thing) <= 36864) //Is this monster alive and in range?
		if(!it.thing.bDormant && it.thing.bShootable && !it.thing.bFriendly) //Can we damage this entity? should we?
		  if(CheckSight(it.thing, SF_IGNOREVISIBILITY|SF_SEEPASTSHOOTABLELINES|SF_IGNOREWATERBOUNDARY)) //Can we even see it?
			if(!DF_Monster(it.thing).bStunned && !it.thing.InStateSequence(it.thing.CurState, it.thing.ResolveState("Pain.Stun"))) //is it already stunned?
			  if(power > 0)
			  {
				DeliverShock(it.thing);
				power--;
				hit = true;
				
				if(fakeTrap)
				  VoltPickup(fakeTrap).charge = power;
			  }
			  
	return hit;
  }
  
  Default
  {
    Health 50;
    Radius 14;
	Height 13;
	Scale 0.95;
    DamageFactor 0.0;
	-SHOOTABLE
	+FLOORCLIP
	+THRUACTORS
	+FORCEPAIN
  }
  States
  {
    Spawn:
	  0000 A 2 A_CheckFloor("PreScan");
	  Wait;
	PreScan:
	  TNT1 A 0 A_StartSound("Volt_Use", CHAN_AUTO, attenuation: 0.5);
	  0000 A 4 Light("ElectroMineLight");
	  TNT1 A 0
	  {
	    if(power == 2)
		  return ResolveState("PreFirst");
		else if(power == 1)
		  return ResolveState("PreSecond");
		
		return ResolveState(null);
	  }
	  0000 AM 4 Light("ElectroMineLight");
	  0000 M 4 Light("ElectroMineLight")
	  {
	    if(CheckForTarget())
		  return ResolveState("First");
		  
		return ResolveState(null);
	  }
	  Wait;
	PreFirst:
	  0000 AI 4 Light("ElectroMineLight");
	  Goto First+4;
	PreSecond:
	  0000 AE 4 Light("ElectroMineLight");
	  Goto Second+4;
	First:
	  0000 MLKJ 2 Light("ElectroMineLight");
	  0000 I 4 Light("ElectroMineLight")
	  {
	    if(CheckForTarget())
		  return ResolveState("Second");
		  
		return ResolveState(null);
	  }
	  Wait;
	Second:
	  0000 IHGF 2 Light("ElectroMineLight");
	  0000 E 4 Light("ElectroMineLight")
	  {
	    if(CheckForTarget())
		  return ResolveState("Third");
		  
		return ResolveState(null);
	  }
	  Wait;
	Third:
	  TNT1 A 0
	  {
	    if(fakeTrap)
		  fakeTrap.Destroy();
	  }
	  0000 EDCBA 2 Light("ElectroMineLight");
	  TNT1 A 1 A_SpawnItemEx("ElectroMineDead");
	  Stop;
  }
}

	Scale 0.5;
	+FLOORCLIP
	+THRUACTORS
	+FORCEXYBILLBOARD
  }
  States
  {
    Spawn:
	  PEEL A 1
	  {
	    BlockThingsIterator it = BlockThingsIterator.Create(self, 64);
		while(it.Next())
		{
		  if(it.thing && it.thing is "DF_Boss" && it.thing.bIsMonster && it.thing.health > 0 && Distance3DSquared(it.thing) <= 4096)
		  {
		    it.thing.bTHRUACTORS = true;
			it.thing.bNONSHOOTABLE = true;
			it.thing.SetState(it.thing.ResolveState("Slip"));
			double spd = sqrt((it.thing.vel.x * it.thing.vel.x) + (it.thing.vel.y * it.thing.vel.y));

			GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
			if(ev)
			  ev.UpdateChat(String.Format("%s fell victim to a sinister prank.", it.thing.GetTag()));
			  
			return ResolveState("Death");
		  }
		}
		
		return ResolveState(null);
	  }
	  Wait;
	Death:
	  PEEL A -1;
	  Stop;
  }
