Class AmmoItem : InventoryItem
{
  string type, snd, msg, rmsg;
  int amount;
  bool hasReserve, isBox, hasLastGFX;
  
  action void A_GiveAmmo(String ammotype, int amount, String snd, String msg)
  {
    A_GiveInventory(ammotype, amount);
	A_PickupMessage(msg);
	A_StartSound(snd, CHAN_AUTO);
	
	if(itemFlash_amount > 0.0)
	  A_SetBlend("#ffffff", itemFlash_amount, 5);
  }
  
  Default
  {
    //$Category "Item/Ammo"
	//$Color 6
	Inventory.PickupMessage "";
	Inventory.PickupSound "";
	+INVENTORY.NOSCREENFLASH
	+FORCEXYBILLBOARD
  }

  States
  {
    Pickup:
	  TNT1 A 0
	  {
	    let ammo = FindInventory(invoker.type);
		if(ammo)
		{
		  int xtr_ammo = CVar.FindCVar("modf_ammo").GetInt();
		  //Whole thing is picked up
		  if(ammo.Amount + invoker.amount <= ammo.MaxAmount)
		  {
		    A_GiveAmmo(invoker.type, invoker.amount * xtr_ammo, invoker.snd, String.Format("+%d %s", invoker.amount * xtr_ammo, invoker.msg));
		  }
		  //Whole thing is NOT picked up. Leftovers.
		  else if(invoker.hasReserve && ammo.Amount + invoker.amount < ammo.MaxAmount + invoker.amount)
		  {
		    int gAmount = ammo.MaxAmount - ammo.Amount;
			A_GiveAmmo(invoker.type, gAmount * xtr_ammo, invoker.snd, "");
			
			GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
			if(ev)
			  ev.UpdateItem(String.Format("+%d %s", gAmount * xtr_ammo, invoker.msg));
			
			if(invoker.isBox)
			{
			  invoker.SetTag(String.Format("%s (%d)", invoker.rmsg, invoker.amount - gAmount));
			  if(invoker.hasLastGFX && invoker.amount - gAmount == 1)
			    invoker.frame++;
			}
			else
			  invoker.frame += gAmount;
			  
			invoker.amount -= gAmount;
			
			if(!CheckInventory("AmmoFullMsg", 1))
		      A_GiveInventory("AmmoFullMsg", 1);
			
			return invoker.ResolveState("Full");
		  }  
		  else 
		  {
		    if(!CheckInventory("AmmoFullMsg", 1))
		      A_GiveInventory("AmmoFullMsg", 1);
			  
			return invoker.ResolveState("Full");
		  }
		} 
		return invoker.ResolveState(null);
	  }
	  TNT1 A 0 A_RailWait;
	  Stop;
    Full:
	  TNT1 A 0;
	  Fail;
  }
}

Class GasVialSmoke : VisualThinker

  bool left;
  int time;
  
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	left = random(0,1);
  }
  
  override void Tick()
  {
	if(bDESTROYED || IsFrozen())
	  return;
	
	Super.Tick();
	time++;
	
	if(time > 5)
	{
	  alpha -= 0.002;
	  scale.x += 0.01;
	  scale.y = scale.x * 0.83;
	  roll += left ? 2.0 : -2.0;
	}
	else
	  alpha += 0.014;
	  
	if(Alpha <= 0.0)
	  Destroy();
  }
}

Class PsiPool : Actor

    Height 4;
    +NOINTERACTION
  }
  States
  {
    Spawn:
      TNT1 A 4 Light("PsiPool")
	  {
	    let pmo = players[consoleplayer].mo;
		if(pmo)
		{
	      if(Distance3D(pmo) < 1500 && !CheckIfSeen())
	      {
	        A_SpawnItemEx("PsiFX3", random(-24, 24), random(-24, 24), random(2, 40), 0, 0, 1);
	        A_SpawnItemEx("PsiFX3", random(-24, 24), random(-24, 24), random(2, 40), 0, 0, 1);
	      }
		  if(Distance3D(pmo) <= 144)
		  {
		    let item = pmo.FindInventory("Psi");
			if(!item || (item && item.Amount < item.MaxAmount))
		      return ResolveState("Death");
		  }
		}
		return ResolveState(null);
	  }
	  Wait;
	Death:
	  TNT1 A 1
	  {
	    let pmo = players[consoleplayer].mo;
		if(pmo)
		{
		  pmo.A_GiveInventory("Psi", 1);
		  pmo.A_GiveInventory("PsiPickupFX", 1);
		  pmo.A_SetBlend("Orange", 0.1, 20);
		  pmo.A_StartSound("Psi_Up", CHAN_AUTO);
		  
		  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		  if(ev)
			ev.UpdateItem("+1 \c[Orange]Psi");
			
		  for(int i = 0; i < 22; i++)
			A_SpawnItemEx("PsiFX3", random(-24, 24), random(-24, 24), random(2, 40), random(-4, 4), random(-4, 4), random(2, 3));
		}
	  }
	  Stop;
  }
}
