Class BlobGas : Actor
{
  int dmg_tic, lifetic;
  int dmg_bonus;
  void Gas_Damage()
  {
    BlockThingsIterator it = BlockThingsIterator.Create(self, 100); 
	while(it.Next())
	{
	  if(it.thing && it.thing.health > 0 && Distance2DSquared(it.thing) <= 10000)
	  {
		if(it.thing == self)
	      continue;
      
	    if(!it.thing.bShootable)
	      continue;
		  
		if(it.thing is "DFPlayer" || it.thing.bFriendly)
		  continue;
		  
		if(!CheckSight(it.thing, SF_IGNOREVISIBILITY | SF_IGNOREWATERBOUNDARY))
          continue; 
		  
		it.thing.DamageMobj(self, self, 5 + dmg_bonus, "Poison");
		
	  }
    }
  }
  
  override void BeginPlay()
  {
    dmg_tic = random(0,6);
	lifetic = 0;
	roll = random(0,359);
	bSPRITEFLIP = random(0,1);
	A_SetScale(frandom(0.2,0.3));
  }
  
  override void Tick()
  {
    if(lifetic < 10)
	{
	  A_FadeIn(frandom(0.02,0.04));
	  A_SetScale(Scale.x + frandom(0.025,0.05), Scale.Y + frandom(0.025,0.05));
	}
	
    if(lifetic >= 35)
	  A_FadeOut(0.01);
	
	if(dmg_tic >= 10)
	{
	  Gas_Damage();
	  dmg_tic = 0;
	}
	
	if(master)
	  SetOrigin(master.pos, false);
	
	dmg_tic++;
	lifetic++;
	
    Super.Tick();
  }
  
  Default
  {
	Alpha 0.0;
	RenderStyle "Translucent";
	+BRIGHT
    +FORCEXYBILLBOARD
	+NODAMAGETHRUST
    +NOINTERACTION
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  TNT1 A 0;
	  TNT1 A 0 A_Jump(127, "Smoke");
	Idle:
	  PGS1 ABCDEFGHIJKLMNOPQRSTUVWXYZ 1;
	  PGS2 ABCDEFGHIJKLMNOPQRSTUVWXYZ 1;
	  PGS3 ABCDEFGHIJKL 1;
      Loop;
	Smoke:
	  TNT1 A 0 A_Jump(256, random(1,4));
	  GAS2 ABCD -1;
	  Stop;
  }
}

    Super.DoEffect();
	
	if(cd > 0)
	  cd--;
	  
	if(owner && fxTime == 37)
	{
	  owner.roll = 0.0;
	  int bonusDmg = DFPlayer(owner).rad * 2;
	  owner.A_Explode(420 + bonusDmg, 1024, XF_NOTMISSILE, fulldamagedistance: 1024, damagetype: "Pen");

	  if(Health == 1 || Health > 50)
	    owner.A_DamageSelf(50, "Drowning", DMSS_NOPROTECT|DMSS_NOFACTOR);
	  else
	    owner.A_DamageSelf(Health - 1, "Drowning", DMSS_NOPROTECT|DMSS_NOFACTOR);
		
      let wep = players[consoleplayer].ReadyWeapon;
	  if(wep)
	    players[consoleplayer].SetPSprite(22, wep.ResolveState("Psi_Blast"));
		
	  let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
	  {
	    ev.UpdateChat(String.Format("%s \cRSacrifices \cJ50 HP.", owner.GetTag()));
		ev.UpdateChat(String.Format("%s \cJexpels \cJall Radiation.", owner.GetTag()));
	  }
	  let sh = ShaderHandler(EventHandler.Find("ShaderHandler"));
	  if(sh)
		sh.SetParry(1.0, 0.04, 0.04, 15);
	}
	else if(owner && fxTime == 55)
	{
	  A_StartSound("Will_02", CHAN_AUTO, volume: 1.0, attenuation: 0.0);
	  let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	  if(ev)
		ev.imageglitcher = true;
	}
	else if(owner && fxTime == 45)
	{
	  owner.A_SpawnItemEx("PsyhicBlast");
	}
	else if(fxTime == 10)
	{
	  let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	  if(ev)
		ev.imageglitcher = false;
	}
	
	if(owner && fxTime > 38)
	  owner.roll = frandompick(-0.5,0.5);
	  
	if(owner && (fxTime < 37 && fxTime > 0))
	{
	  for(int i = 0; i < 4; i++)
	  {
		double ang = random(0,359);
		double pit = random(-50,50);
		Vector3 SpawnPos = 72 * (cos(pit) * cos(ang), cos(pit) * sin(ang), sin(pit));
		Vector3 SpawnVel = SpawnPos * 0.05;
		owner.A_SpawnParticleEx("D2B80D", TexMan.CheckForTexture("SPFLA0"), STYLE_AddShaded, SPF_FULLBRIGHT|SPF_RELATIVE, 35, 4, 0,
		SpawnPos.x, SpawnPos.y, SpawnPos.z + 64, SpawnVel.x, SpawnVel.y, SpawnVel.z);
	  }
	}
	
	if(fxTime > 0)
	  fxTime--;
	
	if(playSnd)
	{
	  A_StartSound("Physique/Flux", CHAN_AUTO, volume: 1.0, attenuation: 0.0);
	  playSnd = false;
	}
  }
  
  override bool Use(bool pickup)
  {
	if(!owner || cd > 0)
	  return false;
	  
	if(cd <= 0)
	{
	  cd = 105;
	  playSnd = true;
	  fxTime = 60;
	  
	  if(owner)
	    owner.A_QuakeEx(2,2,2,50,0,100,"dsempty",QF_SCALEDOWN);
	}

	return false;
  }
}
