 Class Skill_Undying : Skill
{
  bool save;
  int cd;
  override void BeginPlay()
  {
    invName = "Path of Burial| Liberation";
	invIcon = "Graphics/Inventory Icon/Skill_Undying.png";
	invText = "$S_IronWill";
	invLore = "$S_IronWillLore";
	invSnd = "ATD_Use";
	invLog = "";
	save = true;
	
	Super.BeginPlay();
  }
  
  override void ModifyDamage (int damage, Name damageType, out int newdamage, bool passive, Actor inflictor, Actor source, int flags)
  {
    if(passive && damage > 0 && damageType != "InstantDeath")
      if(owner)
	    if(save && damage >= owner.health)
		{
		  newdamage = 0;
		  owner.A_SetHealth(1);
		  owner.GiveInventory("Invulnerability", 1);
		  save = false;
		  cd = 10500;
		}
  }
  
  override void DoEffect()
  {
    Super.DoEffect();
	
	if(cd > 0)
	  cd--;
	
	if(cd == 1)
	{
	  save = true;
	  cd--;
	}
  }
  
  override bool Use(bool pickup)
  {
	return false;
  }
}

Class Invulnerability : InvulnerabilitySphere
{
  Default
  {
    Powerup.Duration -10;
  }
}
