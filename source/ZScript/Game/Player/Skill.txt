Class Skill : CustomInventoryItem
{
  Default
  {
    Inventory.Amount 1;
	Inventory.MaxAmount 1;
	+INVENTORY.UNCLEARABLE
	+INVENTORY.UNDROPPABLE
  }
  States
  {
    Use:
	  TNT1 A 0;
	  Fail;
	Pickup:
	  TNT1 A 0 A_RailWait();
	  Stop;
  }
}

Class Skill_Exit : Skill
{
  override void BeginPlay()
  {
    invName = "Exit Duel";
	invIcon = "Graphics/Inventory Icon/Skill_Duel.png";
	invText = "$S_DuelExit";
	invLore = "$S_DuelExitLore";
	
	Super.BeginPlay();
  }
}
