class ShaderHandler : EventHandler
{
  int resScale, dr_time, radBlurTime, ig_time;
  double ca_amt, ca_str, dr_amt, dr_str, dr_alpha, blr_alpha, blink_alpha, dst_Mltp, ig_amt, ig_speed, vhsLineRes, vhsLineAlpha;
  double hz_speed, hz_blocksize, hz_max_x, hz_max_y;
  bool retroHidden, comeUp, HighAF, isClose, zUnstable, zHazard, isMad, vhs_on, marifx_on, ColorGlitch, ImageGlitcher;
  
  //Pulse
  double CT_Time, CT_Stepf;
  int PulseTime;
  
  //Radial Blur
  double rblr_alpha, rblr_stepf;
  bool rblr_fade;
  
  //Inverse Shader
  bool shader_invert;
  
  bool init;
  
  override void OnRegister()
  {
    if(!init)
	{
      dst_Mltp = 0.05;
	  ig_amt = 0.06;
	  ig_speed = 0.4;
	  hz_speed = 7.0;
	  hz_blocksize = 15.0;
	  hz_max_x = 0.3;
	  hz_max_y = 0.3;
	  vhs_on = true;
	  marifx_on = true;
	  init = true;
	  shader_invert = false;
	}
  }
  
  void SetBlur(int duration)
  {
    blr_alpha = 0.0;
  }
  
  void SetRadialBlur(double amount)
  {
    rblr_alpha = amount;
  }
  
  void FadeRadialBlur(double amount)
  {
    rblr_stepf = amount;
	rblr_fade = true;
  }
  
  void SetParry(double blurAmount, double blurFadeAmount, double step, int time)
  {
    if(blurAmount > 0.0)
	{
      rblr_alpha = blurAmount;
      rblr_stepf = blurFadeAmount;
	  rblr_fade = true;
	}
	CT_Time = 0.0;
	CT_Stepf = step;
	PulseTime = time;
  }
  
  static void ForceVHS(bool on)
  {
    let handler = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(handler)
	  handler.vhs_on = on;
  }
  
  static void SetAcidVisual(bool on)
  {
    let handler = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(handler)
	  if(on)
	    handler.comeUp = true;
	  else
	    handler.comeUp = false;
  }
  
  static void SetColorGlitch(bool on)
  {
    let handler = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(handler)
	  if(on)
	    handler.ColorGlitch = true;
	  else
	    handler.ColorGlitch = false;
  }
  
  static void SetColorPP(bool on)
  {
    let handler = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(handler)
	  if(on)
	    handler.marifx_on = true;
	  else
	    handler.marifx_on = false;
  }
  
  static void SetDrunk(double intensity, double zoom, int duration)
  {
    let handler = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(handler)
	{
      handler.dr_amt = intensity;
	  handler.dr_str = zoom;
	  handler.dr_time = duration;
	  handler.dr_alpha = 0.5;
	}
  }
  
  static void SetFractal(bool on)
  {
    let handler = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(handler)
	  handler.HighAF = on;
  }
  
  static void SetImageGlitcher(bool on)
  {
    let handler = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(handler)
	  if(on)
	    handler.ImageGlitcher = true;
	  else
	    handler.ImageGlitcher = false;
  }
  
  static void SetInvert(bool on)
  {
    let handler = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(handler)
	  if(on)
	    handler.shader_invert = true;
	  else
	    handler.shader_invert = false;
  }
  
  override void WorldTick()
  {
    Super.WorldTick();
	
    PlayerInfo p = players[consoleplayer];
	switch(vhs_strength)
    {
	  case 0:
      resScale = 0;
	  ca_amt = 0.0;
	  ca_str = 0.0;
	  vhsLineAlpha = 0.1;
      break;
	  
      case 1:
      resScale = 0;
	  ca_amt = 5.0;
	  ca_str = 0.0;
	  vhsLineAlpha = 0.3;
      break;
	  
	  case 2:
      resScale = 0;
	  ca_amt = 5.0;
	  ca_str = 0.0;
	  vhsLineAlpha = 0.7;
      break;
	  
	  case 3:
      resScale = 320;
	  ca_amt = 10.2;
	  ca_str = 0.366;
	  vhsLineAlpha = 0.5;
      break;
    }
	
	//ImageGlitching call
	if(ig_time > 0)
	  ig_time--;
	
	//Radial Blur
	if(rblr_fade)
	{
	  if(rblr_alpha > 0.0)
	  {
	    rblr_alpha -= rblr_stepf;
	  }
	  else if(rblr_alpha <= 0.0)
	  {
	    rblr_alpha = 0.0;
	    rblr_stepf = 0.0;
	    rblr_fade = false;
	  }
	}
	
	//Pulse
	if(PulseTime > 0)
	{
	  CT_Time += CT_Stepf;
	  PulseTime--;
	}
	
	if(dr_time > 0)
	  dr_time--;
	else if(dr_time <= 0 && dr_alpha > 0.0)
	  dr_alpha -= 0.01;
	
	//Radiation business
	if(p.mo && DFPlayer(p.mo).bPhy_Tendon)
	  return;
	  
	if(DFPlayer(p.mo).rad >= DFPlayer(p.mo).radMax * 0.5)
	{
	  if(radBlurTime == 546)
	  {
	    radBlurTime = 0;
	    blr_alpha = 0.0;
	  }
	  else if(radBlurTime == 545)
	  {
	    blink_alpha = 0.6;
	    if(p.mo)
	      p.mo.A_StartSound("Skill/NightVision", CHAN_AUTO, volume: 0.4);
	  }
	  else if(radBlurTime >= 440)
	  {
	    blr_alpha += 0.01;
	  }
	  radBlurTime++;
	}
	else
	{
	  radBlurTime = 0;
	  blr_alpha = 0.0;
	}
	
	if(blink_alpha > 0.0)
	  blink_alpha -= 0.2;
  }
  
  override void RenderOverlay(RenderEvent e)
  {
	PlayerInfo p = players[consoleplayer];
	double timer = (gametic +  e.FracTic) * 1/35;
	
	///////// LOW HEALTH
	if(p && p.mo.health < 50)
	{
	  double alpha = 1.0 - ((p.mo.health - 25) * 0.04);
	  double grey = 0.4 - ((p.mo.health - 25) * 0.016);
	  PPShader.SetUniform1f("Low_Health", "greyscale", clamp(grey, 0.0, 0.5));
	  PPShader.SetUniform1f("Low_Health", "opacity", clamp(alpha, 0.0, 1.0));
	  PPShader.SetUniform1f("Low_Health", "timer", timer);
	  PPShader.SetEnabled("Low_Health", true);
	}
	else
	  PPShader.SetEnabled("Low_Health", false);
	
	/////////LSD Visuals
	if(comeUp)
	{
	  PPShader.SetUniform1f("LSD_Visual", "timer", timer);
	  PPShader.SetEnabled("LSD_Visual", true);
	}
	else
	  PPShader.SetEnabled("LSD_Visual", false);
	  
	/////////Color Glitcher
	if(ColorGlitch)
	{
	  PPShader.SetUniform1f("ColorGlitch", "timer", timer);
	  PPShader.SetEnabled("ColorGlitch", true);
	}
	else
	  PPShader.SetEnabled("ColorGlitch", false);
	  
	/////////Counter Attack trigger -- Afterimage pulse
	if(PulseTime > 0)
	{
	  PPShader.SetUniform1f("CounterAttack", "CT_Timer", CT_Time);
	  PPShader.SetEnabled("CounterAttack", true);
	}
	else
	  PPShader.SetEnabled("CounterAttack", false);
	  
	/////////Drunk Shader/////////////////
	if(dr_alpha > 0.0)
	{
	  PPShader.SetUniform1f("drunk", "drunkStrength", dr_amt);
	  PPShader.SetUniform1f("drunk", "drunkZoom", dr_str);
	  PPShader.SetUniform1f("drunk", "drunkAlpha", dr_alpha);
	  PPShader.SetUniform1f("drunk", "timer", timer);
	  PPShader.SetEnabled("drunk", true);
	}
	else
	  PPShader.SetEnabled("drunk", false);
	
	/////////Gaussian Blur Shader/////////////////
	if(blr_alpha > 0.0)
	{
	  PPShader.SetUniform1f("Blur", "blurAlpha", blr_alpha);
	  PPShader.SetEnabled("Blur", true);
	}
	else
	  PPShader.SetEnabled("Blur", false);
	  
	/////////Radial Blur Shader/////////////////
	if(rblr_alpha)
	{
	  PPShader.SetUniform1f("Radial_Blur", "rBlurAlpha", rblr_alpha);
	  PPShader.SetEnabled("Radial_Blur", true);
	}
	else
	  PPShader.SetEnabled("Radial_Blur", false);
	  
	/////////MariFX Shader/////////////////
	if(p.CrouchFactor < 1.0)
	{
	  double amt = 1.0 - p.CrouchFactor;
	  PPShader.SetEnabled("mfx_vignette", true);
	  PPShader.SetUniform1f("mfx_vignette", "vigMul", 0.25 + amt);
	}
	else
	  PPShader.SetEnabled("mfx_vignette", false);
	  
	/////////MariFX Shader/////////////////
	if(marifx_on)
	{
	  PPShader.SetEnabled("mfx_lumasharp_static", true);
	  PPShader.SetEnabled("mfx_grading_static", true);
	  PPShader.SetEnabled("mfx_lutgrading_static", true);
	  //PPShader.SetEnabled("mfx_vignette_static", true);
	}
	else
	{
	  PPShader.SetEnabled("mfx_lumasharp_static", false);
	  PPShader.SetEnabled("mfx_grading_static", false);
	  PPShader.SetEnabled("mfx_lutgrading_static", false);
	  PPShader.SetEnabled("mfx_vignette_static", false);
	}
	/////////VHS Shader/////////////////
	if(vhs_on && vhs_strength > 0)
	{
	  if(vhs_on && vhs_strength > 1)
	  {
	    PPShader.SetUniform1f("vhsFX", "noise", 1);
		 
		PPShader.SetUniform1f("vhsSmear2", "timer", timer);
	    PPShader.SetEnabled("vhsSmear2", true);
	  }
	  else
	  {
	    PPShader.SetEnabled("vhsSmear2", false);
		PPShader.SetUniform1f("vhsFX", "noise", 0);
	  }
		
	  PPShader.SetUniform1f("ChromeAb", "amount", ca_amt);
	  PPShader.SetUniform1f("ChromeAb", "strength", ca_str);
	  PPShader.SetEnabled("ChromeAb", true);
	  
	  PPShader.SetUniform1f("vhsFX", "lineAlpha", vhsLineAlpha);
	  PPShader.SetUniform1f("vhsFX", "lineRes", vhsLineRes);
	  PPShader.SetUniform1f("vhsFX", "res", resScale);
	  PPShader.SetUniform1f("vhsFX", "timer", timer);
	  PPShader.SetUniform1f("vhsSmear", "timer", timer);
	  PPShader.SetEnabled("vhsSmear", true);
	 /* PPShader.SetUniform1f("vhsFX", "val_a", cvar.findcvar("val_a").getfloat());
	  PPShader.SetUniform1f("vhsFX", "val_b", cvar.findcvar("val_b").getfloat());
	  PPShader.SetUniform1f("vhsFX", "val_c", cvar.findcvar("val_c").getfloat());
	  PPShader.SetUniform1f("vhsFX", "val_d", cvar.findcvar("val_d").getfloat());*/
	  
	  if(vhs_scroll)
	  {
	    PPShader.SetUniform1f("vhsFX", "tear_rate", 0.996);
	    PPShader.SetUniform1f("vhsFX", "scroll", true);
	  }
	  else
	    PPShader.SetUniform1f("vhsFX", "scroll", false);
		
	  PPShader.SetEnabled("vhsFX", true);
	}
	else
	{
	  PPShader.SetEnabled("vhsFX", false);
	  PPShader.SetEnabled("ChromeAb", false);
	  PPShader.SetEnabled("vhsSmear", false);
	  PPShader.SetEnabled("vhsSmear2", false);
	}
	
	////////Toxin Shader /////////////////////
	if(p.mo && !DFPlayer(p.mo).bPhy_Tendon)
	{
	  double tox = double(DFPlayer(p.mo).tox);
	  double toxMax = double(DFPlayer(p.mo).toxMax);
	  if(tox > 0 && p.mo.waterlevel < 3)
	  {
	    double pStrScale, pStrVal, pFadeScale, pFadeVal;
	  
	    if(tox < toxMax * 0.5)
	    {
	      pStrScale = 1.75 / toxMax;
	      pFadeScale = 2.5 / toxMax;
		  pStrVal = clamp(tox * pStrScale, 0.0, 0.7);
		  pFadeVal = clamp(tox * pFadeScale, 0.0, 1.0);
	    }
	    else
	    {
	      pStrScale = 1.4 / toxMax;
	      pFadeScale = 2.0 / toxMax;
		  pStrVal = tox * pStrScale;
		  pFadeVal = tox * pFadeScale;
	    }
	  
	    PPShader.SetUniform1f("Poison" ,"Timer",(gametic+e.fractic)/35);
	    PPShader.SetUniform1f("Poison" ,"Strength", pStrVal);
	    PPShader.SetUniform1f("Poison" ,"Tint", 0.5);
	    PPShader.SetUniform1f("Poison" ,"Fade", pFadeVal);
	    PPShader.SetEnabled("Poison", true);
	  }
	  else
	    PPShader.SetEnabled("Poison", false);
	}
	else
	  PPShader.SetEnabled("Poison", false);
	  
	if(isMad)
	{
	  PPShader.SetUniform1f("Poison" ,"Timer",(gametic+e.fractic)/35);
	  PPShader.SetUniform1f("Poison" ,"Strength", 1.1);
	  PPShader.SetUniform1f("Poison" ,"Tint", 0.5);
	  PPShader.SetUniform1f("Poison" ,"Fade", 1.7);
	  PPShader.SetEnabled("Poison", true);
	}
	  
	 ////////Unstable-Hazardous Zones Shader /////////////////////
	if(DFPlayer(p.mo).inZone)
	{
	  if(zHazard && hzv_enable)
	  {
	    PPShader.SetUniform1f("Hazardous", "timer", timer);
		PPShader.SetUniform1f("Hazardous", "_Speed", hz_speed);
		PPShader.SetUniform1f("Hazardous", "_BlockSize", hz_blocksize);
		PPShader.SetUniform1f("Hazardous", "_MaxRGBSplitX", hz_max_x);
		PPShader.SetUniform1f("Hazardous", "_MaxRGBSplitY", hz_max_y);
	    PPShader.SetEnabled("Hazardous", true);
	  }
	  else
	  {
	    PPShader.SetUniform1f("vhsBreathe", "timer", timer);
		PPShader.SetUniform1f("vhsFX", "tear_rate", 0.85);
		PPShader.SetEnabled("vhsBreathe", true);
	    PPShader.SetEnabled("Hazardous", false);
	  }
	}
	else
	{
	  PPShader.SetEnabled("vhsBreathe", false);
	  PPShader.SetEnabled("Hazardous", false);
	}
	
	/////////IMAGE GLITCHER
	if(ImageGlitcher || ig_time > 0)
	{
	  PPShader.SetEnabled("ImageGlitcher", true);
	  PPShader.SetUniform1f("ImageGlitcher", "timer", timer);
	  PPShader.SetUniform1f("ImageGlitcher", "AMT", ig_amt);
	  PPShader.SetUniform1f("ImageGlitcher", "SPEED", ig_speed);
	}
	else
	  PPShader.SetEnabled("ImageGlitcher", false);
	  
	if(p.mo.waterlevel >= 3 || dst_Mltp != 0.05) 
	{
	  PPShader.SetUniform1f("Underwater", "scale", 1.0);
	  PPShader.SetUniform1f("Underwater", "distortionMultiplier", dst_Mltp);
	  PPShader.SetUniform1f("Underwater", "subtractiveDistortionMultiplier", 0.015);
	  PPShader.SetUniform1f("Underwater", "NewTime", level.time * 0.005);
	  PPShader.SetEnabled("Underwater", true);
	}
    else 
      PPShader.SetEnabled("Underwater", false);
	
	if(HighAF)
	{
	  PPShader.SetUniform1f("Fractal", "timer", timer);
	  PPShader.SetEnabled("Fractal", true);
	}
	else
	  PPShader.SetEnabled("Fractal", false);
	  
	///////// Inverted Image Shader
	PPShader.SetEnabled("Invert", shader_invert);
  } 
  
  override void RenderUnderlay(RenderEvent e)
  {
    /////////////BLINK
	if(blink_Alpha > 0.0)
	  Screen.Dim(Color(0, 0, 0), blink_Alpha, 0, 0, Screen.GetWidth(), Screen.GetHeight());
  }
}

Class LowZone : Inventory
{
  override void AttachToOwner(Actor user)
  {
	if(DFPlayer(user))
	{
	  DFPlayer(user).inZone = false;
	  DFPlayer(user).inHighZone = false;
	  let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	  if(ev)
	  {
	    ev.zUnstable = false;
		ev.zHazard = false;
	  }
	}
	  
    Super.AttachToOwner(user);
  }
  
  override void PostBeginPlay()
  {
    Destroy();
	Super.PostBeginPlay();
  }
}

Class MidZoneFX : Inventory
{
  bool blink;
  
  override void Tick()
  {
    Super.Tick();
	
	if(!owner)
	  return;
	
	if(GetAge() % 4 == 0)
	{
	  if(!blink)
	  {
	    DFPlayer(owner).zoneLevel = 6;
		blink = true;
	  }
	  else
	  {
	    DFPlayer(owner).zoneLevel = 0;
		blink = false;
	  }
	}
	
	if(GetAge() > 16)
	  Destroy();
  }
}

Class HighZoneFX : Inventory
{
  bool blink;
  int build;
  
  override void Tick()
  {
    Super.Tick();
	
	if(!owner)
	  return;
	
	if(GetAge() < 25)
	  DFPlayer(owner).zoneLevel = randompick(0,1,2,3,4,6);
	else if(GetAge() > 24 && GetAge() < 55)
	{
	  DFPlayer(owner).zoneLevel = 5;
	  if(!blink)
	  {
	    DFPlayer(owner).zoneLevel = random(1,4);
		blink = true;
	  }
	  else
	  {
	    DFPlayer(owner).zoneLevel = 5;
		blink = false;
	  }
	}
	else if(GetAge() > 56)
	{
	  if(GetAge() % 2 == 0)
	  {
	    if(!blink)
	    {
	      DFPlayer(owner).zoneLevel = 0;
		  blink = true;
	    }
	    else
	    {
	      DFPlayer(owner).zoneLevel = 7;
		  blink = false;
	    }
	  }
	}
	
	if(GetAge() > 76)
	  Destroy();
  }
}

Class BlastOffToken : Inventory
{
  int baseFOV;
  
  override void DoEffect()
  {
    Super.DoEffect();
	
	if(GetAge() < 110)
	{
	  let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	  if(ev)
	  {
	    ev.rblr_alpha += 0.0125;
		owner.player.SetFOV(baseFOV + (GetAge() * 0.55));
	    //owner.A_ZoomFactor(1.0 - (GetAge() * 0.00375));
	  }
	}
	else
	  Destroy();
  }
  
  override void AttachToOwner(Actor other)
  {
    if(other)
      baseFOV = other.player.FOV;
	  
	let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(ev)
	  ev.rblr_alpha = 0.0;
	  
	Super.AttachToOwner(other);
  }
  
  override void DetachFromOwner()
  {
    let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(ev)
	{
	  ev.rblr_alpha = 0.0;
	  owner.player.SetFOV(baseFOV);
	}
  }
  
  Default
  {
    Inventory.Amount 1;
  }
}

Class ZoomShaderHit_Hand : Inventory
{
  override void DoEffect()
  {
    Super.DoEffect();
