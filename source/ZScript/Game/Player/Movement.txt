class MovementHandler : EventHandler 
{
  bool sprintReady, tryDodge, tryHoldSlide, canBeginRegenWait;
  int baseRegen, dodgeCD, slideHoldCount, jumpSlideWindow, regenExpireCD, jumpTime;
  double ang, speed_buff_alpha;
  StaticEventHandler ev;
  
  override void WorldLoaded(WorldEvent e)
  {
    Super.WorldLoaded(e);
  }
  
  double GetPAngle(PlayerPawn pmo)
  {
	int btn = pmo.GetPlayerInput(0);
	
	if(btn & BT_FORWARD && btn & BT_MOVELEFT)
	  ang = 45;
	
	else if(btn & BT_BACK && btn & BT_MOVELEFT)
	  ang = 135;
	
	else if(btn & BT_BACK && btn & BT_MOVERIGHT)
	  ang = 225;
	
	else if(btn & BT_FORWARD && btn & BT_MOVERIGHT)
	  ang = 315;
	
	else if(btn & BT_FORWARD)
	   ang = 0;
	   
    else if(btn & BT_MOVELEFT)
	  ang = 90;
	
	else if(btn & BT_BACK)
	  ang = 180;
	
	else if(btn & BT_MOVERIGHT)
	  ang = 270;
	  
	else
	  ang = VectorAngle(pmo.GetPlayerInput(INPUT_FORWARDMOVE, AAPTR_DEFAULT), pmo.GetPlayerInput(INPUT_SIDEMOVE, AAPTR_DEFAULT) * -1.0);
	
    return ang;
  }  
  
  override void OnRegister()
  {
    ev = StepEventHandler(StaticEventHandler.Find("StepEventHandler"));
	baseRegen = 1;
  }
  
  override bool InputProcess(InputEvent e)
  {
    if(DFPlayer(players[consoleplayer].mo) && DFPlayer(players[consoleplayer].mo).gov)
      return true;
	
    if(e.Type == InputEvent.Type_KeyDown)
	{
	  Array<int> iKey;
	  iKey.Clear();
	  Bindings.GetAllKeysForCommand(iKey, "Sprint");
	  for(int i = 0; i < iKey.Size(); i++)
	    if(e.KeyScan == iKey[i])
	      SendNetworkEvent("TryRun");
		  
	  iKey.Clear();
	  Bindings.GetAllKeysForCommand(iKey, "+jump");
	  for(int i = 0; i < iKey.Size(); i++)
	    if(e.KeyScan == iKey[i])
	      SendNetworkEvent("Jumped");
		  
	  iKey.Clear();
	  Bindings.GetAllKeysForCommand(iKey, "Dodge");
	  for(int i = 0; i < iKey.Size(); i++)
	    if(e.KeyScan == iKey[i])
	      SendNetworkEvent("Evade");
	  
	  iKey.Clear();
	  Bindings.GetAllKeysForCommand(iKey, "Crouch");
	  for(int i = 0; i < iKey.Size(); i++)
	    if(e.KeyScan == iKey[i] && sprintReady)
	      SendNetworkEvent("TrySlide");
		  
	  iKey.Clear();
	  Bindings.GetAllKeysForCommand(iKey, "+crouch");
	  for(int i = 0; i < iKey.Size(); i++)
	    if(e.KeyScan == iKey[i] && sprintReady)
	      SendNetworkEvent("TryHoldSlide");

	  iKey.Clear();
	  Bindings.GetAllKeysForCommand(iKey, "+jump");
	  for(int i = 0; i < iKey.Size(); i++)
	    if(e.KeyScan == iKey[i] && jumpSlideWindow > 0)
	      SendNetworkEvent("TryJumpSlide");
	}
	  
	if(e.Type == InputEvent.Type_KeyUp)
	{
	  Array<int> iKey;
	  iKey.Clear();
	  Bindings.GetAllKeysForCommand(iKey, "Sprint");
	  for(int i = 0; i < iKey.Size(); i++)
	    if(e.KeyScan == iKey[i])
	      SendNetworkEvent("StopRun");
	}
	  
    return false;
  }
  
  override void NetworkProcess(ConsoleEvent e)
  {
	Array<string> command;
	e.Name.Split (command, ":");
	
	if(players[e.player].health <= 0)
	  return;
		
	if(players[e.player].cheats & (CF_FROZEN|CF_TOTALLYFROZEN)) //Credit Nash (cheats)
	  return;
	  
	if(command[0] ~== "TryRun")
	{
	  sprintReady = true;
	  let pmo = players[e.player].mo;
	  if(pmo.CountInv("Stamina") <= 0)
	  {
	    let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		if(ev)
		{
		  if(ev.warn_staminaF <= 0.0)
			ev.warn_staminaF = 1.5;
		}
	  }
	}
	else if(command[0] ~== "StopRun")
	  sprintReady = false;
	  
	if(command[0] ~== "Jumped")
	  jumpTime = 18;
	  
	if(command[0] ~== "Evade")
	{
	  let pmo = players[e.player].mo;
	  
	  if(pmo.waterlevel > 0)
	  {
	    pmo.ACS_NamedExecute("WaterDodge");
		return;
	  }
	  
	  if(dodgeCD <= 0 && pmo.CountInv("Stamina") > 0)
	  {
	    tryDodge = true;
		DFPlayer(pmo).iFrames = 5 + (DFPlayer(pmo).Str * 2) + DFPlayer(pmo).Str;
		DFPlayer(pmo).isDodging = True;
		tryDodge = false;
		
		if(jumpTime > 0)
	    {
		  pmo.A_TakeInventory("EvadeSpeedBuff", 1);
		  pmo.A_GiveInventory("EvadeSpeedBuff", 1);
		  pmo.Thrust(48, pmo.angle + GetPAngle(pmo));
		  pmo.A_StartSound("DodgeX", CHAN_AUTO);
		  dodgeCD = 36;
		  if(CVar.FindCVar("modf_stamina").GetBool())
		    pmo.TakeInventory("Stamina", 125);
	    }
	    else
	    {
		  pmo.Thrust(44, pmo.angle + GetPAngle(pmo));
		  pmo.A_GiveInventory("DashStabilizer", 1);
		  
		  if(CVar.FindCVar("modf_stamina").GetBool())
		    pmo.TakeInventory("Stamina", 50);
		  
		  if(pmo.pos.z - pmo.floorz == 0)
		    pmo.A_StartSound("Dodge", CHAN_AUTO, volume: 0.7);
			
		  dodgeCD = 26;
	    }
	  }
	  else if(pmo.CountInv("Stamina") <= 0)
	  {
	    let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		if(ev)
		{
		  if(ev.warn_staminaF <= 0.0)
			ev.warn_staminaF = 1.5;
		}
	  }
	}
	
	if(command[0] ~== "TryHoldSlide")
	{
	  slideHoldCount = 4;
	  jumpSlideWindow = 13;
	  players[e.player].mo.Thrust(24, players[e.player].mo.angle + GetPAngle(players[e.player].mo));
	  players[e.player].mo.A_StartSound("Slide", CHAN_AUTO, volume: 1.0);
	  DFPlayer(players[e.player].mo).iFrames = 5 + players[e.player].mo.CountInv("WitToken");
	  if(CVar.FindCVar("modf_stamina").GetBool())
	    players[e.player].mo.TakeInventory("Stamina", 25);
	}

	if(command[0] ~== "TrySlide")
	{
	  if(players[e.player].crouchfactor == 1.0 && players[e.player].onground && players[e.player].mo.CountInv("Stamina") > 0)
	  {
	    jumpSlideWindow = 13;
	    players[e.player].crouching = -1;
		players[e.player].mo.Thrust(24, players[e.player].mo.angle + GetPAngle(players[e.player].mo));
		players[e.player].mo.A_StartSound("Slide", CHAN_AUTO, volume: 1.0);
		DFPlayer(players[e.player].mo).iFrames = 5 + players[e.player].mo.CountInv("WitToken");
		if(CVar.FindCVar("modf_stamina").GetBool())
	      players[e.player].mo.TakeInventory("Stamina", 25);
	  }
	}
	
	if(command[0] ~== "TryJumpSlide")
	{
	  if(players[e.player].mo.CountInv("Stamina") > 0)
	  {
	    jumpSlideWindow = 0;
		players[e.player].mo.vel.z += 14;
		players[e.player].mo.Thrust(5, players[e.player].mo.angle + GetPAngle(players[e.player].mo));
		if(CVar.FindCVar("modf_stamina").GetBool())
		  players[e.player].mo.TakeInventory("Stamina", 50);
	  }
	}
  }
  
  override void WorldTick()
  {
    PlayerPawn pmo = players[consoleplayer].mo;
	int btn = pmo.GetPlayerInput(0);
	int curStamina = pmo.CountInv("Stamina");
	
	//Auto-Regen
	if(canBeginRegenWait && curStamina <= 0)
	{
	  regenExpireCD = 70;
	  sprintReady = false;
	  canBeginRegenWait = false;
	}
	
	if(regenExpireCD > 0)
	  regenExpireCD--;
	else
	  canBeginRegenWait = true;
	  
	if(regenExpireCD <= 0 && pmo.health > 0)
	  pmo.GiveInventory("Stamina", baseRegen);
	
	//Sprint
	if(sprintReady && curStamina)
	{
	  if((btn & BT_FORWARD) || (btn & BT_BACK) || (btn & BT_MOVELEFT) || (btn & BT_MOVERIGHT))
	  {
	    pmo.speed = 2.33;
		if(CVar.FindCVar("modf_stamina").GetBool())
	      pmo.TakeInventory("Stamina", 3);
	  }
	}
	else
	  pmo.speed = 1.0;
	  
	//Dodge
	if(dodgeCD > 0)
	  dodgeCD--;
	
	//Slide
	if(slideHoldCount > 0)
	{
	  slideHoldCount--;
	  players[consoleplayer].crouching = -1;
	}
	
	//Slide-Jumping
	if(jumpSlideWindow > 0)
	  jumpSlideWindow--;
	
	//Footstep-Handler
	if(dodgeCD > 0 || slideHoldCount > 0)
	  if(ev)
	    StepEventHandler(ev).quiet = true;
	else
	  if(ev)
	    StepEventHandler(ev).quiet = false;
		
	//Jump related
	if(jumpTime > 0)
	  jumpTime--;
	  
	//
	if(speed_buff_alpha > 0.0)
	  speed_buff_alpha -= 0.00125;
	  
	//console.printf("%d", jumpTime);
  }
  
  override void RenderUnderlay(RenderEvent e)
  {
    if(speed_buff_alpha > 0.0)
      Screen.Dim(Color(0, 204, 0), speed_buff_alpha, 0, 0, Screen.GetWidth(), Screen.GetHeight());
  }
}

Class DashStabilizer : Inventory
{
  Vector3 oldvel;
  int tic;
  
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
	
	oldvel = other.vel;
  }
  
  
  override void DoEffect()
  {
	Super.DoEffect();
	
	if(owner)
	{
	  if(tic < 10)
	    tic++;
		
	  if(tic == 6)
	    owner.vel *= 0.4;
		
	  if(tic == 7)
	    owner.vel = oldvel * 0.2;
	  
	  if(tic == 10)
	  {
	    DFPlayer(owner).isDodging = false;
		Destroy();
	  }
	}
	else
	  Destroy();
  }
  
  Default
  {
    Inventory.Amount 1;
	+Inventory.UNDROPPABLE
  }
}

Class EvadeSpeedBuff : Inventory
{
  bool land;
  int tic, pTic;
  
  override void DoEffect()
  {
	Super.DoEffect();
	
	if(owner && owner is "DFPlayer")
	{
	  if(owner.waterlevel > 0)
	  {
	    Destroy();
		return;
	  }
	  
	  if(pTic == 7)
	     owner.vel *= 0.5;
	
	  if(pTic == 10)
	    DFPlayer(owner).isDodging = false;
	  
	  pTic++;
	  
	  if(!land && owner.pos.z <= owner.floorz)
	  {
	    tic = 88;
		land = true;
		let ev = MovementHandler(EventHandler.Find("MovementHandler"));
		if(ev)
		  ev.speed_buff_alpha = 0.1;
	  }
		 
	  if(land && tic > 0)
	  {
	    owner.speed = 1.0 + Clamp(tic / 35.0, 0.0, 2.5);
	    tic--;
	  }
	
	  if(land && tic <= 0)
	    Destroy();
	}
	else
	  Destroy();
  }
  
  Default
  {
    Inventory.Amount 1;
	+Inventory.UNDROPPABLE
  }
}
