Class DF_Monster : Actor
{
  int sHeight, StunTime, moddedDmg, AP, APMax, parry_time;
  double killerAng, radar_alpha, radar_height;
  bool Armored, died, hasSight, Poisoned, pOpen, TNTDeath, stunInit, unScannable, radar_sight;
  uint DFMFlags;
  flagdef Stunned : DFMFlags, 0;
  flagdef PossessImmune : DFMFlags, 1;
  flagdef StunSoft : DFMFlags, 2;
  flagdef Dazed : DFMFlags, 3;
  flagdef PoisonResist : DFMFlags, 4;
  flagdef StunResist : DFMFlags, 5;
  flagdef Possessed : DFMFlags, 6;
  flagdef Large : DFMFlags, 7;
  flagdef NoRadar : DFMFlags, 8;
  
  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)
  {
    super.Die(source, inflictor, dmgFlags, MeansOfDeath);
	
	A_StopSound(24);
  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod)
  {
    if(!source)
	  return Super.DamageMobj(inflictor, source, damage, mod);
	  
	if(mod == "Poison")
	{
	  if(bPoisonResist)
	    damage = 0;
	  
	  if(ApplyDamageFactor("Poison", 10) > 0)
	  {
	    bool left = random(0,1);
	    A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("BBLLD0"), STYLE_NONE, SPF_ROLL|SPF_RELATIVE, random(24,35), random(6,9), 0,
		random(-radius, radius), frandom(-radius, radius), frandom(8, height - 8), velz: frandom(0.0,0.8), fadestepf: 0.0, startroll: random(0,359), rollvel: left ? -2.0 : 2.0);
	  }
	}
	
	if(!bStunResist && !stunInit && mod == "Stun")
	{
	  stunInit = true;
	}
	
	return Super.DamageMobj(inflictor, source, damage, mod);
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(stunInit)
	{
	  if(health <= 0)
	    A_Die("Shock");
	  else
	  {
	    stunInit = false;
	    SetStateLabel("Stun");
	  }
	}
  }

  Default
  {
    PainChance "Poison", 256;
    PainChance "Sting", 0;
    PainChance "Stun", 256;
	DamageFactor "Wave", 1.0;
	MaxStepHeight 32;
	MaxDropOffHeight 64;
    //$Color 4
  }
  
  override void PostBeginPlay()
  {
    sHeight = self.Height;
	StunTime = 0;
	//console.printf("%d", sHeight);
  }
  
  void A_GiveExp(int amount)
  {
    Actor xpg = Spawn("ExpGiver", pos);
	if(xpg)
	  ExpGiver(xpg).Amount = amount;
  }
  
  void A_DazeEffect()
  {
    bNOPAIN = true;
	bNOINFIGHTING = true;
	bDazed = true;
	A_GiveInventory("DazeToken", 1);
	
	let pmo = players[consoleplayer].mo;
	if(pmo)
	  pmo.A_SetBlend("Orange", 0.13, 17);
	  
	if(target)
	  angle = AngleTo(target);
  }
  
  void A_EndDaze()
  {
    bNOPAIN = false;
	bNOINFIGHTING = false;
	bDazed = false;
	A_TakeInventory("DazeToken", 1);
  }
  
  void A_StunEffect()
  {
    bNOPAIN = true;
	bNOINFIGHTING = true;
	
	if(!bStunSoft)
	{
	  A_SpawnItemEx("StunPrimary", zofs: self.height / 2);
	  
	  let itm = FindInventory("StunToken");
	  if(itm)
	    StunToken(itm).lifeTime = 210;
		
	  A_GiveInventory("StunToken", 1);
	}
	else
	{
	  let itm = FindInventory("StunTokenSoft");
	  if(itm)
	    StunTokenSoft(itm).lifeTime = 210;
		
	  A_GiveInventory("StunTokenSoft", 1);
	}
  }
  
  void A_ClearStun()
  {
    A_TakeInventory("StunToken", 1);
	A_TakeInventory("StunTokenSoft", 1);
	bStunned = false;
	bStunSoft = false;
	bNOPAIN = false;
	bNOINFIGHTING = false;
	StunTime = 0;
  }
  
  Actor A_DoAimedMissile(String mType, Vector3 SpawnPos)
  {
    Actor msl;
    double ang = self.angle; //Credit: Boondorl
	double s = sin(ang);
	double c = cos(ang);

	Vector3 newPos = Vec3Offset(SpawnPos.x * c + SpawnPos.y * s, SpawnPos.x * s - SpawnPos.y * c, SpawnPos.z);

    if(target && target is "DFPlayer")
	{
	  if(!DFPlayer(target).isDodging) //If target is player
	  {
	    msl = SpawnMissileXYZ(newPos, target, mType);
		if(msl)
		  msl.VelIntercept(target, -1);
	  }
	  else
	  {
	    SpawnMissileXYZ(newPos, target, mType);
	  }
	}
	
	else if(target) //If target isn't player
	{
	    msl = SpawnMissileXYZ(newPos, target, mType);
		if(msl)
		  msl.VelIntercept(target, -1);
	}
	
	return msl;
  }
  
  void A_TriggerBloodSplat(int distance)
  {
    let pmo = DFPlayer(players[consoleplayer].mo);
	if(pmo && Distance3D(pmo) <= distance)
	  pmo.BloodSplat();
  }
	
  action state A_TNTDeath()
  {
    int type = random(0,1);
	if(type == 1)
	{
	  ThrustThing(random(0,255), random(0,30), 0, 0);
	  ThrustThingZ(0, random(80,127), 0, 0);
	  DF_Monster(self).TNTDeath = true;
	  return ResolveState("Death");
	}
	return ResolveState(null);
  }
  
  void SetArmor(int value)
  {
    AP = value;
	APMax = value;
  }
  
  void SetCritical()
  {
    A_StartSound("Parry01", CHAN_7, 0, 1.0, 0.0);
	A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("BWAVA0"), STYLE_Add, SPF_RELATIVE, 35, 200,
	zoff: self.height * 0.5, startalphaf: 1.0, fadestepf: 0.1, sizestep: 50);
	A_SpawnParticleEx(Color(128, 64, 0), TexMan.CheckForTexture("EXPFA0"), STYLE_ADD, SPF_FULLBRIGHT, 50, 500, 0,
	zoff: self.height * 0.5, startalphaf: 0.5, fadestepf: 0.05);
	//Gotteem -- Critmark size is 1.3 if revert is needed
	let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	if(ev)
	  ev.SetParry(1.0, 0.04, 0.02, 30); 
  }
  
  void SetDrunk(double intensity, double zoom, double duration)
  {
    let pmo = players[consoleplayer].mo;
	if(pmo)
	{
      ShaderHandler ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	  if(ev)
	    ev.SetDrunk(intensity, zoom, duration);
	}
  }
  
  void SetParryWindow(bool on)
  {
    pOpen = on;
  }
  
  void A_SpawnGore(string texName, double xofs, double yofs, double zofs, double xvel, double yvel, double zvel, int angle, int size, int lifeTime, double acelz = -0.05)
  {
    A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture(texName),
	STYLE_NONE, SPF_ROLL|SPF_RELATIVE|SPF_LOCAL_ANIM, lifeTime, size, angle,
	xofs, yofs, zofs, xvel, yvel, zvel, accelz: acelz, startroll: random(0,359));
  }
  
  void SpawnGibPile(double scale = 1.0)
  {
    string cls = String.Format("GibPile%d", random(1,4));
    let actr = Spawn(cls, (pos.x, pos.y, floorz));
	if(actr)
	{
	  actr.scale.x = scale;
	  actr.scale.y = actr.scale.x * 0.83;
	  actr.bSpriteFlip = random(0,1);
	}
  }
  ////////////////////////////
  //PARRYING nd SHIXTZ zzz
  Actor flash;
  void ParryFlash(double xofs, double yofs, double zofs, double time)
  {
    double c = cos(angle);
	double s = sin(angle);
	Vector3 spawnPos = Vec3Offset(xofs * c + yofs * s, xofs * s - yofs * c, zofs);
	flash = Spawn("CounterFlash", spawnPos);
	if(flash)
	{
	  flash.master = self;
	  CounterFlash(flash).xw = xofs;
	  CounterFlash(flash).yw = yofs;
	  CounterFlash(flash).zw = zofs;
	  CounterFlash(flash).duration = time;
	}
  }
  
  void ParryFlashPosition(int xofs, int yofs, int zofs)
  {
    if(flash)
	{
	  CounterFlash(flash).xw = xofs;
	  CounterFlash(flash).yw = yofs;
	  CounterFlash(flash).zw = zofs;
	}
  }
  
   void SetParryTime(int time)
  {
    parry_time = time + CVar.FindCVar("modf_parry").GetInt();
  }
  
  
  States
  {
	Decay:
	  "####" "#" 0
	  {
	    if(self.TNTDeath)
		  return A_Jump(256, 2);
		return ResolveState(null);
	  }
	  "####" "#" 4;
	  Goto Poof;
	  "####" "#" 2 A_CheckFloor("Poof");
	  Wait;
	Poof:
	  TNT1 A 0 A_KillChildren("Fire", KILS_KILLMISSILES);
	  //TNT1 A 0 A_SpawnItemEx("BurnDeathAlt");
	  Stop;
  }
}

Class Duel_Prep : Inventory
{
  int bullet, ht_max, he_max, shell, gas, void_shell, heavy_bullet;
  override void AttachToOwner(Actor other)
  {
    if(other)
	{
	  other.GiveInventory("Skill_Exit", 1);
	  DFPlayer(other).bDuel_Buff = true;
	  DFPlayer(other).DoHeal(300);
	  DFPlayer(other).armorStack = DFPlayer(other).armorMax;
	  
	  let bl = other.FindInventory("BulletLoaded");
	  let ht = other.FindInventory("PistolAmmo");
	  let he = other.FindInventory("PierceAmmo");
	  let sh = other.FindInventory("ShellLoaded");
	  let gs = other.FindInventory("GasAmmo");
	  let vs = other.FindInventory("VoidLoaded");
	  let hb = other.FindInventory("HeavyLoaded");
	  
	  if(bl)
	    bullet = bl.amount;
		
	  if(ht)
	    ht_max = ht.MaxAmount;
		
	  if(he)
	    he_max = he.MaxAmount;
		
	  if(sh)
	    shell = sh.amount;
		
	  if(gs)
	    gas = gs.amount;
		
	  if(vs)
	    void_shell = vs.amount;
		
	  if(hb)
	    heavy_bullet = hb.amount;
	}
	
    Super.AttachToOwner(other);
  }
  
  override void DetachFromOwner()
  {
    let pmo = players[consoleplayer].mo;
    if(pmo)
	{
	  pmo.TakeInventory("Skill_Exit", 1);
	  DFPlayer(pmo).bDuel_Buff = false;
	  
	  let bl = pmo.FindInventory("BulletLoaded");
	  let ht = pmo.FindInventory("PistolAmmo");
	  let he = pmo.FindInventory("PierceAmmo");
	  let sh = pmo.FindInventory("ShellLoaded");
	  let gs = pmo.FindInventory("GasAmmo");
	  let vs = pmo.FindInventory("VoidLoaded");
	  let hb = pmo.FindInventory("HeavyLoaded");
	  
	  if(bl)
	    bl.amount = bullet;
		
	  if(ht)
	    ht.MaxAmount = ht_max;
		
	  if(he)
	    he.MaxAmount = he_max;
		
	  if(sh)
	    sh.amount = shell;
		
	  if(gs)
	    gs.amount = gas;
		
	  if(vs)
	    vs.amount = void_shell;
		
	  if(hb)
	    hb.amount = heavy_bullet;
	}
	
    Super.DetachFromOwner();
  }
  
  Default
  {
    Inventory.Amount 1;
	Inventory.MaxAmount 1;
  }
}

Class CounterFlash_FX : Actor
{
  double duration, fade_step;
  double step_x, step_y;
  
  override void PostBeginPlay()
  {
    fade_step = 2.0 / duration; //duration
	if(duration > 6)
	{
	  step_x = 1.6;
	  step_y = 0.15;
	}
	else if(duration <= 6)
	{
	  step_x = 1.6;
	  step_y = 0.2;
	  scale.x = 6.0;
	  scale.y = 0.75;
	}
	//else
	  //step_y = 0.1;
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(scale.x < 12.0)
	  scale.x *= step_x;
	
	if(scale.y > 0.2)
	  scale.y -= step_y;
	
	A_Warp(AAPTR_MASTER, flags: WARPF_ABSOLUTEANGLE|WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE);
  }
  
  Default
  {
	Alpha 0.0;
    RenderStyle "AddShaded";
	StencilColor "994C00";
    +BRIGHT
	+NOINTERACTION
  }
  States
  {
    Spawn:
	  TNT1 A 0;
      CFLF AA 1 A_FadeIn(0.5);
	  CFLF A 0
	  {
	    Alpha = 2.0;
		SetShade("FFFF66");
	  }
	  CFLF A 1
	  {
	    A_FadeOut(fade_step);
	  }
	  Wait;
  }
}

Class MonsterGlitch : Actor
{
  int offs, offz;
  
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	if(offs == 0)
	  offs = 8;
	  
	if(offz == 0)
	  offz = 1;
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(master)
	{
