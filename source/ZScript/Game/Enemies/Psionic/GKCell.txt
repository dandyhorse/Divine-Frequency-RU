Class GKCell : DF_Monster
{
  double ang;
  bool user_wait;
  Array<Actor> tents;
  
  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)
  {
    Super.Die(source, inflictor, dmgFlags, MeansOfDeath);
	
    A_KillChildren("Fire", KILS_KILLMISSILES);
	A_GiveExp(1000);
  }
  
  void A_DeployTentacles() //Did you know it's actually called arms and not tentacles irl?
  {
    for(int i = 0; i <= 315; i += 45 )
	{
	  double angl = angle + i;
	  Vector3 sPos = Vec3Offset(cos(angl) * 96.0, sin(angl) * 96.0, 3);
	  let actr = Spawn("CellTentacle", sPos);
	  if(actr)
	  {
		actr.angle = angl;
		actr.master = self;
		CellTentacle(actr).ang = angl;
		tents.Push(actr);
	  }
	}
  }
  
  override void BeginPlay()
  {
    radar_height = Height * 0.5;
	unScannable = true;
	A_DeployTentacles();
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(!user_wait)
	{
	  let fade = Spawn("MonsterGlitch", Pos);
	  if(fade)
	  {
	    fade.sprite = sprite; 
	    fade.frame = frame;
	    fade.scale = scale;
	    fade.angle = angle;
	    fade.master = self;
		fade.bFORCEXYBILLBOARD = true;
	    MonsterGlitch(fade).offs = 20;
		MonsterGlitch(fade).offz = 4;
		  
	    int glow = random(0,3);
	    if(glow < 3)
	      fade.A_SetRenderStyle(frandom(0.7, 0.9), STYLE_ADD);
	    else
	      fade.A_SetRenderStyle(frandom(0.7, 0.9), STYLE_Translucent);
	  }
	}
  }

  Default
  {
    //$Category "Enemies/Psionic"
	//$Color 4
    Health 1500;
    Radius 80;
    Height 160;
	XScale 2.0;
	YScale 1.66;
	PainChance 0;
    Speed 3;
	RenderStyle "Translucent";
	FloatBobStrength 0.3;
	PainChance "Poison", 0;
	DamageFactor "Trauma", 2.0;
	DamageFactor "Void", 10.0;
	DamageFactor "Explode", 0.5;
	Mass 500;
    Monster;
	Species "Giant Killer Cell";
	SeeSound "Cosmosgenesis/Active";
	ActiveSound "Cosmosgenesis/Active";
	+DONTFALL
	+DONTTHRUST
	+FLOAT
	+FLOATBOB
	+FORCEXYBILLBOARD
	+NOGRAVITY
	+PAINLESS
	+THRUSPECIES
	+NOSPRITESHADOW
	+DF_Monster.StunResist;
	+DF_Monster.PossessImmune
  }
  States
  {
    Spawn:
	  TNT1 A 0;
	  TNT1 A 0
	  {
	    if(user_wait)
		  return ResolveState("Chill");
		  
		return ResolveState(null);
	  }
	  GRRM ABCDEFGHIJK 5;
	  GRRM A 0 A_StartSound("GK_See", CHAN_AUTO, attenuation: 0.05);
	  GRRM ABCDEFGHIJKABCDEFGHIJK 5;
	Idle:
      GRRM ABCDEFGHIJK 5 A_Look();
	  Loop;
	Chill:
	  TNT1 A 0 { alpha = 0.0; }
	  TNT1 A -1;
	  Stop;
	Ready:
	  TNT1 A 0
	  {
	    A_StartSound("GK_See", CHAN_AUTO, attenuation: 0.05);
	    A_StartSound("GK_Act", CHAN_AUTO, CHANF_LOOP, attenuation: 0.5);
		user_wait = false;
	  }
	  GRRM JJJJJKKKKK 1 { alpha += 0.1; }
	  Goto Idle;
    See:
      GRRM AAAAABBBBBCCCCCDDDDDEEEEEFFFFFGGGGGHHHHHIIIIIJJJJJKKKKK 1 A_Chase();
      Loop;
	Missile:
	  GRRM G 5
	  {
	    A_FaceTarget();
		if(angle > 360)
		{
		  int rotations = round(angle / 360.0);
		  angle = angle - (rotations * 360.0);
		}
		else if(angle < 0)
		  angle = 180 + (180 - abs(angle));

		int indFinder = round(angle / 45.0);
		indFinder += random(-1,1);
		if(indFinder >= tents.Size())
		  indFinder = 0;
		else if(indFinder < 0)
		  indFinder = 7;

	    if(target)
		{
		  CellTentacle(tents[indFinder]).focusTarget = target;
		  tents[indFinder].SetStateLabel("Missile");
		}
	  }
	  GRRM HIJK 5 A_FaceTarget();
	  GRRM AAAAABBBBBCCCCCDDDDDEEEEEFFFFFGGGGGHHHHHIIIIIJJJJJKKKKK 1 A_Wander();
	  Goto See;
	Death:
	  TNT1 A 1 A_SpawnItemEx("GKC_Death", zofs: 78);
	  Stop;
  }
}

Class CellTentacle : Actor
{
  Actor focusTarget;
  int shift;
  double ang;
  override void Tick()
  {
    Super.Tick();
	
	if(master)
	{
	  A_Warp(AAPTR_MASTER, 96, 0, 3, ang, WARPF_ABSOLUTEANGLE|WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE);
	  shift = random(0,2);
	  if(shift == 0)
	    A_SetRenderStyle(frandom(0.8,1.0), STYLE_Add);
	  else if(shift == 1)
	    A_SetRenderStyle(frandom(0.8,1.0), STYLE_Translucent);
	  else
	    A_SetRenderStyle(1.0, STYLE_Normal);
	}
	else
	  A_FadeOut(0.1);
  }
  Default
  {
   // RenderStyle "Translucent";
	Alpha 1.0;
	FloatBobStrength 0.4;
	+BRIGHT
	+FLOATBOB
    +NOINTERACTION
	XScale 1.42;
	YScale 1.42;
  }
  States
  {
    Spawn:
	  TNT1 A 0 NoDelay
	  {
	    A_SetFloatBobPhase(random(0,63));
	    return A_Jump(256, random(1,18));
	  }
	  0000 ABCDEFGHI random(18,30);
	  0000 IHGFEDCBA random(18,30);
	  Goto Spawn+1;
	Missile:
	  0000 "#" 8;
	  0000 JKL 10;
	  0000 M 10
	  {
	    A_StartSound("GKCell/Charge", CHAN_AUTO, attenuation: 0.2);
	    if(focusTarget)
		{
		  Vector3 newPos = Vec3Offset(cos(angle) * 228, sin(angle) * 228, 100);
		  let laserOrigin = Spawn("GKLaserMain", newPos);
		  
		  Vector3 sPos = focusTarget.Vec3Offset(64.0 * cos(focusTarget.angle), 64.0 * sin(focusTarget.angle), 0);
		  Spawn("GKLaserSpark", sPos);
		  let actr = Spawn("mMissileTarget", sPos);
		  if(laserOrigin && actr)
		  {
		    laserOrigin.target = actr;
			GKLaserMain(laserOrigin).targPos = actr.pos;
		  }
		}
	  }
	  0000 LK 10;
	  0000 JIHGFEDCBA random(18,30);
	  Goto Spawn+1;
  }
}

Class GKLaserMain : Actor
{
  Vector3 targPos;
  
  override void Tick() 
  {
    Super.Tick();
	
	if(Alpha < 1.0)
	  A_FadeIn(0.05);
	  
	A_SetScale(scale.x + 0.05);
	
	if(GetAge() == 14)
	  A_SpawnItemEx("GKLaserExplode");
	  
	if(GetAge() >= 14 && GetAge() < 18)
	{
	  let actr = Spawn("mMissileTarget", targPos);
	  if(actr)
	  {
	    pitch = PitchTo(actr) + 90;
	    A_SpawnItemEx("LaserWarning", xvel: Distance2D(actr) / (20.0 - GetAge()), zvel: (actr.pos.z - pos.z) / (20.0 - GetAge()), flags: SXF_TRANSFERPITCH);
	  }
	}
	
	if(GetAge() > 20)
	  A_FadeOut(0.1);
  }
  Default
  {
    Scale 0.5;
    RenderStyle "Add";
    +BRIGHT
	+FORCEXYBILLBOARD
    +NOINTERACTION
  }
  States
  {
    Spawn:
	  TNT1 A 1;

{

  {
    Super.Tick();

	{
	  bool left = random(0,1);
	  A_SpawnParticleEx

			"FFFFFF",
			TexMan.CheckForTexture("SMMMA0"),

			flags: SPF_RELATIVE|SPF_ROLL,

			TexMan.CheckForTexture("SPFLC0"),
			style: STYLE_ADD,

	    if(GetAge() > 20)

		return ResolveState(null);
	  }
	  Wait;

	  TNT1 A 0
	  {
	    vel = (0, 0, 0);
		A_AlertMonsters();
        A_Explode(60, 230, XF_NOTMISSILE , 0, 230, damagetype: "Energy");
		A_QuakeEx(4, 9, 9, 35, 0, 2048, "DSEMPTY", QF_RELATIVE|QF_SCALEDOWN|QF_SHAKEONLY);

		if(master)
		  master.Destroy();
	  }
	  TNT1 A 0
	  {
	    let pmo = players[consoleplayer].mo;
		if(pmo && !DFPlayer(pmo).bResistRad && DFPlayer(pmo).iFrames <= 0)
		{
		  double dist = Distance3D(pmo);
		  
		  if(dist > 270)
		    return;
		  else if(dist <= 270)
		  {
		    double radnum = clamp((1.0 - (dist * 0.004)) * 25, 1, 25);
			DFPlayer(pmo).SetRadiation(int(radnum));
		  }
		}
	  }
	  TNT1 A 1

		A_StartSound("Electric/Explosion", CHAN_AUTO, attenuation: 0.4);
