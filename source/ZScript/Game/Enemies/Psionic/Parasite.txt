Class Parasite : DF_Monster

  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod)
  {
    if(source && source.GetClassName() != self.GetClassName())
	{
	  double dmg = damage * 0.5;
	  source.DamageMobj(self, self, dmg, "Reflect");
	}
	  
	return Super.DamageMobj(inflictor, source, damage, mod);
  }

Class ParasiteMissile : Actor
{
  Default
  {
	Radius 6;
	Height 12;
	Speed 50;
	Damage 5;
	Scale 0.5;
	Projectile;
	Renderstyle "Add";
	DamageType "Psionic";
	+FLATSPRITE
  }
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	let spd = sqrt((vel.x * vel.x) + (vel.y * vel.y));
	let pit = atan2(spd, vel.z);
	pitch = pit;
  }
  States
  {
	Spawn:
	  BLTN B 1 Bright A_SpawnItemEx("ParasiteMissileWave", flags: SXF_TRANSFERPITCH);
	  Wait;
	Death:
	  TNT1 A 1;
	  Stop;
  }
}

Class BobSpineGib : Actor

  }
  
  Default
  {
    Radius 8;
    Height 8;
    Speed 10;
	XScale 1.2;
	YScale 1.0;
	Gravity 1.0;
	BounceType "Doom";
    BounceCount 3;
    BounceFActor 0.6;

    +THRUACTORS
	+MISSILE
    +BLOODLESSIMPACT 
	+FORCEXYBILLBOARD
	+ROLLSPRITE
	+USEBOUNCESTATE
  }
  
  States
  {
    Spawn:
	  TNT1 A 0;
	  BBSD A 0 { frame = random(0,1); }
	  BBSD "#" random(2,4);
	  BBSD "#" 0 A_Jump(128, "ReverseSpin");
	  
	Spin:
	  BBSD "#" 3
	  {
	    roll += 16;
		angle += 32;
	  }
	  Wait;
	ReverseSpin:
	  BBSD "#" 3
	  {
	    roll -= 16;
		angle -= 32;
	  }
	  Wait;
	Bounce:
	  BBSD "#" 3
	  {
	    roll += random(0,359);
		angle += random(0,359);
	  }
	  Goto Spin;
	Death:
	  BBSD "#" 700
	  {
	    bABSMASKANGLE = true;
	    roll = 0;
	  }
	  BBSD "#" 1 A_FadeOut(0.15);
	  Wait;
  }
}
