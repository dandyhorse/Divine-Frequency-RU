  override void Tick()
  {
    Super.Tick();
	
	if(InStateSequence(CurState, ResolveState("Missile")))
	  A_FaceTarget();
	  
	if(stealthCount < 200)
	{
	  if(Health > 0 && stealthCount % 80 == 0)
	    Spawn("Bird_Glitch", pos);
		
	  stealthCount++;
	}
	
	if(active_snd_cd > 0)
	  active_snd_cd--;
  }

  void TriggerActiveSound()
  {
    if(active_snd_cd <= 0 && active_snd.Size() > 0)
	{
	  active_snd_cd = 35 * random(7,10);
      int rand = random(0, active_snd.Size() - 1);
	  A_StartSound(active_snd[rand], CHAN_AUTO, volume: 0.64, attenuation: 0.1);
	  active_snd.Delete(rand);
	}
  }
  
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	LoadActiveSounds();
	unScannable = true;
	DFPlayer(players[consoleplayer].mo).hActr = null;
	stealthCount = 200;
  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
    let pmo = DFPlayer(players[consoleplayer].mo);
    if(inflictor && (inflictor is "PistolPuff" || inflictor is "ARPuff") && (inflictor.pos.z - pos.z >= 107))
	{
	  if(pmo && DFPlayer(pmo).bBluePower)
	    damage *= 2.0;
	  else
	    damage *= 1.5;
	 }
	 
	if(mod == "Pen" && damage > 100)
	  if(DFPlayer(players[consoleplayer].mo).bBase_Str && players[consoleplayer].ReadyWeapon is "Hands")
	    SetStateLabel("Pain.Shotgib");
	   
    return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);        
  }
  
  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)

	{
	  if(DFPlayer(players[consoleplayer].mo).bStrOrigin && players[consoleplayer].ReadyWeapon is "Hands")
		  Hands(players[consoleplayer].ReadyWeapon).Heal();

	    SetStateLabel("CloseShotGib");
	  else if(players[consoleplayer].mo && Distance2D(players[consoleplayer].mo) > 156)
	    SetStateLabel("MediumShotGib");
	}
		
	bStealth = false;
  }
