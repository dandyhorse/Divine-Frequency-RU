Class Antibody : DF_Monster
{
  Actor clone;
  bool duplicating;
  int dup;
  
  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)
  {
    A_KillChildren("Fire", KILS_KILLMISSILES);
	A_GiveExp(220);
	
	super.Die(source, inflictor, dmgFlags, MeansOfDeath);
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(duplicating)
	  A_SpawnItemEx("AntibodyTrail", flags: SXF_TRANSFERSPRITEFRAME|SXF_TRANSFERSCALE);
  }
  
  override void BeginPlay()
  {
    Super.BeginPlay();
	
	dup = 5;
  }

  Default
  {
    //$Category "Enemies/Psionic"
	//$Color 4
    Health 50;
    Radius 20;
    Height 40;
	Scale 0.5;
    Speed 2;
	MeleeRange 64;
	FloatBobStrength 0.3;
    Monster;
	Species "Abstract";
	DamageFactor "Poison", 0;
	DamageFactor "Trauma", 2.0;
	SeeSound "Germ/See";
	ActiveSound "Germ/Active";
	+DF_Monster.NoRadar
	+DONTTHRUST
	+DONTFALL
	+FLOAT
	+FLOATBOB
	+FORCEXYBILLBOARD
	+NOBLOOD
	+NOGRAVITY
  }
  States
  {
    Spawn:
      JERM ABCD 6 A_Look();
	  Loop;
    See:
      JERM AAAAAABBBBBBCCCCCCDDDDDD 1
	  {
	    int chance = random(1,100);
		if(chance <= 20 && dup > 0)
		{
		  dup--;
		  return ResolveState("Duplicate");
		}  
		  
	    A_Chase();
		return ResolveState(null);
	  }
      Loop;
	Melee:
	  "####" "#" 0 A_Die();
	  Goto Death;
	Duplicate:
	  TNT1 A 0 A_StartSound("BGSplit1", CHAN_AUTO, attenuation: 0.5);
	  JERM CD 8;
	  TNT1 A 0
	  {
		duplicating = true;
		bDONTTHRUST = false;
		bSHOOTABLE = false;
		bTHRUSPECIES = true;
		
	    int sangle = randompick(-90,90);
		int cangle;
		
		if(sangle > 0)
		  cangle = -90;
		else
		  cangle = 90;
		
		Thrust(3, angle + sangle);
	    A_SetRenderStyle(1.0, STYLE_ADD);
		A_StartSound("brain/spit", CHAN_AUTO);
		
	    clone = Spawn("AntibodyDup", pos);
		if(clone)
		{
		  clone.A_StartSound("brain/spit", CHAN_AUTO);
		  clone.Thrust(3, angle + cangle);
		  AntibodyDup(clone).duplicating = true;
		}
	  }
	  JERM ABCD 6;
	  TNT1 A 0
	  {
	    A_Stop();
		A_SetRenderStyle(1.0, STYLE_Translucent);
		duplicating = false;
		bDONTTHRUST = true;
		bSHOOTABLE = true;
		bTHRUSPECIES = false;
		
	    if(clone && target)
		{
		  clone.bDONTTHRUST = true;
		  clone.bSHOOTABLE = true;
		  clone.bTHRUSPECIES = false;
		  AntibodyDup(clone).duplicating = false;
		  clone.SetStateLabel("Spawn");
		  clone.A_Stop();
		  clone.A_SetRenderStyle(1.0, STYLE_Translucent);
		}
	  }
	  Goto See;
	Death:
	  "####" "#" 1 A_SpawnItemEx("AntibodyFlare", zofs: 20);
	  TNT1 A 1
	  {
	    A_StartSound("BODSPLT3", CHAN_AUTO, volume: 0.35, attenuation: 0.6);
		A_SpawnItemEx("AntibodyDeath", 0, 6, 27.5);
		A_Explode(random(25,30), 144, XF_NOTMISSILE, 0, 144, damagetype: "Physical");
		
		let pmo = players[consoleplayer].mo;
		if(pmo && Distance3D(pmo) <= 144)
		  if(!DFPlayer(pmo).isDodging && !DFPlayer(pmo).bResistToxin)
			  DFPlayer(pmo).SetToxin(random(8,11));
		
		for(int i = 0; i < 12; i++)
		{
		  int pitch = random(-90,90);
		  int angle = random(0,359);
		  Vector3 newPos = 8 * (cos(pitch) * cos(angle), cos(pitch) * sin(angle), sin(pitch));
		  Vector3 newVel = 0.05 * newPos;
		  Actor puff = Spawn("AntibodyGas", pos + newPos + (0,0,20));
		  if(puff)
		    puff.vel = newVel;	
		}
		for(int i = 0; i < 32; i++)
		{
		  int pitch = random(-90,90);
		  int angle = random(0,359);
		  Vector3 newPos = 8 * (cos(pitch) * cos(angle), cos(pitch) * sin(angle), sin(pitch));
		  Vector3 newVel = 0.1 * newPos;
		  A_SpawnParticleEx(Color(204, 204, 0), TexMan.CheckForTexture("SPFLA"), STYLE_ADD, SPF_FULLBRIGHT|SPF_RELATIVE, 105, 4, 0,
		  newPos.x, newPos.y, newPos.z + 20, newVel.x * 1.2, newVel.y * 1.2, newVel.z);
		}
	  }
	  TNT1 AAAAAAAAAA 2
	  {
	    for(int i = 0; i < 16; i++)
		{
		  int pitch = random(-90,90);
		  int angle = random(0,359);
		  Vector3 newPos = 16 * (cos(pitch) * cos(angle), cos(pitch) * sin(angle), sin(pitch));
		  Vector3 newVel = 0.05 * newPos;
		  A_SpawnParticleEx(Color(204, 204, 0), TexMan.CheckForTexture("SPFLA"), STYLE_ADD, SPF_FULLBRIGHT|SPF_RELATIVE, 105, 4, 0,
		  newPos.x, newPos.y, newPos.z + 20, newVel.x * 1.2, newVel.y * 1.2, newVel.z);
		}
	  }
	  Stop;
  }
}

Class AntibodyDup : Antibody
{
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	dup = 0;
  }
  
  Default
  {
    RenderStyle "Add";
    -DONTTHRUST
	-SHOOTABLE
	+THRUSPECIES
  }
}

Class AntibodyGas : Actor
{
  double rlAmt;
  
  override void BeginPlay()
  {
    Super.BeginPlay();
	
	roll = random(0,359);
	rlAmt = frandom(-0.5, 0.5);
  }
  override void Tick()
  {
    Super.Tick();
	
	A_SetRoll(roll + rlAmt, SPF_INTERPOLATE);
	
	if(scale.x < 1.0)
	{
	  scale.x += 0.05;
	  scale.y += 0.05;
	}
	else
	  if(GetAge() % 4 == 0)
	  {
	    scale.x += 0.005;
	    scale.y += 0.005;
	  }
	
	if(GetAge() < 10)
	  A_FadeIn(frandom(0.05, 0.07));
	
	if(GetAge() > 70)
	  A_FadeOut(0.02);
  }
  
  Default
  {
    RenderStyle "Translucent";
	Alpha 0.0;
	Scale 0.2;
	+FORCEXYBILLBOARD
	+NOINTERACTION
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  GAS1 A 0 NoDelay { frame = random(0,3); }
	  GAS1 "#" -1;
	  Stop;
  }
}
