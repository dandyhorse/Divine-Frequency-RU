Class Mailbox : ModelBase
{
  void SpawnAmmo(string type)
  {
    int velx = random(5,6);
	double vely = frandom(-1.0,1.0);
	Vector3 iVel = (velx * cos(angle) + vely * sin(angle) , velx * sin(angle) - vely * cos(angle), random(4,5));
	let itm = Spawn(type, pos);
	if(itm)
	{
	  itm.A_SetRenderStyle(0.0, STYLE_TRANSLUCENT);
	  itm.vel = iVel;
	  itm.A_GiveInventory("AlphaReset", 1);
	}
  }
  
  Default
  {
	Radius 21;
	Height 73;
	Scale 1.3;
	+SHOOTABLE
	+CANPASS
	+SOLID
	+NOBLOOD
	+NOBLOODDECALS
	+DONTTHRUST
	+DONTFACETALKER
  }
  States
  {
    Talk:
	  0000 A 0
	  {
	    if(players[consoleplayer].mo)
		  angle = AngleTo(players[consoleplayer].mo);
	  }
	  0000 A -1;
	  Stop;
    HT:
	  0000 A 0 SpawnAmmo("HTMag");
	  Goto Spawn;
	AP:
	  0000 A 0 SpawnAmmo("APMag");
	  Goto Spawn;
	Shell:
	  0000 A 0 SpawnAmmo("ShellMag");
	  Goto Spawn;
  }
}

Class Bob_01 : Actor
{
  bool acs;
  int lookTime;
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	let pmo = players[consoleplayer].mo;
	if(pmo)
	  target = pmo;
  }
  override void Tick()
  {
    Super.Tick();
	
	if(CheckIfInTargetLOS(90))
	  lookTime++;
	  
	if(lookTime > 80)
	  A_FadeOut(0.05);
	  
	if(alpha <= 0.05 && !acs)
	{
	  acs = true;
	  ACS_NamedExecute("AlienGrow");
	}
  }
  Default
  {
    RenderStyle "Translucent";
    YScale 0.83;
	+NOINTERACTION
  }
  States
  {
    Spawn:
	  RGD1 PONMLKJIHGFEDCBA 3;
	  Loop;
  }
}
