		  x = random(0,640);

		for(double i = 0.0; i < 360.0; i += 15.0)
		  for(int j = -75; j < 90; j += 15)
		  {
		    Vector3 newPos = 512 * (cos(j) * cos(i), cos(j) * sin(i), sin(j) * 0.83);
			let actr = Spawn("ChampionHammerSpark", pos + newPos);

  bool spawn_heal;
  int tics;
  
  override void Tick()
  {
    Super.Tick();
	
	if(!master)
	  Destroy();
	else
	  SetOrigin(master.pos, false);
  }
  void LaserBeam(double xofs, double yofs, double zofs, double targetDistance)
  {
    double c = cos(angle);
	double s = sin(angle);
	Vector3 spawnPos = Vec3Offset(xofs * c + yofs * s, xofs * s - yofs * c, zofs);
	let pmo = players[consoleplayer].mo;
	if(pmo)
	{
	  Vector3 targetPos = pmo.Vec3Offset(targetDistance * sin(pmo.angle), targetDistance * cos(pmo.angle), 0);
	  let Laser = Spawn("Six_LaserBeam", spawnPos);
	  if(Laser)
	    Six_LaserBeam(Laser).targPos = targetPos;
	}
  }
  Default
  {
    +NOINTERACTION
  }
  States
  {
    Spawn:
	  TNT1 A -1;
	  Stop;
	ShootSlow:
	  TNT1 A 0 { tics = 105; }
	  Goto Fire;
	ShootFast:
	  TNT1 A 0 { tics = 50; }
	  Goto Fire;
	ShootFaster:
	  TNT1 A 0 
	  { 
	     tics = 35; 
		 spawn_heal = true;
	  }
	  Goto Fire;
	NoMercy:
	  TNT1 A 0 { tics = 20; }
	  Goto Fire;
	Fire:
	  TNT1 A 70
	  {
	    A_SetTics(tics);
	    A_StartSound("GKCell/Charge", CHAN_AUTO, attenuation: 0.1);
	    LaserBeam(-136, 188, 154, -64);
		LaserBeam(-136, -188, 154, 64);
		let pmo = players[consoleplayer].mo;
		if(pmo)
		{
		  let actr = Spawn("Six_LaserBeamExplosion", pmo.pos);
		  if(actr && spawn_heal)
		    Six_LaserBeamExplosion(actr).spawn_heal = true;
		}
	  }
	  TNT1 A 70
	  {
	    A_SetTics(tics);
	    A_StartSound("GKCell/Charge", CHAN_AUTO, attenuation: 0.1);
	    LaserBeam(-136, 188, 522, -64);
		LaserBeam(-136, -188, 522, 64);
		let pmo = players[consoleplayer].mo;
		if(pmo)
		{
		  let actr = Spawn("Six_LaserBeamExplosion", pmo.pos);
		  if(actr && spawn_heal)
		    Six_LaserBeamExplosion(actr).spawn_heal = true;
		}
	  }
	  TNT1 A 70
	  {
	    A_SetTics(tics);
	    A_StartSound("GKCell/Charge", CHAN_AUTO, attenuation: 0.1);
	    LaserBeam(-136, 188, 890, -64);
		LaserBeam(-136, -188, 890, 64);
		let pmo = players[consoleplayer].mo;
		if(pmo)
		{
		  let actr = Spawn("Six_LaserBeamExplosion", pmo.pos);
		  if(actr && spawn_heal)
		    Six_LaserBeamExplosion(actr).spawn_heal = true;
		}
