  override void Tick()
  {
    Super.Tick();

	//console.printf("Ability1:%d Ability2:%d\nAbility3:%d Ability4:%d\n", a1cd, a2cd, a3cd, a4cd);

	  SetMusicVolume(mDamp);
	}

	  a1CD--;
	  
	if(a2CD > 0)
	  a2CD--;
	  
	if(a3CD > 0)
	  a3CD--;
	  
	if(a4CD > 0)
	  a4CD--;
	  
	if(parry_time > 0)
	  parry_time--;
	
	if(staggerTrigger == 1)
	{
	  SetStateLabel("Stagger");
	  staggerTrigger--;
	}
	else if(staggerTrigger > 0)
	  staggerTrigger--;
	   
	if(staggerTime == 1)
	{
	  SetStateLabel("StaggerFinish");
	  staggerTime--;
	}
	else if(staggerTime > 0)
	  staggerTime--;
  }
  
  void DirectDamage(string dam_type)
  {
    if(dam_type == "Trauma")
	  boss_StaggerBlock -= 15;
	else if(dam_type == "Explode")
	  boss_StaggerBlock -= 50;
	else if(dam_type == "ShotGib")
	  boss_StaggerBlock -= 5;
	else if(dam_type == "SGShock")
	  boss_StaggerBlock -= 7;
	  
	if(boss_StaggerBlock <= 0 && !CheckInventory("BlockStackRefill", 1))
	  staggerTrigger = 1;
  }
  
 override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
    if(parry_time > 0 && target && Distance2D(target) <= 777 && (mod == "ShotGib" || mod == "SGShock"))
	{
	  damage *= 0.1;
	  staggerTrigger = 1;
	  vel = (0,0,0);
	}
	else if(parry_time <= 0 && !boss_staggered)
	{
	  if(players[consoleplayer].mo && DFPlayer(players[consoleplayer].mo).bPhyOrigin)
	  {
	    damage *= 0.1;
		DirectDamage(mod);
	  }
	  else if(mod == "SGShock" && !user_noFlash)
	  {
	    damage *= 0.1;
	    DirectDamage("SGShock");
	  }
	  else
	  {
	    if(source && source is "DFPlayer")
	    {
		  damage *= 0.1;
	      GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	      if(mod != "None" && ev)
	        ev.SetShieldIcon();
		}
	  }
	}
	if(boss_staggered)
	{
	  if(mod == "Trauma")
		damage *= 3.0;
	  else if(mod == "SGShock" && !user_noFlash)
	    damage *= 0.7;

	  staggerTime = 2;
	  if(source && source is "DFPlayer")
	    source.GiveBody(source.GetMaxHealth(True));
	}
	   
    return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);
  }

  action void ChampMelee(double ang, bool thrustTarget = True, string snd = "PH_Hit")
  {
    A_CustomMeleeAttack(random(4,6)*6, "DSEMPTY", damagetype: "Physical");
	
    if(target && CheckMeleeRange())
	{
	  target.A_StartSound(snd, CHAN_AUTO, attenuation: 0.4);
	  
	  if(thrustTarget)
	    target.Thrust(64, angle + ang);
		  
	  if(target is "DFPlayer")
		target.A_QuakeEx(1, 1, 1, 5, 0, 128);
	}
  }
  
  action void ChampGlitchTrail()
  {
    let fade = Spawn("MonsterGlitch", pos);
	if(fade)
	{
	  fade.sprite = sprite; 
	  fade.frame = frame;
	  fade.scale = scale;
	  fade.angle = angle;
	  fade.master = self;
		  
	  int glow = random(0,3);
	  if(glow < 3)
	    fade.A_SetRenderStyle(frandom(0.7, 0.9), STYLE_ADD);
	  else
	    fade.A_SetRenderStyle(frandom(0.7, 0.9), STYLE_Translucent);
	}
  }
  
  action void ChampSparkPop(int amount)
  {
    for(int i = 0; i < amount; i++)
	{
	  int pitch = random(-90,90);
	  int angle = random(0,359);	  
	  Vector3 newPos = 24 * (cos(pitch) * cos(angle), cos(pitch) * sin(angle), sin(pitch)); //ALL plushiinations
	  Vector3 newVel = 0.5 * newPos;
	  
	  Actor puff = Spawn("ChampionThunderSpark", pos + newPos);
	  if(puff)
		puff.vel = newVel;
	}
  }
  
  action void ChampionSlamSpark()
  {
    for(int i = 0; i < 18; i++)
	{
	  int pitch = random(2,32);
	  int ang = angle + random(-90,90);
		  
	  Vector3 newPos = 32 * (cos(pitch) * cos(ang), cos(pitch) * sin(ang), sin(pitch)); //ALL plushiinations
	  Vector3 basePos = Vec3Offset(94 * cos(ang), 94 * sin(ang), 0);
	  Vector3 newVel = 1.5 * newPos;
		  
	  Actor puff = Spawn("ChampionHammerSpark",  basePos + (random(24,32) * (cos(pitch) * cos(ang), cos(pitch) * sin(ang), sin(pitch))));
	  if(puff)
	    puff.vel = newVel;
    }
  }

	A_SpawnItemEx("SoulOrbVertices", flags: SXF_SETMASTER);
	let orb = Spawn("SoulOrbBall", pos);
	if(orb)

	  orb.A_SetScale(0.45, 0.37);
	  orb.master = self;
	}
  }
  
  override void Tick()
  {
	A_SetRoll(roll + 0.5, SPF_INTERPOLATE);
	A_SpawnItemEx("ElectricArc", random(-32,32), random(-32,32), random(-32,32));
	
	if(!pick_up && GetAge() % 10 == 0)
	{
	  let pmo = players[consoleplayer].mo;
	  if(pmo && Distance2D(pmo) <= 224)
	  {
	    pick_up = true;
		SetStateLabel("Death");
	  }
	}
	
	Super.Tick();
  }
  
  Default
  {
	Scale 0.3;
	+BRIGHT
	+FORCEXYBILLBOARD
	+NOINTERACTION
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  ELB1 ABCDEFGHIJKLMNOPQRSTUVWXYZ 2 Bright;
	  ELB2 ABCDEFGHIJ 2 Bright;
	  Loop;
	Death:
	  TNT1 A 1
	  {
	    let pmo = players[consoleplayer].mo;
		if(pmo)
		{
		  pmo.ACS_NamedExecute("Duel_PowerTip");
		  let wpn = Trenchgun(pmo.FindInventory("Trenchgun"));
		  if(wpn)
		  {
		    wpn.AddCharge();
			
		    let wep = players[consoleplayer].ReadyWeapon;
			let psp = players[consoleplayer].FindPSprite(PSP_Weapon);
		    if(wep && wep is "TrenchGun" && psp && psp.CurState.InStateSequence(wep.ResolveState("Idle")))
			  players[consoleplayer].SetPSprite(3, wep.ResolveState("Charge"));
		  }
		  
		  pmo.A_SetBlend("Light Blue", 0.37, 45);
		  pmo.A_StartSound("Psi_Up3", CHAN_AUTO, volume: 0.6);
		  GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		  if(ev)
		  {
			ev.UpdateChat(String.Format("· %s \cJconsumes \cFRaw Psi.", pmo.GetTag()));
			ev.UpdateChat("· \cITrench Gun Empowered!");
		  }
		}
	  }
	  Stop;
  }
