  {
    if(source && source.GetClassName() != self.GetClassName()) //Good god, can you imagine if these 2 reflect damage back at each other infinitely?
	{
	  double newDam = 0.5 * damage;
	  source.DamageMobj(self, self, newDam, "Reflect");
	  
	  if(source is "DFPlayer" || source is "DF_Monster")
	  {
	    let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	    if(ev)
	      ev.UpdateChat(String.Format("%s reflects \cRDamage (%d) \cJto %s.", self.GetTag(), int(newDam), source.GetTag()));
	  }
	}
	  
	return Super.DamageMobj(inflictor, source, damage, mod);
  }

Class GaryEvasion : Inventory
{
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
	
	if(owner && DF_Monster(owner).StunTime <= 0)
	  if(!owner.InStateSequence(owner.CurState, owner.ResolveState("Bombardment")) && !owner.InStateSequence(owner.CurState, owner.ResolveState("FlameBlast")))
	    owner.SetStateLabel("Evasion");
	  
	Destroy();
  }
  
  Default
  {
    Inventory.Amount 1;
  }
}
