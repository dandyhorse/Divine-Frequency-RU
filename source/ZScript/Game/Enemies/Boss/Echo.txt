  void DirectDamage(string dam_type)
  {
    if(dam_type == "Trauma")
	  boss_StaggerBlock -= 5;
	else if(dam_type == "Explode")
	  boss_StaggerBlock -= 17;
	else if(dam_type == "ShotGib")
	  boss_StaggerBlock -= 2;
	else if(dam_type == "Pen")
	  boss_StaggerBlock -= 5;
	else if(dam_type == "SGShock")
	  boss_StaggerBlock -= 8;
	  
	if(boss_StaggerBlock <= 0)
	  staggerTrigger = 1;
  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
    if(god_mode)
	  return 0;
	  
    if(mod == "Void" && !ssg_res)
	{
	  ssg_res = true;
	  
	  if(!phase3)
	    players[consoleplayer].mo.ACS_NamedExecute("Boss_Resist", 0);
		
	  if(!boss_staggered)
	  {
		ssg_hit = true;
	    vel = (0,0,0);
	  }
	  else
	  {
	    ssg_debuff = true;
		staggerTime = 2;
	    if(source && source is "DFPlayer")
	      source.GiveBody(source.GetMaxHealth(True));
	  }
	}
	
    if(parry_time > 0 && target && Distance2D(target) <= 777 && (mod == "ShotGib" || mod == "SGShock"))
	{
	  damage *= 0.1;
	  staggerTrigger = 1;
	  vel = (0,0,0);
	}
	else if(parry_time <= 0 && !boss_staggered)
	{
	  if(players[consoleplayer].mo && DFPlayer(players[consoleplayer].mo).bPhyOrigin)
	  {
	    damage *= 0.1;
		DirectDamage(mod);
	  }
	  else if(mod == "SGShock")
	  {
	    damage *= 0.1;
	    DirectDamage("SGShock");
	  }
	  else
	  {
	    if(source && source is "DFPlayer")
	    {
		  damage *= 0.1;
	      GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	      if(mod != "None" && ev)
	        ev.SetShieldIcon();
		}
	  }
	}

	if(boss_staggered)
	{
	  true_hit = true;
	  staggerTime = 2;
	  if(source && source is "DFPlayer")
	    source.GiveBody(source.GetMaxHealth(True));
	}
	   
    return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);
  }
  
  void AdjustFlashPos(int xofs, int yofs, int zofs)
  {
    if(flash)
	{
	  N_CounterFlash(flash).xw = xofs;
	  N_CounterFlash(flash).yw = yofs;
	  N_CounterFlash(flash).zw = zofs;
	}
  }

  void SetParryLight(int xofs, int yofs, int zofs, int beginDie, bool shortCounter)
  {
    double c = cos(angle);
	double s = sin(angle);
	Vector3 spawnPos = Vec3Offset(xofs * c + yofs * s, xofs * s - yofs * c, zofs);
	flash = Spawn("N_CounterFlash", spawnPos);
	if(flash)
	{
	  flash.master = self;
	  N_CounterFlash(flash).xw = xofs;
	  N_CounterFlash(flash).yw = yofs;
	  N_CounterFlash(flash).zw = zofs;
	  N_CounterFlash(flash).duration = beginDie;
	}
  }
  
  action void SpawnTrail(Vector3 spos, double alpha)
  {
    let actr = Spawn("AntibodyTrail", self.pos + spos);
	if(actr)
	{
	  actr.angle = self.angle;
	  actr.frame = self.frame;
	  actr.sprite = self.sprite;
	  actr.scale = self.scale;
	  actr.alpha = alpha;
	}
  }
  
  void SpawnColorTrail(string class_name, vector3 spos, string col, double alpha, int r_style, bool bright, double velx, double vely, double velz)
  {
	let actr = Spawn(class_name, spos);
	if(actr)
	{
	  double ang = self.angle;
	  actr.angle = ang;
	  actr.vel = (velx * cos(ang) + vely * sin(ang), velx * sin(ang) - vely * cos(ang), velz);
	  actr.scale = self.scale;
	  actr.sprite = self.sprite;
	  actr.frame = self.frame;
	  actr.bBright = bright;
	  if(phase3 || (phase1 && !phase2))
	  {
	    actr.A_SetRenderStyle(alpha, r_style);
		actr.SetShade(col);
	  }
	}
  }

Class StarPowerup : Inventory

  override void AttachToOwner(Actor other)

	if(other)

	  other.ACS_NamedExecute("Duel_PowerTip");
	  let wpn = Trenchgun(other.FindInventory("Trenchgun"));
	  if(wpn)
	  {
		wpn.AddCharge();
		
		let wep = players[consoleplayer].ReadyWeapon;
		let psp = players[consoleplayer].FindPSprite(PSP_Weapon);
		if(wep && wep is "TrenchGun" && psp && psp.CurState.InStateSequence(wep.ResolveState("Idle")))
		  players[consoleplayer].SetPSprite(3, wep.ResolveState("Charge"));

	  other.A_StartSound("Psi_Up3", CHAN_AUTO, volume: 0.6);

	  {
		ev.UpdateChat(String.Format("· %s \cJconsumes \cFRaw Psi.", other.GetTag()));
		ev.UpdateChat("· \cITrench Gun Empowered!");
	  }
	}
  }
  
  override void DoEffect()
  {
	Super.DoEffect();
	
	if(GetAge() > 35)
	  Destroy();
  }
