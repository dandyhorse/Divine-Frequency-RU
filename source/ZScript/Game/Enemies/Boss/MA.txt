  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
    if(!phase2)
	{
	  if(parry_time > 0 && target && Distance2D(target) <= 777 && (mod == "ShotGib" || mod == "SGShock"))
	  {
	    damage *= 0.1;
	    staggerTrigger = 1;
	    vel = (0,0,0);
	  }
	  else if(parry_time <= 0 && !boss_staggered)
	  {
	    if(players[consoleplayer].mo && DFPlayer(players[consoleplayer].mo).bPhyOrigin)
	    {
		  damage *= 0.1;
		  DirectDamage(mod);
	    }
	    else
	    {
	      if(source && source is "DFPlayer")
	      {
		    damage *= 0.1;
	        GenericScreenStuff ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	        if(mod != "None" && ev)
	          ev.SetShieldIcon();
		  }
	    }
	  }
	
	  if(boss_staggered)
	  {
	    if(mod == "Gauntlet")
	      damage *= 2.3;
		else if(mod == "Explode")
		  damage *= 6.9; //lol
		else if(mod == "Trauma")
		  damage *= 46.0;
	    else
	      damage *= 6.3;
		  
	    staggerTime = 2;
	  
	    if(source && source is "DFPlayer")
	      source.GiveBody(source.GetMaxHealth(True));
	  }
	}
	else //Phase 2 type shit
	{
	  if(mod == "Pen")
	  {
		damage *= 2.0;
		if(damage > 360)
		{
		  damage *= 1.5;
		  staggerThres = 400;
		  
		  if(source && source is "DFPlayer")
	        DFPlayer(source).DoHeal(50);
			
		  if(damage >= 700) //Triggered by a riposte
		    damage = 300;
		}
		damage = clamp(damage, 0, 600);
	  }
	  
	  if(mod == "Explode")
	    damage *= 0.48;
		
	  damage *= 1.9;
	  staggerThres += ApplyDamageFactor(mod, damage);
	  if(staggerThres >= 400)
	  {
	    staggerTrigger = 1;
	    vel = (0,0,0);
		staggerThres = 0;
	  }
	}
	
    return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);
  }
  
  void DirectDamage(string dam_type)
  {
    if(dam_type == "Trauma")
	  boss_StaggerBlock -= 12;
	else if(dam_type == "Explode")
	  boss_StaggerBlock -= 34;
	else if(dam_type == "ShotGib")
	  boss_StaggerBlock -= 3;
	  
	if(boss_StaggerBlock <= 0 && !CheckInventory("BlockStackRefill", 1))
	  staggerTrigger = 1;
  }
