}

Class MeatChunk1 : Actor
{
  int rlAmt;
  bool rollUp;
  
  override void BeginPlay()
  {
    roll = random(0,359);
	rlAmt = random(-8,8);
	bSPRITEFLIP = random(0,1);
	
    let actr = Spawn("MeatBlood", pos);
	if(actr)
	{
	  actr.master = self;
	  actr.A_SetScale(0.6);
	}
  }

  Default
  {
	Gravity 0.7;
    +MISSILE
	+ROLLSPRITE
	+ROLLCENTER
  }
  
  States
  {
    Spawn:
	  MTCH A 0;
	  Goto Idle;
	Idle:
	  MTCH "#" 1 A_SetRoll(roll + rlAmt);
	  Wait;
	  MTCH "#" 2
	  {
	    if(rollUp)
		  if(roll < 6)
			roll++;
		  else
			rollUp = false;
		
		if(!rollUp)
		  if(roll > -6)
			roll--;
		  else
			rollUp = true;
	  }
	  Wait;
	Death:
	  MTCH "#" 0
	  {
		roll = 0;
		
		return A_Jump(127, "BounceB");
	  }
	BounceA:
	  MTCH "#" 0 A_SetScale(scale.x * 0.7, scale.x * 1.4);
	  MTCH "####" 1
	  {
		scale.y -= 0.1;
		scale.x += 0.05;
	  }
	  MTCH "##" 1 { scale.x += 0.05; }
	  MTCH "#" 350;
	  MTCH "#" 1 A_FadeOut(0.05);
	  Wait;
	  Stop;
	BounceB:
	  MTCH "#" 0 A_SetScale(scale.x * 1.4, scale.x * 0.7);
	  MTCH "####" 1
	  {
		scale.x -= 0.1;
		scale.y += 0.05;
	  }
	  MTCH "##" 1 { scale.y += 0.05; }
	  MTCH "#" 350;
	  MTCH "#" 1 A_FadeOut(0.05);
	  Stop;
  }
}
