  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)
  {
    A_RemoveLight("CultistLight");
    A_KillChildren("Fire", KILS_KILLMISSILES);
	
	let pmo = players[consoleplayer].mo;
	if(pmo && pmo.CheckInventory("CultistRoll", 1))
	{
	  pmo.A_TakeInventory("CultistRoll", 1);
	  pmo.roll = 0.0;
	  let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	  if(ev)
	    ev.SetParry(0.0, 0.0, 0.02, 30); 
	}
	
	if(inflictor)
      killerAng = inflictor.angle;
	
	if(MeansOfDeath == "Poison")
	  Poisoned = true;
	  
	if(user_itemDrop != "")
	  A_DropItem(user_itemDrop);
	
	super.Die(source, inflictor, dmgFlags, MeansOfDeath);

  action void CultistFire(Vector3 SpawnPos)
  {
    if(target)
	{
	  int aimFlag = 2;
	  if(Proxy(self).preCrouch)
	    aimFlag = 0;
		
      for(int i = 0; i < 16; i++)
	  {
	    Actor msl = A_SpawnProjectile("CT_Pellet", SpawnPos.z, SpawnPos.y, flags: aimFlag);
		if(msl)
		{
		  if(Proxy(self).aimed && GetCrouchFactor() > 0.5)

		  msl.vel += (frandom(-2.0,2.0), frandom(-2.0,2.0), frandom(-0.5,2.0));
		}
	  }
	}

Class CultistBrute : Proxy
{
  Actor sig;
  int mag, fireWave_cd;
  
  void SlamFire()
  {
    for(int i = 0; i < 16; i++)
	{
	  Actor msl = A_SpawnProjectile("CT_Bullet", 97, 16, flags: CMF_AIMDIRECTION);
	  if(msl)
	  {
	    msl.VelIntercept(target, -1);
		msl.A_ChangeVelocity(0, frandom(-10.0,10.0), frandom(-0.5,4.0), CVF_RELATIVE);
	  }
	}
  }
  
  override void BeginPlay()
  {
    mag = 5;
	Armored = true;
    user_noSeeSound = true;
	A_SpawnItemEx("CultistBruteHead", zofs: 104, flags: SXF_SETMASTER);
  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  { 
	 if(mod == "Trauma" || mod == "ShotGib" || mod == "SGShock" || mod == "Pen")
	 {
	   damage = 1;
	   SetStateLabel("Block");
	 }
	 
     return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);        
  }
  
  Default
  {
    Health 400;
    XScale 1.0;
	YScale 0.83;
	+DONTTHRUST
	+NOBLOOD
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(fireWave_cd > 0)
	  fireWave_cd--;
  }
  
  States
  {
    Block:
	  CLBL A 1;
	  CLBL A 1
	  {
	    A_FaceTarget();
		if(!sig)
		{
		  Vector3 sPos = Vec3Offset(cos(angle) * 72 + sin(angle) * -24, sin(angle) * 72 + cos(angle) * -24, 104);
		  sig = Spawn("CultistSigil", sPos);
		  if(sig)
		    sig.master = self;
		}
		else
		{
		  CultistSigil(sig).time = 0; //Reset the death time
		  sig.Alpha = 1.0;
		}
	  }
	  CLBL AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 1 A_FaceTarget();
	  CLBL BBBCCCDDDEEE 1 A_FaceTarget();
	  Goto Chase;
    Missile:
	  TNT1 A 0
	  {
	    if(target && Distance2D(target) < 350 && fireWave_cd <= 0)
		  return ResolveState("FlameWave");
		  
		return ResolveState(null);
	  }
    Slamfire:
	  CLHF Q 2
	  {
	    A_FaceTarget();
		A_StartSound("Cul_Aim2", CHAN_AUTO, volume: 0.6, attenuation: 0.65);
	  }
	  CLHF RS 3 A_FaceTarget();
	  CLHF AAAA 4 A_FaceTarget();
	SlamfireLoop:
	  CLFL A 4 A_FaceTarget();
	  CLFL A 1 Bright A_FaceTarget();
	  CLFL B 1 Bright 
	  {
		A_FaceTarget();
	    A_StartSound("Cul_Fir4", 10, volume: 0.6, attenuation: 0.5);
		A_SpawnItemEx("CT_Casing", 39, 6, 102, random(-1,0),random(8,12),random(8,10));
		SlamFire();
		mag--;
	  }
	  CLHF CD 1;
	  CLHF CB 2;
	  TNT1 A 0
	  {
	    if(mag <= 0)
		  return ResolveState("SlamfireDone");
		  
		return ResolveState(null);
	  }
	  Loop;
	SlamfireDone:
	  TNT1 A 0 { mag = 5; }
	  CLHF HI 5;
	  CLHF JK 4;
	  CLHF L 2; 
	  CLHF L 1 A_StartSound("CulPump3", CHAN_AUTO, volume: 0.4, attenuation: 0.74);
	  CLHF M 1;
	  CLHF N 2 A_StartSound("CulPump4", CHAN_AUTO, volume: 0.4, attenuation: 0.74);
	  CLHF O 1;
	  CLHF P 4;
	  CLHF QRSS 3;
	  Goto Chase;
    FlameWave:
	  TNT1 A 0
	  {
	    A_FaceTarget();
	    A_StartSound("Cultist/ReadyFireBall", CHAN_AUTO, attenuation: 0.5);
		fireWave_cd = 35*10;
	  }
	  CLTH D 1 CultistPreBall((0, -62, 96));
	  CLTH D 1 CultistPreBall((0, -63, 96));
	  CLTH D 1 CultistPreBall((0, -64, 96));
	  CLTH E 1 CultistPreBall((0, -65, 96));
	  CLTH E 1 CultistPreBall((0, -66, 96));
	  CLTH E 1 CultistPreBall((0, -67, 97));
	  CLTH F 1 CultistPreBall((0, -67, 98));
	  CLTH F 1 CultistPreBall((0, -67, 99));
	  CLTH F 1 CultistPreBall((0, -67, 100));
	  CLTH G 1 CultistPreBall((0, -68, 101));
	  CLTH G 1 CultistPreBall((0, -68, 102));
	  CLTH G 1 CultistPreBall((0, -68, 102));
	  TNT1 A 0
	  {
	    A_FaceTarget();
		A_SpawnItemEx("CultistFireShooter", flags: SXF_NOCHECKPOSITION);
		let actr = Spawn("SoundDummy", pos + (0,0,100));
		if(actr)
		{
		  actr.A_StartSound("Cul_FB3", CHAN_AUTO, volume: 0.4, attenuation: 0.6);
		  actr.vel = (cos(angle) * 6.5, sin(angle) * 6.5, 0.0);

  {
    Super.DoEffect();
	
	if(owner)
	  owner.roll += 0.5;
  }
  
  Default
  {
    Inventory.Amount 1;
    Inventory.Maxamount 1;
  }
}

Class CultistSigil : Actor
{
  int time;
  override void Tick()
  {
    Super.Tick();
	
	if(time > 35)
	  A_FadeOut(0.1);
	  
	if(master)
	  A_Warp(AAPTR_MASTER, 72, -24, 104, flags: WARPF_NOCHECKPOSITION);
	  
	time++;
  }
  
  Default
  {
    RenderStyle "Translucent";
	Alpha 0.7;
	Scale 0.35;
	+BRIGHT
	+NOINTERACTION
    +WALLSPRITE
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  BSPR A 1 { roll += 1; }
	  Wait;
  }
}
