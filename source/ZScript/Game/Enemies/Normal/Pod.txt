Class GreenPod : DF_Monster
{
  uint tic;
  int user_range;
  double spawnRate;
  
  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)
  {
    A_KillChildren("Fire", KILS_KILLMISSILES);
	A_GiveExp(400);
	
	super.Die(source, inflictor, dmgFlags, MeansOfDeath);
  }
  
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	
	user_range = 256;
	spawnRate = user_range * 0.03125;
	
	Armored = true;
	unScannable = true;
	SetArmor(100);
  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
    int newDamage = damage;
	
	if(AP > 0)
	{
	  let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
	  {
	    if(mod == "Gauntlet" || mod == "HeavyCombo")
	    {
	      AP -= 50;
          newDamage = Max(0, damage * 1.5);
		  ev.SetHitMark("CritMark", 1.2);
	    }
	    else if(mod == "Trauma")
	    {
	      AP -= 20;
	      newDamage = Max(0, damage * 0.1);  //.1
		  ev.SetShieldIcon();
	    }
	    else if(mod == "PistolAP")
	    {
	      AP -= 100;
	      newDamage = Max(0, damage * 0.1);
		  ev.SetHitMark("CritMark", 1.2);
	    }
		else if(mod == "ShotGib")
	    {
		  let pmo = players[consoleplayer].mo;
		  if(pmo && !DFPlayer(pmo).bOrangePower)
		  {
	        AP -= 8;
	        newDamage = Max(0, damage * 0.1);
		    ev.SetShieldIcon();
		  }
	    }
	    else if(mod == "Explode")
	    {
	      AP -= 100;
	      newDamage = Max(0, damage * 0.6);
		  ev.SetHitMark("CritMark", 1.2);
	    }
		else if(mod == "Pen")
	    {
	      AP -= 50;
	      newDamage = Max(0, damage);
		  ev.SetHitMark("CritMark", 1.2);
	    }
		else if(mod == "Sting")
	    {
	      AP -= 5;
	      newDamage = Max(0, damage);
	    }
  
	    ev.SetArmorBreak(AP);
	  }
	  
	  if(armored && AP <= 0)
	  {
	    armored = false;
	    A_DamageSelf(2000, mod);
	    ev.shieldBarAlpha = 0.0;
	    ev.shieldVal = 0;
	  }
	}
	
	return super.DamageMobj(inflictor, source, newDamage, mod, flags, angle); 
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(health <= 0)
	  return;
	
	if(!bDormant && GetAge() % 4 == 0)
	{
	  for(int i = 0; i < spawnRate; i++)
	  {
	    int pitch = random(0,90);
		int angle = random(0,359);
		Vector3 inPos = random(0, user_range) * (cos(pitch) * cos(angle), cos(pitch) * sin(angle), sin(pitch));
		Vector3 outPos = user_range * (cos(pitch) * cos(angle), cos(pitch) * sin(angle), sin(pitch));
		
		
		A_SpawnParticleEx("", TexMan.CheckForTexture(String.Format("SPR%dA0", random(1,2))), STYLE_Add, 
		SPF_FULLBRIGHT|SPF_NEGATIVE_FADESTEP|SPF_ROLL|SPF_ROLLCENTER|SPF_FADE_IN_HOLD_OUT, random(48,94), random(12,40), 0,
		outPos.x, outPos.y, outPos.z, velz: frandom(0.0,0.2),
		startalphaf: 0.0, fadestepf: -0.2, startroll: random(0,360), fadeoutstepf: 0.07);
		
		A_SpawnParticleEx("", TexMan.CheckForTexture(String.Format("SPR%dA0", random(1,2))), STYLE_Add, 
		SPF_FULLBRIGHT|SPF_NEGATIVE_FADESTEP|SPF_ROLL|SPF_ROLLCENTER|SPF_FADE_IN_HOLD_OUT, random(48,94), random(12,40), 0,
		inPos.x, inPos.y, inPos.z, velz: frandom(0.0,0.2),
		startalphaf: 0.0, fadestepf: -0.2, startroll: random(0,360), fadeoutstepf: 0.07);
	  }
	}
	
    if(!bDormant && tic % 17 == 0)
	  if(players[consoleplayer].mo && Distance2D(players[consoleplayer].mo) <= user_range && !DFPlayer(players[consoleplayer].mo).bResistToxin)
	    DFPlayer(players[consoleplayer].mo).SetToxin(1);
	  
	if(tic == 17)
	  tic = 0;
	else
	  tic++;
  }
  
  Default
  {
    //$Category "Enemies/Harmless"
    Health 200;
    Radius 42;
    Height 105;
	YScale 1.41;
	XScale 1.7;
	PainChance "Stun", 256;
	DamageFactor 0.1;
	DamageFactor "Poison", 0.0;
	DamageFactor "Sting", 10.0;
	FloatBobStrength 0.08;
	Species "Rot";
	Monster;
	+DONTFALL
	+DONTTHRUST
	+FLOATBOB
	+NOBLOOD
	+NOGRAVITY
	+DF_Monster.PoisonResist
	+DF_Monster.PossessImmune
  }
  
  States
  {
    Spawn:
	  GPOD ABCDEFGHIJKLMN 3;
	See:
	  GPOD OPQRSTU 6;
	  Loop;
	Pain.Shock:
	Pain.Stun:
	Stun:
	  "####" "#" 1 A_SetHealth(1);
	  "####" "#" 0 A_DamageSelf(10, "Stun", DMSS_NOFACTOR);
	Death:
	  TNT1 A 0
	  {
	    A_StartSound("Pod_Die", CHAN_AUTO);
		A_UnsetSolid();
	  }
	  GPOD NMLKJIHGFEDCB 3;
	  TNT1 A 1 A_SpawnItemEx("DeadPod");
	  Stop;
  }
}
