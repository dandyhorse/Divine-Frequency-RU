  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
    if(inflictor && (inflictor is "PistolPuff" || inflictor is "ARPuff") && (inflictor.pos.z - pos.z >= 104))

	  damage *= 0.1;

	  inflictor.Destroy();
	}
    return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);        
  }

Class Cultist_Handgun : Cultist
{
  int ammo_token;
  override void BeginPlay()
  {
    ammo_token = 6;
	Super.BeginPlay();
  }
  void CultistFireHandgun()
  {
    Actor msl = A_SpawnProjectile("CT_Bullet", 118, 22, flags: CMF_AIMDIRECTION);
	if(msl)
	{
	  msl.VelIntercept(target, -1);
		//msl.A_ChangeVelocity(0, frandom(-10.0,10.0), frandom(-0.5,4.0), CVF_RELATIVE);
	}
  }
  States
  {
    Spawn:
	  CPF1 A 10 A_Look();
	  Wait;
    See:
	  CPW1 AAACCCDDDEEEFFFHHHJJJKKKLLLMMMOOOPPP 1 A_Chase();
      Loop;
	Melee:
	  TNT1 A 0
	  {
	    if(target && !(target is "DFPlayer"))
		  return ResolveState("Missile");
		  
		return ResolveState(null);
	  }
	  CPML DEF 2 A_FaceTarget();
	  TNT1 A 0 A_StartSound("Fist_Sw1", CHAN_AUTO);
	  CPML GHIJK 1 A_FaceTarget();
	  CPML K 1
	  {
	    A_CustomMeleeAttack(4, "Cultist/Melee", "", "Physical");
		
		if(target && CheckMeleeRange())
		{
		  target.Thrust(5, angle - 90);
		  target.angle += 35;
		}
	  }
	  CPML LMNOP 2 A_FaceTarget();
	  CPF1 FFF 2 A_FaceTarget();
	  Goto Fire;
	Missile:
	  CPF1 ABCDEF 3 A_FaceTarget();
	Fire:
      CPF2 A 1 Bright Light("MineExplodeLight1")
	  {
	    ammo_token--;
	    A_FaceTarget();
	    A_StartSound("Cul_HG1", CHAN_AUTO, volume: 0.6, attenuation: 0.5);
		A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("EXPFA0"), STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE, 4, 100, 0,
	    93, 29, 118, startalphaf:  0.5, sizestep: 20);
		CultistFireHandgun();
	  }
	  CPF2 A 1 Bright Light("MineExplodeLight2") A_FaceTarget();
	  CPF2 B 1 Bright Light("MineExplodeLight3") A_FaceTarget();
	  CPF2 BCD 1 A_FaceTarget();
	  CPF2 E 4 A_FaceTarget();
	  CPF2 F 5 A_FaceTarget();
	  CPF2 G 5 A_FaceTarget();
	  CPF2 H 4 A_FaceTarget();
	  CPF2 IJKL 3 A_FaceTarget();
	  CPF2 MNOP 3 A_FaceTarget();
	  TNT1 A 0
	  {
	    if(target && CheckSight(target))
		  if(Distance2D(target) > 500)
		  {
		    if(ammo_token <= 3)
			  ammo_token += 2;
			  
		    return ResolveState("WalkFire");
		  }
		  else if(ammo_token > 0)
		    return ResolveState("Fire");
		  
		return ResolveState(null);
	  }
	  TNT1 A 0 { ammo_token = 6; }
	  CPF1 FED 3;
	  CPF1 CB 2;
	  CPF1 A 1;
	  Goto See;
	WalkAim:
	  TNT1 A 0
	  {
	    if(target && !CheckSight(target))
		  return ResolveState("See");
		  
		return ResolveState(null);
	  }
	  CPW2 AAACCCDDDEEEFFFHHHJJJKKKLLLMMMOOOPPP 1
	  {
	    A_FaceTarget();
		Thrust(0.2, angle);
	  }
	  TNT1 A 0
	  {
	    if(target && CheckSight(target))
		  return A_Jump(208, "WalkFire");
		  
		return ResolveState(null);
	  }
	  Loop;
	WalkFire:
      CPW3 A 1 Bright Light("MineExplodeLight1")
	  {
	    ammo_token--;
	    A_StartSound("Cul_HG1", CHAN_AUTO, volume: 0.6, attenuation: 0.5);
		A_SpawnParticleEx("FFFFFF", TexMan.CheckForTexture("EXPFA0"), STYLE_Add, SPF_FULLBRIGHT|SPF_RELATIVE, 4, 100, 0,
	    93, 29, 118, startalphaf:  0.5, sizestep: 20);
		CultistFireHandgun();
		A_FaceTarget();
		Thrust(0.2, angle);
	  }
	  CPW3 A 1 Bright Light("MineExplodeLight2")
	  {
	    A_FaceTarget();
		Thrust(0.2, angle);
	  }
	  CPW3 A 1 Bright Light("MineExplodeLight3")
	  {
	    A_FaceTarget();
		Thrust(0.2, angle);
	  }
	  CPW3 BBB 1 Bright
	  {
	    A_FaceTarget();
		Thrust(0.2, angle);
	  }
	  CPW3 CCCDDDEEEFFFHHHJJJKKKLLLMMMOOOPPP 1
	  {
	    A_FaceTarget();
		Thrust(0.2, angle);
	  }
	  TNT1 A 0
	  {
	    if(ammo_token <= 0)
		{
		  ammo_token = 6;
		  return ResolveState("See");
		}
	    else if(target && CheckSight(target))
		  return A_Jump(127, "WalkFire");
		  
		return ResolveState(null);
	  }
	  Goto WalkAim;
	Pain.Daze:
	  CPF1 A 140 A_DazeEffect();
	DazeEnd:
	  TNT1 A 0 A_EndDaze();
	  Goto See;
	Stun:
	  CPF1 "#" random(1,2)
	  {
	    frame = random(0,1);
		if(!bStunSoft)
	      A_DamageSelf(2, "Shock");
		
	    A_SpriteOffset(frandom(-3.0,3.0), frandom(0.0, -2.0));
		
		if(StunTime <= 0)
		  return ResolveState("StunFinish");
		else
		  StunTime--;
		  
		return ResolveState(null);
	  }
	  Wait;
  }
}
