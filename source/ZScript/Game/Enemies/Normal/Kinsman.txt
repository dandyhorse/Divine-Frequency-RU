  bool user_yell, user_helm, yelling;
  int deathTic, helmHP;
  
  void SpawnGore(String className, int xofs, int yofs, int zofs, double xvel, double yvel, double zvel)
  {
    double ang = random(0,359);
	double cos = cos(angle);
	double sin = sin(angle);
    Vector3 SpawnPos = Vec3Offset(xofs*cos + yofs*sin, xofs*sin - yofs*cos, zofs);
	let actr = Spawn(className, spawnPos);
	if(actr)
	{
	  actr.vel = (xvel, yvel, zvel);
	  actr.scale = (1.5, 1.5);
	}
  }
  
  override void PostBeginPlay()
  {
    deathTic = 4;
	radar_height = Height * 0.5;
	//user_helm = true;
	if(user_helm)
	{
	  SeeSound = "KinHelm/See";
	  ActiveSound = "KinHelm/Active";
	  helmHP = 3;
	}
	
	Super.PostBeginPlay();
  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
     if(inflictor && inflictor is "PistolPuff" && (inflictor.pos.z - pos.z >= 95))
	 {
	   if(user_helm)
	   {
	     helmHP--;
	     damage *= 0.2;
		 inflictor.A_SpawnItemEx("BulletSpark");
		 
		 if(helmHP == 0)
		 {
		   A_DamageSelf(192, "Trauma");
		   SetStateLabel("HelmBreak");
		 }
	   }
	   else
	   {
	     let pmo = DFPlayer(players[consoleplayer].mo);
	     if(pmo && DFPlayer(pmo).bBluePower)
	       damage *= 1.5;
	     else
	       damage *= 1.25;
	   }
	 }
     return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);        
   }
  
  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)
  {
    super.Die(source, inflictor, dmgFlags, MeansOfDeath);
	
    A_KillChildren("Fire", KILS_KILLMISSILES);
	A_GiveExp(250);
	bTHRUACTORS = true;
	yelling = false;
	
	if(bPossessed)
	{
	  SetStateLabel("Explode");
	  A_SpawnItemEx("Swarm_Explosion", zofs: (sHeight * scale.y) * 0.5);
	}
	else if(MeansOfDeath == "Poison")
	  Poisoned = true;
  }
