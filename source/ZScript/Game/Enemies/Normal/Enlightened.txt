    super.Die(source, inflictor, dmgFlags, MeansOfDeath);

    if(source)
      killerAng = source.angle;
	
	if(bPossessed)
	{
	  SetStateLabel("ShotgunTorso");
	  A_SpawnItemEx("Swarm_Explosion", zofs: (sHeight * scale.y) * 0.5);

	else if(inflictor && inflictor is "PistolPuff" && (inflictor.pos.z - pos.z >= 90))
	  SetStateLabel("HandgunBehead");
	else if(inflictor && (inflictor is "ShotgunPuff" || inflictor is "ShockPuff"))
	  if(source && Distance2D(source) <= 156)
	    SetStateLabel("CloseShotGib");
	  else if(source && Distance2D(source) <= 300)
	    SetStateLabel("MediumShotGib");
	  else if(source && Distance2D(source) > 300)
	    SetStateLabel("Death"); 
	 
	if(MeansOfDeath == "Poison")
	  Poisoned = true;
	
	if(dType == 4)
	  A_GiveExp(250);
	else
	  A_GiveExp(150);
	  
	A_KillChildren("Fire", KILS_KILLMISSILES);
  }

	   if(pmo && DFPlayer(pmo).bBluePower)
	     damage *= 2.0;
	   else

	 }

     return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);        
  }

  }
  
  States
  {
    Spawn:
      ZM2I A 10 A_Look; 
	  Wait;
    See:
	  ZM2C AAABBBCCCDDDEEEFFFGGGHHH 1 A_Chase();
	  Loop;
	Melee:
	  ZM2A A 3;
	  ZM2A B 1 A_FaceTarget();
	  ZM2A B 1 A_StartSound("PsychoPipe/Swing", 99, volume: 0.7, attenuation: 0.7);
	  ZM2A BBC 1 A_FaceTarget();

	  ZM2A CD 2 A_FaceTarget();
	  ZM2A E 2
	  {
		A_FaceTarget();
	    A_CustomMeleeAttack(random(11,15), "DSEMPTY", damagetype: "Physical");
		
		if (target && CheckMeleeRange())
		{
		  A_StopSound(99);
		  A_StartSound("PsychoPipe/Impact", CHAN_AUTO, CHANF_OVERLAP, volume: 0.9, attenuation: 0.5);
		  target.A_StartSound("PP_Burn", CHAN_AUTO, volume: 0.7);
		  
		  if(target is "DFPlayer")
		  {
		    A_SetTics(8);
		    target.A_QuakeEx(2, 2, 2, 8, 0, 128);
			if(!DFPlayer(target).isDodging && !DFPlayer(target).bResistToxin)
			  DFPlayer(target).SetToxin(random(3,8));
		  }
		}
	  }
	  ZM2A FFGGHHI 2;
	  Goto See;
	Pain:
      ZM2P A 4;
	  Goto See;
	Pain.Daze:
	  ZM2I A 140 A_DazeEffect();
	DazeEnd:
	  TNT1 A 0 A_EndDaze();
	  Goto See;
	StunFinish:
	  TNT1 A 0
	  {
	    A_ClearStun();
		A_SpriteOffset(0.0, 0.0);
	  }
	  Goto See;
  }
