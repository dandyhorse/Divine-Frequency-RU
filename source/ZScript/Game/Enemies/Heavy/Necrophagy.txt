Class Tick : DF_Monster //6005 SP"DIERMAN
{
  bool frenzy;
  int fCD;
  
  override void Tick()
  {
    Super.Tick();
	
	if(!frenzy && target && fCD > 0)
	  fCD--;
  }
  
  override void BeginPlay()
  {
    fCD = 35*8;
	radar_height = Height * 0.4;
  }
  
  override void Die(Actor source, Actor inflictor)
  {
	super.Die(source, inflictor);
	
	A_KillChildren("Fire", KILS_KILLMISSILES);
	A_GiveExp(750);
	
	if(bPossessed)
	{
	  SetStateLabel("Death.Void");
	  A_SpawnItemEx("Swarm_Explosion", zofs: (sHeight * scale.y) * 0.5);
	}
  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
    if(self && InStateSequence(CurState, ResolveState("See")))
	  SetState(ResolveState("GetAssOffGround"));
	 
	fCD = 0;
	 
    return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);        
  }
  
  Default
  {
    //$Category "Enemies/Heavy"
	Health 1000;
	Speed 4;
	Radius 66;//Radius 75;
	Height 196;//Height 217;
	MeleeThreshold 104;
	MeleeRange 176;
	Mass 2500;
	Scale 1.5; //Scale 1.7;
	PainChance 0;
	PainChance "Stun", 255;
	Tag "Necrophagy";
	DamageFactor "Trauma", 2.0;
	DamageFactor "Explode", 1.0;
	DamageFactor "Poison", 0.5;
	+DONTTHRUST
    +FLOORCLIP
	+DROPOFF
	+LOOKALLAROUND
	+DF_Monster.Large;
	Monster;
	//SeeSound "Necrophagy/Sight";
	ActiveSound "Necrophagy/Active";
	DeathSound "Necrophagy/Death";
  }
  States
  {
	Spawn:
	  TIKI A 10
	  {
	    A_Look();
		
	    if(frenzy)
		  return ResolveState("IdleFrenzy");
		  
	    return ResolveState(null);
	  }
	  Wait;
	IdleFrenzy:
	  TIAT A 10 A_Look();
	  Wait;
	See:
	  TIKI A 4
	  {
	    BlockThingsIterator it = BlockThingsIterator.Create(self, 512);
		while (it.Next())
		  if(it.thing && it.thing is "DFPlayer" && it.thing.health > 0 && Distance3DSquared(it.thing) <= 262144)
		    if(CheckSight(it.thing, SF_IGNOREVISIBILITY|SF_SEEPASTSHOOTABLELINES|SF_IGNOREWATERBOUNDARY))
		      return ResolveState("GetAssOffGround");
			
		return ResolveState(null);
	  }
	  Wait;
	Chill:
	  TIKI A -1;
	  Stop;
	GetAssOffGround:
	  TNT1 A 0 A_StartSound("Necrophagy/Sight", CHAN_AUTO, attenuation: 0.8);
	  TIKI BCDEFFF 2;
	  TIKI GHIJK 3;
	  TIKI K 2;
	  Goto Stalk;
    Stalk:
	  TICK AAABBBCCCDDDEEEFFFGGGHHH 1
	  {
	    A_Chase(null, "Missile");
		if(fCD <= 0)
	    {
	      frenzy = true;
	      SetStateLabel("ChaseStart");
	    }
	  }
      Loop;
	ChaseStart:
	  TNT1 A 0
	  {
		frenzy = true;
		A_StartSound("Necrophagy/Chase", CHAN_AUTO, attenuation: 0.7);
	  }
	  TISP AAABB 1 A_Chase("Melee", null);
	  TISP BCCCDDDEEEFFFGGGHHH 1
	  {
	    A_FaceTarget();
	    A_Chase(null, null);
	  }
	  TNT1 A 0 A_SetSpeed(30);
	Chase:
	  TNT1 A 0 A_SetSpeed(30);
	  TISW AABBCCDDEEFFGGHH 1 A_Chase("Melee", null);
	  Goto Chase+1;
	Melee:
	  TNT1 A 0 A_StartSound("TK_Bite", CHAN_AUTO);
	  TIAT BC 3 A_FaceTarget();
	  TNT1 A 0 Thrust(40, angle);
	  TIAT D 2 A_FaceTarget();
	  TIAT EF 2 A_FaceTarget();
	  TIAT G 1
	  {
	    A_FaceTarget();
		//A_StartSound("Hybrid/Attack", CHAN_AUTO);
		if(target && target is "DFPlayer" && CheckMeleeRange())
		{
		  A_CustomMeleeAttack(random(30,40), "TK_Hit", "DSEMPTY", "Physical");
		  let pmo = players[consoleplayer].mo;
		  if(pmo && !DFPlayer(pmo).bAdvanced_Str && !DFPlayer(pmo).bRecoilPower)
		    SetDrunk(1.2, 0.006, true);
			
		  if(pmo && !DFPlayer(pmo).isDodging && !DFPlayer(pmo).bResistToxin)
			  DFPlayer(pmo).SetToxin(random(7,10));
		}
		else
		    A_CustomMeleeAttack(100, "TK_Hit", "DSEMPTY", "ShotGib");
	  }
	  TIAT H 1;
	  TIAT IJKLM 3;
	  Goto Chase;
	Stun:
	  "####" "#" 2
	  {
	    A_Pain();
	    A_StunEffect();
	  }
	  TISP DEF random(0,3)
	  {
	    if(!bStunSoft)
	      A_DamageSelf(random(4,5), "Shock");
		
		A_SpawnItemEx("ExplosionSmoke",random(-4,4),random(-16,16),random(32,88),
		frandom(-1,1),frandom(-1,1),frandom(-1,1),random(0,360), failchance: 196);
	    if(StunTime <= 0)
		  return ResolveState("StunFinish");
		else
		  StunTime--;
		  
	    return ResolveState(null);
	  }
	  Goto Stun+1;
    StunFinish:
	  TNT1 A 0 A_ClearStun();
	  Goto Chase;
	Death.Void:
	  TNT1 A 0
	  {
	    A_StartSound("BODSPLT2", CHAN_AUTO, attenuation: 0.5);
		let exp = Spawn("MineExplosion2", pos);
		if(exp)
		  exp.scale = (1.5,1.5);

		A_SpawnParticleEx
		  (
			"FFFFFF",
			TexMan.CheckForTexture("BWAVA0"),
			style: STYLE_ADD,
			flags: SPF_RELATIVE|SPF_ROLL,
			lifetime: 35,
			size: 256,
			angle: 0.,
			xoff: 0,
			yoff: 0,
			zoff: 24,
			velx: 0,
			vely: 0,
			velz: 0,
			accelx: 0,
			accely: 0,
			accelz: 0,
			startalphaf: 1.0,
			fadestepf: 0.15,
			sizestep: 100.0,
			startroll: random(0,359),
			rollvel: 0,
			rollacc: 0
		  );
	  }
	  Goto Death+1;
	Death:
	Death.Shock:
	  "####" "#" 10;
	  TNT1 A 0
	  {
	    A_FaceTarget();
	    A_Explode(50, 300);
		for(int i = 0; i < 12; i++)
		  A_SpawnItemEx("PlagueGas", random(-56,56), random(-56,56), random(57,176), frandom(-1.0,1.0), frandom(-0.0,1.0), frandom(-1.0,1.0), random (0, 360));
		  
		for(int i = 0; i < 23; i++)
		{
		  double angle = random(0,359);
		  double pitch = random(4,90);
		  Vector3 newPos = 48 * (cos(pitch) * cos(angle), cos(pitch) * sin(angle), sin(pitch));
		  Vector3 newVel = 0.25 * newPos;
		  
		  Actor gib = Spawn("TickGib", (pos.x, pos.y, pos.z + random(48,128)) + newPos);
		  if(gib)
		    gib.vel = newVel;
			
		  
		  if(i < 17)
		  {
		    angle = random(0,359);
		    pitch = random(4,90);
		    newPos = 48 * (cos(pitch) * cos(angle), cos(pitch) * sin(angle), sin(pitch));
		    newVel = 0.25 * newPos;
			
			Actor eye = Spawn("TickEye", (pos.x, pos.y, pos.z + random(48,128)) + newPos);
		    if(eye)
		      eye.vel = newVel;
		  }
		}
	  }
	  Goto DeathFinish;
	DeathFinish:
	  TNT1 A 0
	  {
	    A_QuakeEx(3, 3, 3, 30, 0, 1024, "DSEMPTY", QF_SCALEDOWN, falloff: 768);
		A_StartSound("Necrophagy/Death", CHAN_AUTO, attenuation: 0.5);
		A_StartSound("BODSPLT5", CHAN_AUTO, volume: 0.7, attenuation: 0.5);
		A_SpawnItemEx("BloodFountain2", zofs: -32);
		for(int i = 0; i < 2; i++)
		  A_SpawnItemEx("PlagueBall", random(-56,56), random(-56,56), random(57,176), random(-2,2), random(-2,2), random(4,12), random (0, 360));
		  
		for(int i = 0; i < 5; i++)
		  A_SpawnItemEx("PlagueBallMini", random(-56,56), random(-56,56), random(57,176), random(-2,2), random(-2,2), random(4,12), random (0, 360));
	  }
	  TKDT A 5;
	  TNT1 A 0
	  {
		A_StartSound("BODSPLT4", CHAN_AUTO);
		A_NoBlocking();
	    for(int i = 0; i < 5; i++)
		  A_SpawnItemEx("BloodCloud", random(-8,8), random(-16,16), random(64,88), frandom(-1,1), frandom(-1,1), frandom(-1,1), random (0, 359));
	  }
	  TKDT BCD 4;
	  TKDT EFG 4;
	  TKDT H 1 A_SpawnItemEx("DeadTick", flags: SXF_NOCHECKPOSITION);
	  Stop;
  }
}

Class PlagueGas : Actor
{
  int tic, lifetic;
  
  void Gas_Damage()
  {
    BlockThingsIterator it = BlockThingsIterator.Create(self, 88); 
	while(it.Next())
	{
	  if(it.thing && it.thing.health > 0 && Distance3DSquared(it.thing) <= 7744)
	  {
	    if(it.thing == self)
	      continue;
      
	    if(!it.thing.bShootable)
	      continue;
		
	    if(!CheckSight(it.thing, SF_IGNOREVISIBILITY | SF_IGNOREWATERBOUNDARY))
          continue; 
		  
		if(it.thing is "DFPlayer")
		{
		  it.thing.A_StartSound("Plague/Burn", 24, CHANF_NOSTOP);
		}
		it.thing.DamageMobj(self, self, 3, "Poison");
	  }
    }
  }
  
  override void BeginPlay()
  {
    tic = 0;
	lifetic = 0;
	roll = random(0,359);
	bSPRITEFLIP = random(0,1);
	A_SetScale(frandom(0.2,0.3));
  }
  
  override void Tick()
  {
    if(lifetic < 10)
	{
	  A_FadeIn(frandom(0.02,0.04));
	  A_SetScale(Scale.x + frandom(0.025,0.05), Scale.Y + frandom(0.025,0.05));
	}
	
    if(lifetic >= 260)
	  A_FadeOut(0.05);
	
	if(tic >= 12)
	{
	  //A_Explode(3, 88, XF_NOTMISSILE, 0, 88, damagetype: "Poison");
	  Gas_Damage();
	  tic = 0;
	}
	
	if(master)
	  SetOrigin(master.pos, false);
	
	tic++;
	lifetic++;
	
    Super.Tick();
  }
  
  Default
  {
	Alpha 0.0;
	RenderStyle "Translucent";
    +FORCEXYBILLBOARD
	+NODAMAGETHRUST
    +NOINTERACTION
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  PGS1 ABCDEFGHIJKLMNOPQRSTUVWXYZ 1 Bright;
	  PGS2 ABCDEFGHIJKLMNOPQRSTUVWXYZ 1 Bright;
	  PGS3 ABCDEFGHIJKL 1 Bright;
      Loop;
  }
}
