  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)
  {

  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
    int newDamage = damage;
	
	if(AP > 0)
	{
	  let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	  if(ev)
	  {
	    if(mod == "Trauma")
	    {
	      AP -= 5;
	      newDamage = Max(0, damage * 0.1);  //.1
		  ev.SetShieldIcon();
	    }
	    else if(mod == "ShotGib")
	    {
		  let pmo = players[consoleplayer].mo;
		  if(pmo && !DFPlayer(pmo).bOrangePower)
		  {
	        AP -= 2;
	        newDamage = Max(0, damage * 0.1);
		    ev.SetShieldIcon();
		  }
	    }
		else if(mod == "ArmorBreak")
		{
		  AP -= 25;
		}
	    else if(mod == "Explode")
	    {
		  if(!soul)
	        AP -= 30;
		  else
		    AP -= 16;
			
	      newDamage = Max(0, damage * 0.6);
		  ev.SetHitMark("CritMark", 1.2);
	    }
	    ev.SetArmorBreak(AP);
	  }
	  if(armored && AP <= 0)
	  {
	    armored = false;
	    SetStateLabel("ArmorBreak");
	    ev.shieldBarAlpha = 0.0;
	    ev.shieldVal = 0;
	  }
	}
	
	return super.DamageMobj(inflictor, source, newDamage, mod, flags, angle); 
  }
  
  override void PostBeginPlay()
