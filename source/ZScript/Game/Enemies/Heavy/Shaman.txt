  void ShamanMelee(Actor mo, double ThrustAng, double Rotate)
  {
    if(!mo.bDontThrust)
	  mo.Thrust(24, angle + ThrustAng);
	  
	A_QuakeEx(1, 4, 1, 64, 0, 384, "DSEMPTY", QF_SCALEDOWN, falloff: 96);
	A_SoundVolume(10, 0.9);
	if(mo is "DFPlayer")
	{
	  if(mo && !DFPlayer(mo).bAdvanced_Str && !DFPlayer(mo).bRecoilPower)
	  {
		mo.roll = Rotate;
		SetDrunk(0.5, 0.05, 12);
		let item = mo.FindInventory("HeadRoll");
		if(item)
		  HeadRoll(item).amt = 0.5;
		else
		{
		  let item = mo.GiveInventoryType("HeadRoll");
		  if(item)
			HeadRoll(item).amt = 0.5;
		}
	  }
	}
  }
  
  override void BeginPlay()
  {
    tp_charge = random(5,6);
	min_dmg = 40;
	max_dmg = 50;
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(tp_cd > 0)
	  tp_cd--;
  }
  
  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle) 
  {
     if(inflictor && (inflictor is "PistolPuff" || inflictor is "ARPuff") && (inflictor.pos.z - pos.z >= 104))
	   damage *= 1.25;

     return super.DamageMobj(inflictor, source, Damage, mod, flags, angle);        
   }
  
  override void Die(Actor source, Actor inflictor, int dmgFlags, Name MeansOfDeath)
  {
    A_KillChildren("Fire", KILS_KILLMISSILES);
	A_GiveExp(8);
	
	super.Die(source, inflictor, dmgFlags, MeansOfDeath);
  }
