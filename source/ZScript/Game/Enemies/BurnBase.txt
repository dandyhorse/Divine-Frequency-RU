Class BurnSticker : Inventory
{
  double radx, rady, hei, inc;
  int fireCount, fireMax, lifeTime;
  
  override void PostBeginPlay()
  {
    if(owner)
	{
	  fireCount = 0;
	  fireMax = 8;
	  radx = owner.radius * 0.3;
	  rady = owner.radius * 0.6;
	  hei = owner.height * 0.6;
	  inc = hei / fireMax;
	  lifeTime = fireMax * 120;
	}
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(owner && fireCount < fireMax && GetAge() % 4 == 0)
	{
	  double ang = owner.angle;
	  double xofs = random(-radx, radx);
	  double yofs = random(-rady, rady);
	  double zofs = 8 + random(fireCount * inc, (fireCount + 1) * inc);
	  
	  Vector3 spawnPos = owner.Vec3Offset(xofs * cos(ang) + yofs * sin(ang), xofs * sin(ang) - yofs * cos(ang), zofs);
	  
	  let actr = Spawn("BurnFire1", spawnPos);
	  if(actr)
	  {
	    if(fireCount < 2)
		  actr.A_SetScale(frandom(0.3,0.35));
		else
		  actr.A_SetScale(frandom(0.4,0.5));
		  
		actr.master = owner;
		BurnFire1(actr).lockedPos = (xofs, yofs, zofs);
	  }
	  fireCount++;
	}
	
	if(owner && fireCount > 0 && GetAge() % 8 == 0)
	  owner.A_DamageSelf(10, "Fire");
	
	if(lifeTime > 0)
	  lifeTime--;
	else
	{
	  //if(owner && owner is "DF_Monster")
	    //DF_Monster(owner).heat = 0;
		
	  Destroy();
	}
  }
  
  Default
  {
    Inventory.Amount 1;
	Inventory.MaxAmount 1;
  }
}

Class BurnFire1 : Actor
{
  Vector3 lockedPos;
  int dTime;
  
  override void BeginPlay()
  {
    dTime = random(80,140);
	roll = random(-40,40);
  }
  
  override void Tick()
  {
    Super.Tick();
	
    if(master && master.health > 0)
	{
	  if(dTime > 0 && alpha < 1.0)
	    A_FadeIn(0.1);
	
	  if(dTime > 0)
	    dTime--;
	  else
	    A_FadeOut(0.05);
		
	  A_Warp(AAPTR_MASTER, lockedPos.x, lockedPos.y, lockedPos.z, master.angle, WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE);
	}
	else
	{
	  A_FadeOut(0.03);
	  Gravity = 0.07;
	  bNOINTERACTION = false;
	  bTHRUACTORS = true;
	}
  }
  
  Default
  {
    Scale 0.4;
    RenderStyle "Add";
	Alpha 0.0;
	+BRIGHT
	+FORCEXYBILLBOARD
	+NOINTERACTION
	+ROLLSPRITE
  }
  
  States
  {
    Spawn:
	  FRB1 ABCDEFGHIJKLMNOPQRSTUVWXYZ 1;
	  FRB2 ABCDEFGHIJKLMNOPQRSTUVWXYZ 1;
	  Loop;
  }
}

Class FireDripBurn : Actor
{
  int lifeTime;
  
  override void BeginPlay()
  {
    A_SetScale(frandom(0.7,1.0), frandom(0.5,0.9));
	bSPRITEFLIP = random(0,1);
	lifeTime = random(140,175);
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(lifeTime > 0)
	{
	  if(alpha < 1.0)
	    alpha += 0.05;
		
	  lifeTime--;
	}
	
	if(lifeTime == 0)
	  A_FadeOut(0.05);
  }
  
  Default
  {
    RenderStyle "Add";
	Alpha 0.0;
	+BRIGHT
	+NOINTERACTION
  }
  States
  {
    Spawn:
	  DFR1 ABCDEFGHIJKLMNOPQRSTUVWXYZ 1;
	  Loop;
  }
}
