Class BurnFire : Actor
{
  Vector3 sRPos;
  
  override void BeginPlay()
  {
    bSPRITEFLIP = random(0,1);
	scale = (frandom(0.2,0.3), frandom(0.2,0.4));
  }
  
  override void Tick()
  {
    if(master)
	  A_Warp(AAPTR_MASTER, sRPos.x, sRPos.y, sRPos.z, WARPF_ABSOLUTEANGLE|WARPF_NOCHECKPOSITION|WARPF_INTERPOLATE);
	  
	Super.Tick();
  }
  
  Default
  {
    RenderStyle "Translucent";
    Alpha 0.5;
    +NOINTERACTION
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  TNT1 A 0 NoDelay A_Jump(256, random(1,14));
	Looping:
	  BFXR ABCDEFGHIJKLMN 2 Bright;
	  Loop;
  }
}

  }
  States
  {

	  Stop;

Class BurnDeathFire : BurnDeathEmber
{
  override void PostBeginPlay()
  {
    double scale = frandompick(frandom(-0.5, -0.3), frandom(0.3, 0.5));
	
    self.Scale.X = scale;
	self.Scale.Y = abs(scale);
	
    Super.PostBeginPlay();
  }
  Default
  {
    RenderStyle "Translucent";
  }
  States
  {
    Spawn:
	  TNT1 A 0;
	  FTRL AAABBBCCC 1 Bright A_FadeIn(0.1);
	  FTRL DEFGH 3 Bright;
	  Stop;
  }
}
