Class BloodSquirt : Actor
{
  int rlAmt;
  
  override void BeginPlay()
  {
    rlAmt = frandompick(-22.5, 22.5);
  }
  
  Default
  {
	Scale 0.3;
	Gravity 0.5;
	+FORCEXYBILLBOARD
	+MISSILE
	+ROLLSPRITE
	+THRUACTORS
  }
  States
  {
    Spawn:
	  SQRT A 1
	  {
	    A_SpawnItemEx("SquirtTrail");
		A_SetRoll(roll + rlAmt, SPF_INTERPOLATE);
	  }
	  Wait;
	Crash:
	Death:
	  TNT1 A 1 
	  {
	    if(pos.z - floorz == 0)
		{
	      string actr = String.Format("SmallBloodSplat%d", random(1,5));
		  A_SpawnItemEx(actr);
		}
		A_StartSound("Blood/Splat", CHAN_AUTO);
		int ang = random(0,359);
		for(int i = 0; i < random(3,4); i++)
		{
		  A_SpawnItemEx("BloodTrail", xvel: random(3,4), zvel: random(5,6), angle: ang);
		  ang += 90;
		}
		bSPRITEFLIP = random(0,1);
		Alpha = frandom(0.8,0.9);
		Scale = (frandom(0.7, 0.8), frandom(0.7,0.8));
		Roll = 0;
	  }
	  SMB5 ABCDEFGHIJKLMN 1;
	  Stop;
  }
}
