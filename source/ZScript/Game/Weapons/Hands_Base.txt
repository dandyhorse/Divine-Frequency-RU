Class JabPuff : BulletzPuff
{
  Default
  {
    Decal "None";
    DamageType "Physical";
    +PUFFONACTORS
	
  }
  States
  {
    Spawn:
	Melee:
    Crash:
	  TNT1 A 70 NoDelay
	  {
	    A_StartSound("PWallHit", CHAN_AUTO);
		A_QuakeEx(1,0,1,6,0,512,"DSEMPTY");
	    //A_DoSpark("PuffSpark");
	  }
	  Stop;
	XDeath:
	  TNT1 A 3
	  {
	    if(tracer && tracer.bIsMonster)
		{
		  for(int i = 0; i < random(2,3); i++)
		    A_SpawnItemEx("DFBlood", 16, frandom(-16,16), 0, frandompick(-0.5, 0.5), frandompick(-0.5, 0.5), frandompick(-0.5, 0.5));
		   
		  for(int i = 0; i < random(8,12); i++)
		    A_SpawnItemEx("BloodParticle", 12, frandom(-16,16), 0, frandom(3,5), frandom(-2,2), frandom(1,3));
		}
	  }
	  TNT1 A 3
	  {
	    if(tracer)
		{
		  tracer.A_StartSound("Jab/Impact", CHAN_AUTO, volume: 0.5, attenuation: 0.4);
		}
	  }
	  TNT1 A 1
	  {
	    if(tracer)
		{
		  tracer.A_StartSound("Jab/Blood", CHAN_AUTO, volume: 0.4, attenuation: 0.4);
		}
	  }
      Stop;
  }
}

Class Fist_Enhance : Inventory
{
  override void AttachToOwner(Actor other)
  {
    if(other)
	{
	  let inv = Hands(other.FindInventory("Hands"));
	  if(inv)
	    inv.duel_boost = true;
	}
    Super.AttachToOwner(other);
  }
  override void DetachFromOwner()
  {
    let pmo = players[consoleplayer].mo;
    if(pmo)
	{
	  let inv = Hands(pmo.FindInventory("Hands"));
	  if(inv)
	    inv.duel_boost = false;
	}
    Super.DetachFromOwner();
  }
  Default
  {
    Inventory.MaxAmount 1;
  }
}

Class Fist_TackleFX : Inventory
{
  override void AttachToOwner(Actor other)
  {
    if(other)
	  level.SetFrozen(true);
	  
    Super.AttachToOwner(other);
  }
  
  override void DoEffect()
  {
    Super.DoEffect();
	
	if(GetAge() > 3)
	{
	  level.SetFrozen(false);
	  Destroy();
	}
  }
  
  Default
  {
    Inventory.MaxAmount 1;
  }
}
