  void SwapFireMode()
  {
    if(!AltFireMode)

	else
	{
	  AltFireMode = false;

  }

  void ConditionWarn(bool totalFail = false)
  {
    let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	if(ev)
	{
	  if(totalFail)
	  {
	    ev.UpdateChat(String.Format("· \cF%s\cJ has \cRStopped Working!!", GetTag()));
	  }
	  else
	  {
	    let pmo = players[consoleplayer].mo;
		if(pmo && !DFPlayer(pmo).tip_cond)
		{
		  DFPlayer(pmo).tip_cond = true;
		  ev.PromptTutorial("Tip_Condition");
		}
		  
	    ev.UpdateChat(String.Format("· \cF%s\cJ is nearing a \cRCritical State!", GetTag()));
	  }
	}
  }
  
  void Degrade(int threshold = 0, bool can_jam = true)
  {
    if(CVar.FindCVar("modf_weapon").GetInt() == 0 || (owner && DFPlayer(owner).bDuel_Buff))
	  return;
	
    if(wep_protectShot > 0)
	{
	  wep_protectShot--;
	}
	else
	{
	  if(wep_cond <= threshold)
	  {
	    isInDanger = true;
		if(!hasWarned && can_jam)
		{
		  hasWarned = true;
		  ConditionWarn();
		}
	  }
	  wep_cond -= wep_deg;
	}
	
	if(wep_cond <= 0)
	{
	  if(can_jam)
	  {
	    ConditionWarn(true);
	    jam = true;
	  }
	  wep_cond = 0;
	}
  }
  
  void RepairGun(string itm)
  {
    //Make fun of the player because we're evil.
	if(owner)
	{
      if(owner && wep_cond == wep_condMax)
	  {
	    Log(String.Format("· %s \cJwasted a perfectly good \cFRepair Device\cJ...", owner.GetTag()));
	  }
	  else
	  {
	    int restore = 50;
	    if(itm ~== "MiniRepair")
	      restore = 25;
	  
	    if(owner && DFPlayer(owner).bRepairBoost)
	      restore *= 2;
		
	    int cond_prev = wep_cond;
	    wep_cond = clamp(wep_cond + restore, 0, wep_condMax);
	    jam = false;
	    isInDanger = false;
	    hasWarned = false;
	    Log(String.Format("· %s \cJrepairs \cF%s (%d% >> %d%).", owner.GetTag(), GetTag(), cond_prev, wep_cond));
	  }
	}
  }
  
  action void A_SpawnCasing(class<Actor> casingType, Vector3 offset, Vector3 velocity) //From ZScript Weapons Library by dodopod
  {
    offset.y *= -1;
	
    Vector2 xz = RotateVector((offset.x, offset.z), -pitch);
    offset.x = xz.x;
    offset.z = xz.y;
    offset.xy = RotateVector(offset.xy, angle);
    offset.xy += pos.xy;
    offset.z += player.viewZ;

    Actor casing = Spawn(casingType, offset);
	if(casing)
	{
	  double xvel = velocity.x * cos(angle) + velocity.y  * sin(angle);
	  double yvel = velocity.x * sin(angle) - velocity.y  * cos(angle);
	  casing.vel = (xvel, yvel, velocity.z);
	}
  }
  
  action void A_Reload(String ammoLoadedName, String AmmoName, bool plusOne)
  {
	let wLoaded = FindInventory(ammoLoadedName);
	let wAmmo = FindInventory(AmmoName);
	
	if(wLoaded && wAmmo)
	{
	  int req = wLoaded.MaxAmount;
	  
	  if(plusOne && wLoaded.Amount == 0)
	  {
	    req = wLoaded.MaxAmount - 1;
		plusOne = false; //counter intuitive way to check for stuff. but u cant stop me lol
	  }
	  else
	  {
	    req = wLoaded.MaxAmount - wLoaded.Amount;
		plusOne = true;
	  }
	  
	  if(wAmmo.Amount > req)
	  {
	    if(plusOne)
	      wLoaded.Amount = wLoaded.MaxAmount;
		else
		  wLoaded.Amount = wLoaded.MaxAmount - 1;
		  
		wAmmo.Amount -= req;
	  }
	  else
	  {
	    req = wAmmo.Amount;
		wLoaded.Amount += req;
		wAmmo.Amount = 0;
	  }
	}
  }
  
  action state A_CheckMax(String Loaded){
    let wLoaded = FindInventory(Loaded);
	if(wLoaded != null && wLoaded.Amount == wLoaded.MaxAmount)
	{
	  return A_Jump(256, "Empty");
	}
	return ResolveState(null);
  }
  
  int A_GetRaiseTime(int baseTime, int statMult)
  {
    int time = baseTime - (DFPlayer(owner).Wit * statMult);
	return time;
  }
  
  int GetMaxAmmoCount(String ammoType){
    let inv = FindInventory(ammoType);
	if(inv != null)
	{
	  return inv.MaxAmount;
	}
	return 0;
  }

Class Psi_HG : Inventory
{
  override void AttachToOwner(Actor other)
  {
    Super.AttachToOwner(other);
	
	if(other && other is "DFPlayer")
	{
	  DFPlayer(other).bPsi_HG = true;
	  
	  let inv = other.FindInventory("Handgun");
	  if(inv)
	    Handgun(inv).hasAlt = true;
		
	  if(other && !DFPlayer(other).tip_psi)
	  {
	    DFPlayer(other).tip_psi = true;
	    let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
	    if(ev)
	      ev.PromptTutorial("Psi");
	  }
	}
  }
  Default
  {
	-INVENTORY.INVBAR
	+INVENTORY.KEEPDEPLETED
  }
}

Class Repair_Script : Actor
{
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	let pmo = players[consoleplayer].mo;
    if(pmo)
	{
      let wpn = Handgun(pmo.FindInventory("Handgun"));
	  if(wpn)
	  {
	    wpn.jam = false;
	    wpn.isInDanger = false;
	    wpn.hasWarned = false;
		wpn.wep_cond = 50;
	  }
	}
	Destroy();
  }
  Default
  {
    +NOINTERACTION
  }
}
