Class BulletzPuff : BulletPuff 
{
  int rollAmt;
  
  action void A_DoSpark(String type)
  {
    for(int i = 0; i < 7; i++)
	  A_SpawnItemEx(type ,0,0,0,frandom(-4,4),frandom(-4,4),frandom(2,8),random(0,359),0,64);
  }
  
  Default
  {
	RenderStyle "Add";
    Decal "BulletzDecal";
	+FORCEXYBILLBOARD
	+HITTRACER
    +PUFFONACTORS
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  TNT1 A 0 {self.Scale.X = randompick(-1, 1);}
      PUFD ABCDE 2 Bright;
	  Stop;
	XDeath:
	  TNT1 A 1;
	  Stop;
	Sparks:
	  TNT1 A random(0,2)
	  {
		A_SetScale(frandom(0.8,1.0));
		roll = random(0,359);
		bXFLIP = random(0,1);
		bYFLIP = random(0,1);
		rollAmt = random(-6,6);
		
		return A_Jump(256, "Spark1", "Spark2");
	  }
	Spark1:
	  SPRK AAABCDEFGH 1 Bright A_SetRoll(roll + rollAmt, SPF_INTERPOLATE);
	  Stop;
	Spark2:
	  SPRK IIIJKLMNOPQR 1 Bright A_SetRoll(roll + rollAmt, SPF_INTERPOLATE);
	  Stop;
  }
}

Class NormalSmoke : Actor
{
  bool left;
  override void PostBeginPlay()
  {
    Super.PostBeginPlay();
	left = random(0,1);
    bSPRITEFLIP = random(0,1);
	roll = random(0,359);
	scale.x = frandom(1.75, 2.5);
	scale.y = scale.x * 0.83;
  }
  Default
  {
    RenderStyle "Shaded";
	StencilColor "3d3d3d";
	Alpha 0.0;
	+BRIGHT
	+FORCEXYBILLBOARD
    +NOINTERACTION
	+ROLLSPRITE
  }
  States
  {
    Spawn:
	  TNT1 A 0;
	  TNT1 A 0
	  {
	    name spr = String.Format("SGV%dA0", random(1,5));
	    sprite = GetSpriteIndex(spr);
	  }
	  "####" AAAAA 1 { alpha += 0.02; }
	  "####" A 1
	  {
		A_FadeOut(0.001);
	    scale.x += 0.05;
		scale.y = 0.83 * scale.x;
		roll += left ? 0.5 : -0.5;
	  }
	  Wait;
	Dummy:
	  SGV1 A -1;
	  SGV2 A -1;
	  SGV3 A -1;
	  SGV4 A -1;
	  SGV5 A -1;
	  Stop;
  }
}
