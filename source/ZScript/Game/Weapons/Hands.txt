Class Hands : DF_Weapon
{
  bool hasAttacked, elbowHit, AllowFastUP, no_stamina_raise;
  
  //Default melee
  int melee_brawl_window, melee_cycle_delay;
  bool melee_cycle;
  
  //Charged melee
  double charge_zoom, charge_mult;
  double charge_layer_scale, charge_right_x, charge_right_y, charge_left_x, charge_left_y;
  bool charge_hit_confirm;
  int charge_drain_time;
  
  //Blocking mechanics
  int shake_time, block_power_charge;
  bool block_hold, block_buff;
  
  //Parry-Riposte
  bool perfect_start, riposte_land;
  int perfect_time;
  double riposte_ang_turn;
  
  //Punched Data
  Actor taggedActr, executeActr, focusActr;
  bool hit_land;
  bool headbutt_land;
  
  //YEEEEAAAAHHHHH BOi
  bool duel_boost;
  
  //misc- ???
  int elbowTime;
  bool restrict_;
  
  override void AttachToOwner(Actor other)
  {
    if(other)
	  DFPlayer(other).wep1 = true;

	Super.AttachToOwner(other);
  }
  
  override void BeginPlay()
  {
    Super.BeginPlay();
	
	wep_order = 0;
	wep_name = "Fists";
	wep_info = "$Hand_Info";
	wep_ability = "$Hand_Ability";
	wep_quirk = "$Hand_Quirk";
	wep_dmgType = "Physical";
	wep_condMax = 0;
	wep_no_alt = true;
	
	charge_zoom = 1.0;
  }
  
  override void DoEffect()
  {
    Super.DoEffect();
	
	//console.printf("%d", melee_brawl_window);

	if(shake_time > 1)
	{
	  owner.A_OverlayOffset(3, -50 + random(-2,2), 90 + random(0,10), WOF_INTERPOLATE);
	  owner.A_OverlayOffset(2, 10 + random(-2,2), 54 + random(0,10), WOF_INTERPOLATE);
	  shake_time--;
	}
	
	if(perfect_start && owner && (owner.GetPlayerInput(0) & BT_ALTATTACK))
	{
	  if(perfect_time < 10)
	    perfect_time++;
	}
	else
	  perfect_time = 0;
	  
	if(charge_mult > 0.0 && charge_zoom == 1.0) //Building from standard melee - but implies not charging
	{
	  if(charge_drain_time > 9)
	  {
	    charge_drain_time = 0;
		charge_mult -= 0.01;
	  }
	  charge_drain_time++;
	}
	  
	if(melee_brawl_window > 0)
	  melee_brawl_window--;
	else
	  melee_cycle_delay = 9;
	
	//console.printf("%d", fire_hold_time);
	///console.printf("Damage Buff: %2f   Damage Time: %d", dmg_buff, dmg_buff_time);
  }
  
  void HitBuff(Actor hitactr)
  {
    if(DFPlayer(owner).bStrOrigin && hitactr.Health <= 0)
	  Heal();
    
	if(duel_boost)
	  DFPlayer(owner).DoHeal(15);
  }
  
  void HitBlocked(int stamina_drain)
  {
	shake_time = 8;
    block_power_charge += 25;
	if(block_power_charge >= 100)
	  block_power_charge = 100;
	  
	if(owner && CVar.FindCVar("modf_stamina").GetBool())
	  owner.TakeInventory("Stamina", stamina_drain);
  }
  
  void Heal()
  {
    DFPlayer(owner).DoHeal(50);
	Log(String.Format("· %s \cJactivates the \cFChain of Life.", owner.GetTag()));
	Log("\cD+50 Health.");
	
	if(DFPlayer(owner).bStrOrigin_Buff)
	{
	  DFPlayer(owner).RadHeal(20);
	  DFPlayer(owner).ToxHeal(20);
	  Log("\cD-20 Radiation.");
	  Log("\cD-20 Toxin.");
	}
  }
  
  action void Melee_Default()
  {
	double dmg = 20 + (20 * (DFPlayer(self).Str + DFPlayer(self).isDrunk));
	int stm = 50 - (DFPlayer(self).Str * 5);
	
	if(CVar.FindCVar("modf_stamina").GetBool())
	  TakeInventory("Stamina", stm);
	  
	FTranslatedLineTarget t;
	LineAttack(angle, 156, pitch, dmg, "Pen", "MeleePuff", LAF_ISMELEEATTACK, t);
	if(t.linetarget && t.linetarget.bIsMonster)
	{
	  A_QuakeEx(2, 4, 2, 10, 0, 128, "DSEMPTY");
	  A_StartSound("HardHook", CHAN_AUTO);
	  invoker.hit_land = true;
	  invoker.HitBuff(t.linetarget);
	  invoker.charge_mult += 0.5;
	  
	  if(invoker.charge_mult > 2.0)
	    invoker.charge_mult = 2.0;
		
	  if(!t.linetarget.bDontThrust)
	    t.linetarget.Thrust(16, angle);
	  
	  if(DFPlayer(self).bHigh_Str && invoker.melee_cycle_delay > 0)
	    invoker.melee_cycle_delay--;
	}
  }
  
  action void Melee_Charge()
  {
    double dmg = 80 + (20 * (DFPlayer(self).Str + DFPlayer(self).isDrunk));
	dmg *= invoker.charge_mult;
	FTranslatedLineTarget t;
	LineAttack(angle, 192, pitch, int(dmg), "Pen", "SoulPunchPuff", LAF_ISMELEEATTACK, t);
	if(t.linetarget && t.linetarget.bIsMonster)
	{
	  invoker.charge_hit_confirm = true;
	  angle = t.angleFromSource;
	  invoker.HitBuff(t.linetarget);
	  A_QuakeEx(2, 4, 2, 10, 0, 128, "DSEMPTY");
	  A_StartSound("HardCros", CHAN_AUTO);
	  if(t.linetarget && t.linetarget is "DF_Boss" && t.linetarget.Health > 0)
	  {
        if(DF_Boss(t.linetarget).parry_time > 0)
		{
		  DF_Boss(t.linetarget).staggerTrigger = 1;
		  DF_Boss(t.linetarget).disrespect++;
		  DF_Boss(t.linetarget).boss_fisted = true;
		  invoker.A_TryCounter(t.linetarget);
		}
	  }
	  else if(t.linetarget && t.linetarget is "DF_Monster")
	  {
	    if(DF_Monster(t.linetarget).parry_time > 0)
	      t.linetarget.DamageMobj(self, self, 10, "Shotgib", DMG_PLAYERATTACK);
		else if(DF_Monster(t.linetarget).Armored && dmg >= 150)
	      t.linetarget.DamageMobj(self, self, 10, "ArmorBreak", DMG_PLAYERATTACK);
	  }
	  if(!t.linetarget.bDontThrust)
	    t.linetarget.Thrust(50, angle);
	}
  }
  
  action void Melee_Retaliate()
  {
    int dmg = 150;
	if(DFPlayer(self).bHigh_Str)
	  dmg = 300;
	
	int stm = 250 - (DFPlayer(self).Str * 20 );
	
	if(CVar.FindCVar("modf_stamina").GetBool())
	  TakeInventory("Stamina", stm);
	  
	A_QuakeEx(2, 4, 2, 10, 0, 128, "DSEMPTY");
	//invoker.charge_mult = 0.0;
	invoker.block_power_charge = 0;
	//UHHHGHGHGHGUHGUHGUIRN RKJ NKNEKRTJNEJTKLN
    BlockThingsIterator it = BlockThingsIterator.Create(self, sqrt(512 * 512 + 128)); //front + side
    while (it.Next())
	{
      Actor mo = it.thing;
	  
      if(mo == self)
	    continue;
      
	  if(!mo.bShootable)
	    continue;
		
      if(mo.Health <= 0)
	    continue;

      double dx = mo.pos.x - pos.x;
      double dy = mo.pos.y - pos.y;
      double dz = mo.pos.z - pos.z;
      double fwd = dx * cos(angle) + dy * sin(angle);
      double sid = -dx * sin(angle) + dy * cos(angle);
	  
	  //behind or too far front
	  if(fwd < 0 || fwd > 512)
	    continue;
		
	  //too far left or right	
	  if(abs(sid) > 128)
	    continue; 
		
      //too high up      
	  if(dz < -mo.height || dz > height)
	    continue;
		
	  //Can you ever see it?
	  if(!self.CheckSight(mo, SF_IGNOREVISIBILITY))
	    continue;
	  
	  if(!mo.bDontThrust)
	    mo.Thrust(40, angle);
		
	  if(mo is "DF_Monster" && DF_Monster(mo).parry_time > 0)
	    mo.DamageMobj(self, self, 10, "Shotgib", DMG_PLAYERATTACK);
		
	  if(mo is "DF_Boss")
	  {
        if(DF_Boss(mo).parry_time > 0)
		{
		  DF_Boss(mo).staggerTrigger = 1;
		  DF_Boss(mo).disrespect++;
		  DF_Boss(mo).boss_fisted = true;
		  invoker.A_TryCounter(mo);
		}
	  }
		
      mo.DamageMobj(self, self, dmg, "Pen", DMG_PLAYERATTACK);
	  invoker.HitBuff(mo);
	}
	  
	let splash = Spawn("MineExplosion2", pos + (0,0,1));
	if(splash)
	  splash.A_SetScale(4.0);
	
	for(int i = 0; i < 20; i++)
	{
	  bool left = random(0,1);
	  A_SpawnParticleEx("888888", TexMan.CheckForTexture(String.Format("SMOKE0%d", random(1,5))), STYLE_Translucent, SPF_RELATIVE|SPF_ROLL, random(35,65),
	  random(224,256), 0, random(-64,400), random(-64,64), random(4,64), frandom(8.0,15.0), frandom(-3.0,-3.0), frandom(-3.0,0.0),
	  accelz: -0.001, startalphaf: 0.4, sizestep: 4, startroll: random(0,359), rollvel: Left ? 0.5 : -0.5, rollacc: Left ? 0.02 : -0.02);
	  A_SpawnParticleEx("888888", TexMan.CheckForTexture(String.Format("SMOKE0%d", random(1,5))), STYLE_Translucent, SPF_RELATIVE|SPF_ROLL, random(35,65),
	  random(224,256), 0, random(-64,400), random(-128,-88), random(4,64), frandom(8.0,15.0), frandom(-12.0,-8.0), frandom(-3.0,0.0),
	  accelz: -0.001, startalphaf: 0.4, sizestep: 4, startroll: random(0,359), rollvel: Left ? 0.5 : -0.5, rollacc: Left ? 0.02 : -0.02);
	  A_SpawnParticleEx("888888", TexMan.CheckForTexture(String.Format("SMOKE0%d", random(1,5))), STYLE_Translucent, SPF_RELATIVE|SPF_ROLL, random(35,65),
	  random(224,256), 0, random(-64,400), random(88,128), random(4,64), frandom(8.0,15.0), frandom(8.0,12.0), frandom(-3.0,0.0),
	  accelz: -0.001, startalphaf: 0.4, sizestep: 4, startroll: random(0,359), rollvel: Left ? 0.5 : -0.5, rollacc: Left ? 0.02 : -0.02);
	}
  }
  
  action void Melee_Riposte(Actor hitactr)
  {
    int dmg = 250 + (50 * (DFPlayer(self).Str + DFPlayer(self).isDrunk));
	FTranslatedLineTarget t;
	if(hitactr is "DF_Boss")
	{
	  DFPlayer(self).DoHeal(20);
	  let ev = ShaderHandler(EventHandler.Find("ShaderHandler"));
	  if(ev)
	    ev.SetParry(1.0, 0.04, 0.04, 15); 
	  
	  DF_Boss(hitactr).boss_StaggerBlock -= 13;
	  if(DF_Boss(hitactr).boss_StaggerBlock <= 0)
	    DF_Boss(hitactr).staggerTrigger = 1;
	}
	A_GiveInventory("ZoomShaderHit_Hand", 1);
	A_QuakeEx(3, 4, 7, 8, 0, 128, "DSEMPTY");
	A_StartSound("UpperCut", CHAN_AUTO, volume: 1.0, attenuation: 0.0);
	A_SetBlend("Orange", 0.15, 35);
	angle = AngleTo(hitactr);
	hitactr.DamageMobj(self, self, dmg, "Pen", DMG_PLAYERATTACK);
	hitactr.Thrust(18, angle);
	invoker.HitBuff(hitactr);
	invoker.riposte_land = true;
	if(DFPlayer(self).bHigh_Str)
	  invoker.block_power_charge = 100;
	else
	  invoker.block_power_charge += 25;
  }
  
  action void Melee_Headbutt(Actor hitactr)
  {
    invoker.headbutt_land = true;
    int dmg = 50 + (25 * (DFPlayer(self).Str + DFPlayer(self).isDrunk));
	FTranslatedLineTarget t;
	if(hitactr is "DF_Boss")
	{
	  if(DF_Boss(hitactr).parry_time > 0)
	  {
	    DF_Boss(hitactr).staggerTrigger = 1;
		DF_Boss(hitactr).disrespect++;
		DF_Boss(hitactr).boss_fisted = true;
		invoker.A_TryCounter(hitactr);
	  }
	}
	else if(hitactr is "DF_Monster")
	{
	  if(DF_Monster(hitactr).parry_time > 0)
	    hitactr.DamageMobj(self, self, 10, "Shotgib", DMG_PLAYERATTACK);
	  else if(DF_Monster(hitactr).Armored)
	    hitactr.DamageMobj(self, self, 10, "ArmorBreak", DMG_PLAYERATTACK);
	}
	
	if(Health == 1 || Health > 20)
	  A_DamageSelf(20, "Drowning", DMSS_NOPROTECT|DMSS_NOFACTOR);
	else
	  A_DamageSelf(Health - 1, "Drowning", DMSS_NOPROTECT|DMSS_NOFACTOR);
	   
	A_GiveInventory("Fist_TackleFX", 1);
	A_QuakeEx(3, 4, 7, 13, 0, 128, "DSEMPTY");
	A_StartSound("MA_Imp2", CHAN_AUTO, volume: 1.0);
	A_Stop();
	hitactr.DamageMobj(self, self, dmg, "Pen", DMG_PLAYERATTACK);
	hitactr.DamageMobj(self, self, 10, "Daze", DMG_PLAYERATTACK);
	if(!hitactr.bDontThrust)
	  hitactr.Thrust(17, angle);
	invoker.HitBuff(hitactr);
	invoker.shake_time = 8;
	invoker.block_power_charge += 25;
	invoker.Log(String.Format("· %s \cJsuffers \cRDamage (20)\cJ from headbutt.", self.GetTag()));
	if(invoker.block_power_charge >= 100)
	  invoker.block_power_charge = 100;
  }
  
  action void A_TryExecuteLeft()
  {
    FTranslatedLineTarget t;
    LineAttack(angle, 256, pitch, 0, "LeftHook", "BulletPuff", LAF_ISMELEEATTACK, t);
	if(t.linetarget && t.linetarget.bIsMonster)
	{
	  A_StartSound("LeftHook", CHAN_AUTO);
	  invoker.executeActr = t.linetarget;
	  invoker.HitBuff(t.linetarget);
	}
  }
  
  action void A_TryCounter(Actor actr)
  {
    actr.A_SpawnItemEx("CounterImpact", 10, 0, 0.5 * actr.height);
	actr.DamageMobj(self, self, 10, "CounterAttack");
	A_StartSound("Counter1", CHAN_AUTO);
	A_StartSound("Counter2", CHAN_AUTO);
	A_QuakeEx(2, 4, 2, 10, 0, 128, "DSEMPTY");
  }
  
  action void A_TryPunish()
  {
    FTranslatedLineTarget t;
    LineAttack(angle, 256, pitch, 250, "Gauntlet", "CounterImpact", LAF_ISMELEEATTACK, t);
	
	if(t.linetarget && t.linetarget.bIsMonster)
	{
	  A_QuakeEx(2, 4, 2, 10, 0, 128, "DSEMPTY");
	  A_StartSound("Fist_CR", CHAN_AUTO);
	}
  }
  
  action void A_TryElbow()
  {
    FTranslatedLineTarget t;
    LineAttack(angle, 256, pitch, 250, "Gauntlet", "CounterImpact", LAF_ISMELEEATTACK, t);
	if(t.linetarget && t.linetarget.bIsMonster)
	{
	  invoker.elbowHit = true;
	  A_QuakeEx(2, 4, 2, 10, 0, 128, "DSEMPTY");
	  A_StartSound("Fist_CR", CHAN_AUTO);
	  invoker.HitBuff(t.linetarget);
	}
  }

  {
    if(CountInv("Stamina") > 0)// && elbowTime > 0 )

  }

	+WEAPON.NOAUTOAIM

	Weapon.BobRangeY 0.2;
	Weapon.BobSpeed 0.7;
	Weapon.BobStyle "Inverse";
  }
  States
  {
    Select:
	  TNT1 A 0
	  {
	    A_SetCrosshair(83);
	    invoker.bIsRaising = true;
	    invoker.raiseTime = invoker.A_GetRaiseTime(10, 2);
	  }
	  TNT1 A 0 A_Raise;
	  Wait;
	Deselect:
      TNT1 A 0
	  {
	    DFPlayer(self).fist_fighting = false;
		invoker.hit_land = false;
		A_Overlay(2);
		A_Overlay(3);
		
	    if(DFPlayer(self).bWitOrigin_Buff || invoker.bIsRaising)
	      return invoker.ResolveState("FastDeselect");

	    return invoker.ResolveState(null);
	  }
	  MLSL FEDCBA 1;
	  TNT1 A 0 A_Lower;
	  Wait;
	Ready:
	  TNT1 A 1
	  {
	    A_WeaponReady(WRF_NOFIRE);
	    if(DFPlayer(self).bWitOrigin_Buff)
	      return invoker.ResolveState("FastRaise");
	    else if(invoker.raiseTime <= 0)
	      return invoker.ResolveState("WeaponRaise");
	    else
	      invoker.raiseTime--;
	    return ResolveState(null);
	  }
	  Wait;
    FastRaise:
      TNT1 A 0
	  {
	    A_Overlay(2);
		A_Overlay(3);
	    invoker.bIsRaising = false;
	  }
	  Goto WeaponRaise+4;
	WeaponRaise:
	  TNT1 A 0
	  {
	    A_Overlay(2);
		A_Overlay(3);
	    invoker.bIsRaising = false;
	  }
	  MLSL ABCDEF 1;
	  TNT1 A 0 { invoker.no_stamina_raise = false; }
	  Goto Idle;
	Idle:
	  TNT1 A 0
	  {
	    A_Overlay(2, "LeftIdle");
		A_Overlay(3, "RightIdle");
		invoker.shake_time = 0;
		invoker.elbowHit = false;
		DFPlayer(self).fist_fighting = true;
	  }
	  TNT1 A 1 A_WeaponReady(WRF_ALLOWRELOAD);
      Wait;
	LeftIdle:
	  MLI1 B 6;
	  MLI1 CDEF 3;
	  MLI1 GHI 2;
	  MLI1 J 12;
	  MLI1 IHGFE 3;
	  MLI1 DCB 2;
	  MLI1 A 15;
	  Loop;
	RightIdle:
	  MLI2 B 6;
	  MLI2 CDEF 3;
	  MLI2 GHI 2;
	  MLI2 J 12;
	  MLI2 IHGFE 3;
	  MLI2 DCB 2;
	  MLI2 A 15;
	  Loop;
	///////////////LFET HAND STUFF ************************
	/////////////////
	////////////////
    Fire:
      TNT1 A 0
	  {
	    /////////// Priority:
		/////////// No stamina (can't do anything)
	    if(CountInv("Stamina") <= 0)
		{
		  let ev = GenericScreenStuff(EventHandler.Find("GenericScreenStuff"));
		  if(ev)
		  {
			if(ev.warn_staminaF <= 0.0)
			  ev.warn_staminaF = 1.5;
		  }
		  
		  if(invoker.no_stamina_raise)
		    return ResolveState("WeaponRaise"); //raise hands
		  else
		    return ResolveState("Empty"); //dont fuck with the overlays
		}
		
		if(invoker.block_hold)
		  return ResolveState("Headbutt");
		  
		A_ClearRefire();
		/////////// Priority:
		/////////// Check for easy kills - Execution
		FLineTraceData RemoteRay;
		if(DFPlayer(self).bMax_Str)
		{
		  bool kill_search = LineTrace(angle, 220, pitch, offsetz: height-12, data: RemoteRay);
		  if(kill_search && RemoteRay.HitType == TRACE_HitActor && RemoteRay.HitActor.bIsMonster)
		  {
		    invoker.taggedActr = RemoteRay.HitActor;
		    //if(!RemoteRay.HitActor.bBoss && RemoteRay.HitActor.Health <= 0.4 * RemoteRay.HitActor.SpawnHealth())
			  return ResolveState("ExecuteJab");
		  }
		}
		/////////// Priority:
		/////////// Duel Counter
		for(int i = -128; i < 128; i += 32)
	    {
          bool hit = LineTrace(angle, 260, pitch, offsetz: height-12, offsetside: i, data: RemoteRay);
          if(hit && RemoteRay.HitType == TRACE_HitActor)
		  {
		    //Boss enemies
		    if(RemoteRay.HitActor && RemoteRay.HitActor is "DF_Boss" && RemoteRay.HitActor.Health > 0 && DFPlayer(self).bBase_Str)
			{
              if(DF_Boss(RemoteRay.Hitactor).parry_time > 0)
			  {
			    DF_Boss(RemoteRay.HitActor).staggerTrigger = 1;
				DF_Boss(RemoteRay.HitActor).disrespect++;
				{
				  DF_Boss(RemoteRay.Hitactor).boss_fisted = true;
				  A_TryCounter(RemoteRay.HitActor);
			      return ResolveState("CounterAttack");
				}
			  }
			  else if(DF_Boss(RemoteRay.Hitactor).boss_staggered)
			    return ResolveState("CounterPunish");
			  else
			    return ResolveState(null);
			}
		  }
	    }
	    return ResolveState(null);
	  }
	  /////////////
	  //Priority Resolved - Proceed as usual
	  TNT1 A 0
	  {
	    //No Strength - Do default stuff
	    if(!DFPlayer(self).bBase_Str || invoker.restrict_)
		  return ResolveState("Fire_Default");
		else if(invoker.restrict_)
		  return ResolveState("Fire_Default");
		  
		return ResolveState(null);
	  }
	//////////////
	///////////// REGULAR SWINGS
	Base_Attack:
	  TNT1 A 0
	  {
		invoker.no_stamina_raise = true;
		A_ClearRefire();
		A_Overlay(2);
		A_Overlay(3);

		  //Allow the right hand to come through for sake of continuity -

		  else
		    invoker.melee_cycle = false;
			
		  return ResolveState("Base_Attack2");
		}
		else
		{
		  invoker.melee_cycle = true;
		  if(invoker.melee_brawl_window > 0)
		    return ResolveState("Base_Attack_Swing");
		}
		
		return ResolveState(null);

	Base_Attack_Swing:
	  TNT1 A 0
	  {
	    if(invoker.melee_brawl_window > 0 && !invoker.duel_boost)

		  A_SetTics(invoker.melee_cycle_delay);

	    invoker.melee_brawl_window = 45; 
	  }
	  TNT1 AAAA 1 A_WeaponReady(WRF_NOSWITCH|WRF_NOPRIMARY|WRF_ALLOWRELOAD);
      TNT1 A 0 Melee_Default();
	  TNT1 A 0
	  {
	    if(mcam_enable)
		{
	      A_SetRoll(roll + 4, SPF_INTERPOLATE); 
	      A_ZoomFactor(1.1);
		}
	  }
	  MLEM AB 1
