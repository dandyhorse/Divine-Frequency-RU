Class Flare : Actor
{
  double alp;
  int zofs;
  
  void Init(Actor ptr, double heightOffset, int flareRad, double opacity)
  {
    master = ptr;
	zofs = heightOffset * master.scale.y;
	scale = (flareRad/32.0, flareRad/32.0);
	alpha = opacity;
	alp = alpha;
	
	if(master && master.bFloatBob)
	{
	  bFloatBob = true;
	  FloatBobPhase = master.FloatBobPhase;
	  FloatBobStrength = master.FloatBobStrength;
	}
	
	SetOrigin(master.pos + (0, 0, zOfs), false);
  }
  
  override void Tick()
  {
    Super.Tick();
	
	if(master)
	  vel = master.vel;
	else
	  Destroy();
  }
  
  action void Flicker(double alphaRange)
  {
    alpha = frandom(alphaRange * invoker.alp, invoker.alp);
  }
	
  Default
  {
	RenderStyle "Add";
	+BRIGHT
	+FORCEXYBILLBOARD
	+NOINTERACTION
  }
}
